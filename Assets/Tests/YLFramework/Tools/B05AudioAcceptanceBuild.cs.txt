        private static void VerifyProductAssets(JObject install)
        {
            var manifestPath = Path.Combine(Delivery, Package + ".files.json");
            if (Sha(manifestPath) != (string)install["PackageFilesSha256"])
            {
                throw new InvalidDataException("Product manifest changed.");
            }
            var manifest = JArray.Parse(File.ReadAllText(manifestPath));
            var allowed = new HashSet<string>(StringComparer.Ordinal);
            var parents = new HashSet<string>(StringComparer.Ordinal);
            foreach (var row in manifest)
            {
                var path = (string)row["Path"];
                for (var parent = path; parent.Contains("/");)
                {
                    parent = parent.Substring(0, parent.LastIndexOf('/'));
                    parents.Add(parent);
                }
                if (!Regex.IsMatch(path, "^Assets/YLFramework(?:/[^<>:\"|?*\\\\]+)?$") ||
                    Regex.IsMatch(path, "(^|/)\\.\\.?(/|$)|/Tests(?:/|$)"))
                {
                    throw new InvalidDataException("Unknown product path.");
                }
                var folder = (bool)row["Folder"];
                var omittedEmpty = folder && !Exists(path) &&
                    !manifest.Any(item => ((string)item["Path"]).StartsWith(path + "/", StringComparison.Ordinal));
                if (omittedEmpty)
                {
                    continue;
                }
                if (!allowed.Add(path + ".meta") || Sha(path + ".meta") != (string)row["MetaSha256"] ||
                    !folder && (!allowed.Add(path) || Sha(path) != (string)row["Sha256"]))
                {
                    throw new InvalidDataException("Imported product/meta differs: " + path);
                }
            }
            foreach (var path in SafeFiles(Application.dataPath))
            {
                var relative = Path.GetRelativePath(Project, path).Replace('\\', '/');
                if (!allowed.Contains(relative) && relative.EndsWith(".meta", StringComparison.Ordinal))
                {
                    var folder = relative.Substring(0, relative.Length - 5);
                    if (parents.Contains(folder) && Directory.Exists(folder))
                    {
                        var metadata = File.ReadAllText(path);
                        if (!Regex.IsMatch(metadata, "(?m)^guid: [0-9a-f]{32}\\r?$") ||
                            !Regex.IsMatch(metadata, "(?m)^folderAsset: yes\\r?$"))
                        {
                            throw new InvalidDataException("Invalid imported parent folder metadata.");
                        }
                        continue;
                    }
                }
                if (!allowed.Contains(relative) && relative != Root + ".meta" &&
                    !relative.StartsWith(Root + "/", StringComparison.Ordinal))
                {
                    throw new InvalidDataException("Unknown receiver asset/harness: " + relative);
                }
            }
        }

        private static void VerifyOpenScene(bool allowOwned)
        {
            if (SceneManager.sceneCount != 1)
            {
                throw new InvalidOperationException("Never save or close unknown/multiple scenes.");
            }
            var scene = SceneManager.GetActiveScene();
            if (!scene.IsValid() || !scene.isLoaded || scene.isDirty)
            {
                throw new InvalidOperationException("Never save or close a dirty/unknown scene.");
            }
            if (allowOwned && scene.path == ScenePath)
            {
                VerifyFixtureScene(scene);
                return;
            }
            var roots = scene.GetRootGameObjects();
            if (scene.path.Length != 0 || roots.Length != 0 &&
                (roots.Length != 2 || roots.Count(item => item.name == "Main Camera") != 1 ||
                    roots.Count(item => item.name == "Directional Light") != 1))
            {
                throw new InvalidOperationException("Only the clean empty/default CreateProject bootstrap may be replaced.");
            }
            foreach (var item in roots)
            {
                var types = item.GetComponents<Component>().Select(c => c ? c.GetType() : null).ToArray();
                var camera = item.name == "Main Camera" && types.Length == 3 && types.Contains(typeof(Transform)) &&
                    types.Contains(typeof(Camera)) && types.Contains(typeof(AudioListener)) && item.CompareTag("MainCamera");
                var light = item.name == "Directional Light" && types.Length == 2 && types.Contains(typeof(Transform)) &&
                    types.Contains(typeof(Light)) && item.GetComponent<Light>().type == LightType.Directional;
                if (item.transform.childCount != 0 || !item.activeSelf || !camera && !light)
                {
                    throw new InvalidOperationException("Unknown bootstrap content; do not save/close it.");
                }
            }
        }

        public static void Generate()
        {
            var install = Verify("generate");
            if (Exists(Owned) || Exists(Owned + ".meta"))
            {
                throw new IOException("Never adopt/overwrite generated Audio assets.");
            }
            var mixer = AssetDatabase.LoadAssetAtPath<AudioMixer>(Root + "/AudioMixerFixture.mixer");
            var music = mixer ? mixer.FindMatchingGroups("Music").Where(group => group.name == "Music").ToArray() : null;
            if (!mixer || music == null || music.Length != 1 || music[0].audioMixer != mixer ||
                !mixer.GetFloat("MusicVolume", out var volume) || Mathf.Abs(volume) > .0001f)
            {
                throw new InvalidDataException("Current engine must import the real Music group/exposed MusicVolume at 0 dB.");
            }
            var receiptRoot = Argument("-b05-receipt");
            ClaimDirectory(receiptRoot);
            WriteNew(Path.Combine(receiptRoot, "Owner.json"), Receipt("GenerateStarted", install));
            if (string.IsNullOrEmpty(AssetDatabase.CreateFolder(Root, "Owned")) ||
                string.IsNullOrEmpty(AssetDatabase.CreateFolder(Owned, "Resources")))
            {
                throw new IOException("Could not create new owned Audio folders.");
            }
            Wave(Owned + "/Resources/audio45-tone.wav", 330);
            Wave(Owned + "/Signal.wav", 440);
            Wave(Owned + "/Second.wav", 660);
            var library = ScriptableObject.CreateInstance<AudioClipLibrary>();
            library.groups.Add(new AudioClipGroup
            {
                name = "signal", selection = AudioClipSelectionMode.Uniform,
                entries = new List<AudioClipEntry>
                {
                    new AudioClipEntry { source = AudioClipSource.Resource, projectId = "yl-audio-acceptance", key = "signal" }
                }
            });
            AssetDatabase.CreateAsset(library, Owned + "/Library.asset");
            var config = ScriptableObject.CreateInstance<AudioMixerConfig>();
            using (var serialized = new SerializedObject(config))
            {
                Property(serialized, "mixer").objectReferenceValue = mixer;
                Property(serialized, "musicGroup").objectReferenceValue = music[0];
                serialized.ApplyModifiedPropertiesWithoutUndo();
            }
            AssetDatabase.CreateAsset(config, Owned + "/MixerConfig.asset");
            // Verify 已拒绝所有未知/脏场景；这里仅替换已证明的 CreateProject bootstrap。
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            library = AssetDatabase.LoadAssetAtPath<AudioClipLibrary>(Owned + "/Library.asset");
            config = AssetDatabase.LoadAssetAtPath<AudioMixerConfig>(Owned + "/MixerConfig.asset");
            if (!library || !config || !config.Mixer || !config.MusicGroup)
            {
                throw new InvalidDataException("Reload persistent audio bindings after replacing the bootstrap scene.");
            }
            var root = new GameObject("B05 Audio Acceptance", typeof(Camera), typeof(AudioListener),
                typeof(AudioGroupPlayer), typeof(AudioAcceptanceEntry));
            var entry = root.GetComponent<AudioAcceptanceEntry>();
            var groupPlayer = root.GetComponent<AudioGroupPlayer>();
            using (var serialized = new SerializedObject(groupPlayer))
            {
                Property(serialized, "library").objectReferenceValue = library;
                Property(serialized, "groupName").stringValue = "signal";
                Property(serialized, "channelName").stringValue = "b05/audio45";
                Property(serialized, "loadAsync").boolValue = true;
                Property(serialized, "autoPlay").intValue = (int)AudioGroupAutoPlay.Manual;
                Property(serialized, "options").FindPropertyRelative("Loop").boolValue = true;
                serialized.ApplyModifiedPropertiesWithoutUndo();
            }
            entry.group = groupPlayer;
            entry.signal = AssetDatabase.LoadAssetAtPath<AudioClip>(Owned + "/Signal.wav");
            entry.secondSignal = AssetDatabase.LoadAssetAtPath<AudioClip>(Owned + "/Second.wav");
            entry.mixerConfig = config;
            entry.resourceKey = "audio45-tone";
            VerifyFixtureScene(scene);
            if (!EditorSceneManager.SaveScene(scene, ScenePath))
            {
                throw new IOException("Could not save the newly owned Audio scene.");
            }
            AssetDatabase.Refresh(ImportAssetOptions.ForceSynchronousImport);
            var receipt = Receipt("GeneratedOnly", install);
            receipt["Scene"] = ScenePath;
            receipt["SceneGuid"] = AssetDatabase.AssetPathToGUID(ScenePath);
            receipt["SceneSha256"] = Sha(ScenePath);
            receipt["SceneMetaSha256"] = Sha(ScenePath + ".meta");
            receipt["MusicGroup"] = music[0].name;
            receipt["MusicVolumeDb"] = volume;
            receipt["Files"] = AssetInventory();
            WriteNew(Path.Combine(receiptRoot, "Generate.json"), receipt);
            Debug.Log("B05_AUDIO_GENERATED_ONLY RuntimePassed=false InputInjected=false SourceProjectTested=false");
        }

                (string)generation["PackageSha256"] != PackageSha ||
                (string)generation["InstallReceiptSha256"] != Sha(Root + "/InstallReceipt.json") ||

            var receiptRoot = Argument("-b05-receipt");
            var output = Argument("-b05-player-root");

            ClaimDirectory(output);
            WriteNew(Path.Combine(receiptRoot, "Owner.json"), Receipt("BuildStarted", install));
            WriteNew(Path.Combine(output, "Owner.json"), Receipt("PlayerOutputReserved", install));
            BuildReport report;
            var backend = "";

            try

                backend = PlayerSettings.GetScriptingBackend(NamedBuildTarget.Standalone).ToString();
                stripping = PlayerSettings.GetManagedStrippingLevel(NamedBuildTarget.Standalone).ToString();
                if (backend != "Mono2x" || stripping != "Disabled" || EditorUserBuildSettings.development)
                {
                    throw new InvalidOperationException("Actual Mono/Disabled/non-Development build settings differ.");
                }
                report = BuildPipeline.BuildPlayer(new BuildPlayerOptions
                {
                    scenes = new[] { ScenePath }, locationPathName = Path.Combine(output, "B05Audio.exe"),
                    target = BuildTarget.StandaloneWindows64, targetGroup = BuildTargetGroup.Standalone,
                    subtarget = (int)StandaloneBuildSubtarget.Player, options = BuildOptions.None,
                    extraScriptingDefines = Array.Empty<string>()
                });
            }
            finally
            {
                // 与原 owned builder 一致：先恢复内存对象，再恢复本 receiver 的原始磁盘字节。
                EditorJsonUtility.FromJsonOverwrite(json, settings);
                EditorUserBuildSettings.development = development;
                EditorUserBuildSettings.allowDebugging = debugging;
                EditorUserBuildSettings.connectProfiler = profiler;
                if (EditorJsonUtility.ToJson(settings) != json)
                {
                    throw new InvalidOperationException("Exact in-memory PlayerSettings restoration failed.");
                }
                if (Sha(settingsPath) != diskHash)
                {
                    File.WriteAllBytes(settingsPath, bytes);
                }
                EditorUtility.ClearDirty(settings);
                if (Sha(settingsPath) != diskHash || !buildScenes.SequenceEqual(EditorBuildSettings.scenes.Select(s => (s.path, s.enabled))) ||
                    EditorUserBuildSettings.development != development || EditorUserBuildSettings.allowDebugging != debugging ||
                    EditorUserBuildSettings.connectProfiler != profiler)
                {
                    throw new IOException("Exact PlayerSettings/build settings restoration failed.");
                }
            }
            if (report == null)
            {
                throw new InvalidOperationException("No actual BuildReport returned.");
            }
            var assemblyPath = Path.Combine(output, "B05Audio_Data/ScriptingAssemblies.json");
            var actualNames = File.Exists(assemblyPath)
                ? ((JArray)Read(assemblyPath)["names"]).Select(item => Path.GetFileNameWithoutExtension((string)item)).ToArray()
                : Array.Empty<string>();
            var managed = SafeFiles(output).Where(path => path.EndsWith(".dll", StringComparison.OrdinalIgnoreCase) &&
                path.IndexOf(Path.DirectorySeparatorChar + "Managed" + Path.DirectorySeparatorChar, StringComparison.Ordinal) >= 0)
                .Select(Path.GetFileNameWithoutExtension).ToArray();
            var cleanAssemblies = actualNames.Contains(FixtureAssembly) && actualNames.Contains("YLFramework.Runtime") &&
                managed.Contains(FixtureAssembly) && managed.Contains("YLFramework.Runtime") &&
                actualNames.Concat(managed).All(AllowedPlayerAssembly) &&
                !SafeFiles(output).Where(path => path.EndsWith(".dll", StringComparison.OrdinalIgnoreCase))
                    .Select(Path.GetFileNameWithoutExtension).Any(ForbiddenAssembly);
            var receipt = Receipt("BuiltOnly", install);
            receipt["GenerationSha256"] = Sha(generationPath);
            receipt["Scene"] = ScenePath;
            receipt["SceneSha256"] = Sha(ScenePath);
            receipt["SceneMetaSha256"] = Sha(ScenePath + ".meta");
            receipt["Inputs"] = AssetInventory();
            receipt["Backend"] = backend;
            receipt["Stripping"] = stripping;
            receipt["Target"] = report.summary.platform.ToString();
            receipt["Development"] = (report.summary.options & BuildOptions.Development) != 0;
            receipt["Options"] = report.summary.options.ToString();
            receipt["Result"] = report.summary.result.ToString();
            receipt["Errors"] = report.summary.totalErrors;
            receipt["Warnings"] = report.summary.totalWarnings;
            receipt["BuildGuid"] = report.summary.guid.ToString();
            receipt["Executable"] = Path.Combine(output, "B05Audio.exe");
            receipt["SettingsRestored"] = true;
            receipt["PlayerSettingsBeforeSha256"] = diskHash;
            receipt["PlayerSettingsAfterSha256"] = Sha(settingsPath);
            receipt["PlayerAssemblies"] = JArray.FromObject(actualNames);
            receipt["ManagedAssemblies"] = JArray.FromObject(managed);
            receipt["PlayerAssemblyClosurePassed"] = cleanAssemblies;
            receipt["InputIdentityUnchanged"] = JToken.DeepEquals(generation["Files"], AssetInventory());
            receipt["Messages"] = JArray.FromObject(report.steps.SelectMany(step => step.messages)
                .Where(message => message.type != LogType.Log).Select(message => new { message.type, message.content }));
            receipt["PlayerFiles"] = Inventory(output, SafeFiles(output));
            WriteNew(Path.Combine(receiptRoot, "BuildReport.json"), receipt);
            if (report.summary.result != BuildResult.Succeeded || report.summary.totalErrors != 0 ||
                report.summary.options != BuildOptions.None || report.summary.platform != BuildTarget.StandaloneWindows64 ||
                report.summary.guid.ToString() == new string('0', 32) || !cleanAssemblies || !(bool)receipt["InputIdentityUnchanged"])
            {
                throw new InvalidOperationException("Build/assembly/input gates failed; preserve the immutable report/output, no DSP claim.");
            }
            Debug.Log("B05_AUDIO_BUILT_ONLY RuntimePassed=false InputInjected=false SourceProjectTested=false");
        }

        private static void VerifyFixtureScene(Scene scene)
        {
            var roots = scene.GetRootGameObjects();
            var entry = roots.Length == 1 ? roots[0].GetComponent<AudioAcceptanceEntry>() : null;
            var components = roots.Length == 1 ? roots[0].GetComponents<Component>() : Array.Empty<Component>();
            if (!entry || roots[0].transform.childCount != 0 || !entry.enabled || components.Length != 5 ||
                components.Where(c => c).Select(c => c.GetType()).Distinct().Count() != 5 ||
                components.Any(c => !c || !new[] { typeof(Transform), typeof(Camera), typeof(AudioListener),
                    typeof(AudioGroupPlayer), typeof(AudioAcceptanceEntry) }.Contains(c.GetType())) ||
                !entry.GetComponent<AudioListener>().enabled || entry.transform.position != Vector3.zero ||
                !entry.group || entry.group.gameObject != entry.gameObject || !entry.signal || !entry.secondSignal ||
                entry.signal == entry.secondSignal || !entry.mixerConfig || !entry.mixerConfig.MusicGroup ||
                entry.mixerConfig.MusicGroup.audioMixer != entry.mixerConfig.Mixer || entry.resourceKey != "audio45-tone")
            {
                throw new InvalidDataException("Only one Entry/Listener/Camera/Group is allowed; no root/UI/other Source/ResourceSystem. " +
                    "roots=" + roots.Length + "; components=" + string.Join(",", components.Select(component => component ? component.GetType().FullName : "missing")) +
                    "; children=" + (roots.Length == 1 ? roots[0].transform.childCount : -1) +
                    "; entry=" + (bool)entry + "; position=" + (entry ? entry.transform.position.ToString() : "none") +
                    "; group=" + (entry && entry.group) + "; signal=" + (entry && entry.signal) + "; second=" + (entry && entry.secondSignal) +
                    "; mixer=" + (entry && entry.mixerConfig && entry.mixerConfig.Mixer) +
                    "; music=" + (entry && entry.mixerConfig && entry.mixerConfig.MusicGroup) +
                    "; resourceKey=" + (entry ? entry.resourceKey : "none"));
            }
            using var serialized = new SerializedObject(entry.group);
            if (Property(serialized, "library").objectReferenceValue != AssetDatabase.LoadAssetAtPath<AudioClipLibrary>(Owned + "/Library.asset") ||
                Property(serialized, "groupName").stringValue != "signal" || Property(serialized, "channelName").stringValue != "b05/audio45" ||
                !Property(serialized, "loadAsync").boolValue || Property(serialized, "autoPlay").intValue != (int)AudioGroupAutoPlay.Manual ||
                !Property(serialized, "options").FindPropertyRelative("Loop").boolValue ||
                Property(serialized, "borrowedSource").objectReferenceValue || !string.IsNullOrEmpty(Property(serialized, "persistenceId").stringValue))
            {
                throw new InvalidDataException("Async/manual/loop/channel/owned-source fixture contract differs.");
            }
        }

        private static void Wave(string path, int frequency)
        {
            const int Rate = 44100;
            const int Count = Rate * 3;
            RejectLinks(Path.GetFullPath(path));
            using (var stream = new FileStream(path, FileMode.CreateNew, FileAccess.Write, FileShare.None))
            using (var writer = new BinaryWriter(stream))
            {
                writer.Write(Encoding.ASCII.GetBytes("RIFF"));
                writer.Write(36 + Count * 2);
                writer.Write(Encoding.ASCII.GetBytes("WAVEfmt "));
                writer.Write(16);
                writer.Write((short)1);
                writer.Write((short)1);
                writer.Write(Rate);
                writer.Write(Rate * 2);
                writer.Write((short)2);
                writer.Write((short)16);
                writer.Write(Encoding.ASCII.GetBytes("data"));
                writer.Write(Count * 2);
                for (var i = 0; i < Count; i++)
                {
                    writer.Write((short)(Math.Sin(2 * Math.PI * frequency * i / Rate) * 4000));
                }
            }
            AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceSynchronousImport);
            var importer = AssetImporter.GetAtPath(path) as AudioImporter;
            if (!importer)
            {
                throw new InvalidDataException("Real AudioImporter unavailable: " + path);
            }
            var settings = importer.defaultSampleSettings;
            settings.compressionFormat = AudioCompressionFormat.PCM;
            settings.loadType = AudioClipLoadType.DecompressOnLoad;
            settings.sampleRateSetting = AudioSampleRateSetting.PreserveSampleRate;
            settings.preloadAudioData = true;
            importer.defaultSampleSettings = settings;
            importer.SaveAndReimport();
            var clip = AssetDatabase.LoadAssetAtPath<AudioClip>(path);
            if (!clip || clip.channels != 1 || clip.frequency != Rate || clip.samples != Count)
            {
                throw new InvalidDataException("Current engine did not import the exact mono PCM wave: " + path);
            }
        }

        private static bool ForbiddenAssembly(string name) => Regex.IsMatch(name,
            "Editor|NUnit|TestRunner|Analyzer|CodeAnalysis|Roslyn|Tests|Acceptance", RegexOptions.IgnoreCase);

        private static bool AllowedPlayerAssembly(string name) => !ForbiddenAssembly(name) &&
            (name == "YLFramework.Runtime" || name == "YLFramework.ResourceStartup.Sample" || name == FixtureAssembly || name == "Newtonsoft.Json" ||
                name == "BouncyCastle.Cryptography" || name == "mscorlib" || name == "netstandard" || name == "System" ||
                name == "Microsoft.CSharp" || name == "UnityEngine" || name == "Unity.TextMeshPro" ||
                name.StartsWith("System.", StringComparison.Ordinal) || name.StartsWith("Mono.", StringComparison.Ordinal) ||
                name.StartsWith("UnityEngine.", StringComparison.Ordinal) ||
                Regex.IsMatch(name, "^Unity\\.InternalAPIEngineBridge\\.[0-9]+$"));

        private static SerializedProperty Property(SerializedObject target, string name) => target.FindProperty(name) ??
            throw new InvalidDataException("Real serialized field missing: " + target.targetObject.GetType().FullName + "." + name);

        private static JObject Receipt(string stage, JObject install) => new JObject
        {
            ["Schema"] = "B05Audio-v1", ["Stage"] = stage, ["Delivery"] = Delivery, ["Project"] = Project,
            ["Run"] = Argument("-b05-run"), ["OwnerSha256"] = Argument("-b05-owner-sha256"),
            ["Unity"] = Application.unityVersion, ["EditorExecutable"] = EditorApplication.applicationPath,
            ["PackageSha256"] = PackageSha, ["InstallReceiptSha256"] = Sha(Root + "/InstallReceipt.json"),
            ["SourceInputs"] = install["Inputs"].DeepClone(),
            ["RuntimePassed"] = false, ["InputInjected"] = false, ["SourceProjectTested"] = false
        };

        private static JArray AssetInventory() => Inventory(Project, SafeFiles(Path.GetFullPath(Root))
            .Append(Path.GetFullPath(Root + ".meta")));

        private static JArray Inventory(string root, IEnumerable<string> files) => new JArray(files
            .OrderBy(path => Path.GetRelativePath(root, path).Replace('\\', '/'), StringComparer.Ordinal).Select(path => new JObject
            {
                ["Path"] = Path.GetRelativePath(root, path).Replace('\\', '/'),
                ["Length"] = new FileInfo(path).Length, ["Sha256"] = Sha(path)
            }));

        private static IEnumerable<string> SafeFiles(string directory)
        {
            RejectLinks(Path.GetFullPath(directory));
            foreach (var child in Directory.GetFileSystemEntries(directory))
            {
                RejectLinks(Path.GetFullPath(child));
                if (Directory.Exists(child))
                {
                    foreach (var file in SafeFiles(child))
                    {
                        yield return file;
                    }
                }
                else
                {
                    yield return child;
                }
            }
        }

        private static void ClaimDirectory(string path)
        {
            RejectLinks(path);
            if (Exists(path))
            {
                throw new IOException("Never adopt an existing output/receipt root: " + path);
            }
            var parent = Path.GetDirectoryName(path);
            Directory.CreateDirectory(parent);
            var staging = Path.Combine(parent, ".b05-" + Guid.NewGuid().ToString("N"));
            Directory.CreateDirectory(staging);
            Directory.Move(staging, path);
        }

        private static bool False(JObject value, string key) => value[key]?.Type == JTokenType.Boolean && !(bool)value[key];
        private static bool Exists(string path) => File.Exists(path) || Directory.Exists(path);
        private static JObject Read(string path)
        {
            RejectLinks(Path.GetFullPath(path));
            return JObject.Parse(File.ReadAllText(path));
        }
        private static string Sha(string path)
        {
            RejectLinks(Path.GetFullPath(path));
            using var hash = SHA256.Create();
            using var stream = File.OpenRead(path);
            return BitConverter.ToString(hash.ComputeHash(stream)).Replace("-", "").ToLowerInvariant();
        }
        private static void WriteNew(string path, JObject value)
        {
            RejectLinks(Path.GetFullPath(path));
            using var stream = new FileStream(path + ".tmp", FileMode.CreateNew, FileAccess.Write, FileShare.None);
            var bytes = new UTF8Encoding(false).GetBytes(value.ToString(Formatting.Indented));
            stream.Write(bytes, 0, bytes.Length);
            stream.Flush(true);
            stream.Dispose();
            File.Move(path + ".tmp", path);
        }
        private static void RejectLinks(string path)
        {
            for (var cursor = path; !string.IsNullOrEmpty(cursor); cursor = Path.GetDirectoryName(cursor))
            {
                try
                {
                    if ((File.GetAttributes(cursor) & FileAttributes.ReparsePoint) != 0)
                    {
                        throw new IOException("Linked path rejected: " + cursor);
                    }
                }
                catch (FileNotFoundException) { }
                catch (DirectoryNotFoundException) { }
            }
        }
        private static string Argument(string name)
        {
            var args = Environment.GetCommandLineArgs();
            var index = Array.IndexOf(args, name);
            if (index < 0 || index + 1 >= args.Length || Array.LastIndexOf(args, name) != index ||
                args[index + 1].StartsWith("-", StringComparison.Ordinal))
            {
                throw new ArgumentException("Exactly one explicit value required: " + name);
            }
            return args[index + 1];
        }
