// Install both templates only in an audited independent receiver. No auto-run/load hooks or source-side entry.
public static class Stage9NativeInputEditor
{
    private const string Root = "Assets/Stage9NativeInput";
    private const string Owned = Root + "/Owned";
    private const string ScenePath = Owned + "/NativeInput.unity";
    private const string FullPrefab = "Assets/YLFramework/Prefabs/YLFramework.prefab";
    private const string ColorPrefab = "Assets/YLFramework/Prefabs/ColorSelector.prefab";
    private static string Project => Path.GetFullPath(Application.dataPath + "/..");

    private static JObject Verify()
    {
        RejectLinks(Project);
        if (Project.IndexOf("CGJ2024", StringComparison.OrdinalIgnoreCase) >= 0)
        {
            throw new InvalidOperationException("Never install or run native-input generation in Source CGJ2024.");
        }
        var ownerPath = Path.Combine(Project, "stage9-owner.json");
        RejectLinks(ownerPath);
        var owner = JObject.Parse(File.ReadAllText(ownerPath));
        var delivery = (string)owner["Delivery"];
        if (delivery != @"G:\Unity\U63-20261027-c63a1" || (string)owner["Purpose"] != "Stage9-independent-receiver" ||
            (string)owner["Project"] != Project || !Project.StartsWith(delivery + "\\", StringComparison.Ordinal) ||
            !Regex.IsMatch(Path.GetFileName(Project), "^(Receiver[A-Z][0-9]*|[a-z][0-9]?)$"))
        {
            throw new InvalidDataException("Exact owned receiver identity required.");
        }
        RejectLinks(delivery);
        RejectLinks(Path.Combine(delivery, "owner.json"));
        var deliveryOwner = JObject.Parse(File.ReadAllText(Path.Combine(delivery, "owner.json")));
        if ((string)deliveryOwner["purpose"] != "YLFramework-Stage9-independent-delivery" ||
            (string)deliveryOwner["root"] != delivery || string.IsNullOrEmpty((string)deliveryOwner["repository"]))
        {
            throw new InvalidDataException("Wrong delivery owner.");
        }
        var version = (string)owner["UnityVersion"];
        if ((version != "2022.3.62f3c1" && version != "6000.3.27f1") || Application.unityVersion != version ||
            (string)owner["UguiVersion"] != (version == "6000.3.27f1" ? "2.0.0" : "1.0.0") ||
            EditorApplication.isPlayingOrWillChangePlaymode || EditorApplication.isCompiling ||
            EditorUtility.scriptCompilationFailed || EditorUserBuildSettings.activeBuildTarget != BuildTarget.StandaloneWindows64)
        {
            throw new InvalidOperationException("Wrong actual Unity/version/UGUI/build-target state.");
        }
        if (Directory.Exists(Path.Combine(Project, "Assets/YLFramework/Tests")))
        {
            throw new InvalidOperationException("Native production consumers must not import a test or acceptance tree; use the Editor-only profile helper.");
        }
        var projectVersion = Path.Combine(Project, "ProjectSettings/ProjectVersion.txt");
        RejectLinks(projectVersion);
        if (!Regex.IsMatch(File.ReadAllText(projectVersion), "(?m)^m_EditorVersion: " + Regex.Escape(version) + "\\r?$"))
        {
            throw new InvalidDataException("ProjectVersion mismatch; never upgrade a receiver.");
        }
        for (var i = 0; i < EditorSceneManager.sceneCount; i++)
        {
            if (EditorSceneManager.GetSceneAt(i).isDirty)
            {
                throw new InvalidOperationException("Never save or close a foreign dirty scene.");
            }
        }
        var package = (string)owner["Package"];
        if (package == null || !Regex.IsMatch(package, "^[A-Za-z0-9][A-Za-z0-9._-]*\\.unitypackage$") ||
            Sha(Path.Combine(delivery, package)) != (string)owner["PackageSha256"])
        {
            throw new InvalidDataException("Package differs from this receiver's source owner.");
        }
        if (Directory.Exists(Path.Combine(Project, ".git")) || File.Exists(Path.Combine(Project, ".git")))
        {
            throw new InvalidDataException("A git repository/worktree is never an independent receiver.");
        }
        foreach (var path in new[] { Root, Root + "/Stage9NativeInputProbe.cs", Root + "/Editor/Stage9NativeInputEditor.cs",
            Owned, ScenePath,
            FullPrefab, ColorPrefab, "ProjectSettings/ProjectSettings.asset" })
        {
            RejectLinks(Path.Combine(Project, path));
            RejectLinks(Path.Combine(Project, path + ".meta"));
        }
        var registered = UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages();
        if (!registered.Any(p => p.name == "com.unity.ugui" && p.version == (string)owner["UguiVersion"]))
        {
            throw new InvalidDataException("Actual registered UGUI version mismatch.");
        }
        return owner;
    }

    public static void Generate()
    {
        var owner = Verify();
        if (!AssetDatabase.IsValidFolder(Root) || Directory.Exists(Owned) || File.Exists(Owned) || File.Exists(Owned + ".meta"))
        {
            throw new IOException("Install templates first; never adopt or overwrite an existing Owned fixture.");
        }
        var full = AssetDatabase.LoadAssetAtPath<GameObject>(FullPrefab);
        var color = AssetDatabase.LoadAssetAtPath<GameObject>(ColorPrefab);
        if (!full || !full.GetComponentInChildren<UIRootContext>(true) ||
            !color || !color.GetComponent<ColorSelectorWindow>() || !color.GetComponentInChildren<ColorSelector>(true) ||
            color.GetComponentsInChildren<Button>(true).Length != 1)
        {
            throw new InvalidDataException("Formal full root/ColorSelector/close Button bindings are unavailable.");
        }
        var previous = SceneManager.GetActiveScene();
        var untitled = Enumerable.Range(0, SceneManager.sceneCount).Select(SceneManager.GetSceneAt)
            .Where(item => string.IsNullOrEmpty(item.path)).ToArray();
        if (untitled.Length != 0)
        {
            if (untitled.Length != 1 || SceneManager.sceneCount != 1)
            {
                throw new InvalidOperationException("Never replace a mixed or unknown untitled scene set.");
            }
            var roots = untitled[0].GetRootGameObjects();
            if (roots.Length != 0 && (roots.Length != 2 ||
                !roots.Any(item => item.name == "Main Camera") || !roots.Any(item => item.name == "Directional Light")))
            {
                throw new InvalidOperationException("Only an empty or exact official Camera/Light bootstrap may be replaced.");
            }
            foreach (var item in roots)
            {
                var types = item.GetComponents<Component>().Select(component => component ? component.GetType() : null).ToArray();
                var camera = item.name == "Main Camera" && types.Length == 3 && types.Contains(typeof(Transform)) &&
                    types.Contains(typeof(Camera)) && types.Contains(typeof(AudioListener));
                var light = item.name == "Directional Light" && types.Length == 2 && types.Contains(typeof(Transform)) &&
                    types.Contains(typeof(Light));
                if (item.transform.childCount != 0 || !camera && !light)
                {
                    throw new InvalidOperationException("Unknown bootstrap object/component; do not save or replace it.");
                }
            }
        }
        AssetDatabase.CreateFolder(Root, "Owned");
        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,
            untitled.Length == 0 ? NewSceneMode.Additive : NewSceneMode.Single);
        try
        {
            SceneManager.SetActiveScene(scene);
            var camera = new GameObject("Owned native-input camera", typeof(Camera));
            camera.GetComponent<Camera>().clearFlags = CameraClearFlags.SolidColor;
            camera.GetComponent<Camera>().backgroundColor = new Color(0.025f, 0.04f, 0.07f);
            var probe = new GameObject("Owned native-input consumer").AddComponent<Stage9NativeInputProbe>();
            probe.fullTemplate = full;
            probe.colorPrefab = color;
            probe.receiverProject = Project;
            probe.deliveryRoot = (string)owner["Delivery"];
            probe.expectedUnity = Application.unityVersion;
            if (!EditorSceneManager.SaveScene(scene, ScenePath))
            {
                throw new IOException("Could not save the newly owned scene.");
            }
            WriteNew(Owned + "/Identity.json", new JObject
            {
                ["Schema"] = "Stage9NativeInput-v1", ["Project"] = Project, ["Unity"] = Application.unityVersion,
                ["PackageSha256"] = owner["PackageSha256"], ["Scene"] = ScenePath,
                ["SceneGuid"] = AssetDatabase.AssetPathToGUID(ScenePath), ["SceneSha256"] = Sha(ScenePath),
                ["ProbeSha256"] = Sha(Root + "/Stage9NativeInputProbe.cs"),
                ["EditorSha256"] = Sha(Root + "/Editor/Stage9NativeInputEditor.cs"),
                ["FullPrefabGuid"] = AssetDatabase.AssetPathToGUID(FullPrefab), ["FullPrefabSha256"] = Sha(FullPrefab),
                ["ColorPrefabGuid"] = AssetDatabase.AssetPathToGUID(ColorPrefab), ["ColorPrefabSha256"] = Sha(ColorPrefab)
            });
            AssetDatabase.ImportAsset(Owned + "/Identity.json", ImportAssetOptions.ForceSynchronousImport);
        }
        finally
        {
            if (previous.IsValid() && previous.isLoaded)
            {
                EditorSceneManager.CloseScene(scene, true);
                SceneManager.SetActiveScene(previous);
            }
        }
        Debug.Log("STAGE9_NATIVE_INPUT_GENERATED");
    }

    public static void BuildMono() => Build(ScriptingImplementation.Mono2x, "Mono");
    public static void BuildIl2Cpp() => Build(ScriptingImplementation.IL2CPP, "Il2Cpp");

    private static void Build(ScriptingImplementation backend, string label)
    {
        var owner = Verify();
        var run = Argument("-s9-run");
        var mode = Argument("-s9-native-mode");
        if (!Regex.IsMatch(run, "^[A-Za-z0-9-]{1,32}$") ||
            (mode != "ExpectedLegacy" && mode != "ExpectedNew" && mode != "ExpectedBoth"))
        {
            throw new ArgumentException("Use safe unique -s9-run and explicit -s9-native-mode ExpectedLegacy/New/Both.");
        }
        var generationPath = Owned + "/Identity.json";
        RejectLinks(Path.Combine(Project, generationPath));
        var generation = JObject.Parse(File.ReadAllText(generationPath));
        if ((string)generation["Schema"] != "Stage9NativeInput-v1" || (string)generation["Project"] != Project ||
            (string)generation["Unity"] != Application.unityVersion || (string)generation["PackageSha256"] != (string)owner["PackageSha256"] ||
            (string)generation["Scene"] != ScenePath || (string)generation["SceneGuid"] != AssetDatabase.AssetPathToGUID(ScenePath) ||
            (string)generation["SceneSha256"] != Sha(ScenePath) || (string)generation["ProbeSha256"] != Sha(Root + "/Stage9NativeInputProbe.cs") ||
            (string)generation["EditorSha256"] != Sha(Root + "/Editor/Stage9NativeInputEditor.cs") ||
            (string)generation["FullPrefabGuid"] != AssetDatabase.AssetPathToGUID(FullPrefab) || (string)generation["FullPrefabSha256"] != Sha(FullPrefab) ||
            (string)generation["ColorPrefabGuid"] != AssetDatabase.AssetPathToGUID(ColorPrefab) || (string)generation["ColorPrefabSha256"] != Sha(ColorPrefab))
        {
            throw new InvalidDataException("Generated owned scene/template/formal prefab identity changed; do not reuse stale generation.");
        }
        var playerAssembly = CompilationPipeline.GetAssemblies(AssembliesType.Player).Single(a => a.name == "Assembly-CSharp");
        var legacy = playerAssembly.defines.Contains("ENABLE_LEGACY_INPUT_MANAGER");
        var newer = playerAssembly.defines.Contains("ENABLE_INPUT_SYSTEM");
        var settings = Resources.FindObjectsOfTypeAll<PlayerSettings>().Single();
        var serialized = new SerializedObject(settings);
        var input = serialized.FindProperty("activeInputHandler");
        var expectedMode = mode == "ExpectedLegacy" ? 0 : mode == "ExpectedNew" ? 1 : 2;
        if (input == null || input.intValue != expectedMode || legacy != (mode != "ExpectedNew") ||
            newer != (mode != "ExpectedLegacy"))
        {
            throw new InvalidOperationException("Input mode/configured Player defines differ; configure/restart via existing TestsLegacy/New/Both first.");
        }
        var packages = UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages();
        var inputPackage = packages.SingleOrDefault(p => p.name == "com.unity.inputsystem");
        if (newer && (inputPackage == null || inputPackage.version != (Application.unityVersion == "6000.3.27f1" ? "1.20.1" : "1.14.2")))
        {
            throw new InvalidOperationException("The exact selected official Input System package is required; this template never installs/configures it.");
        }
        if (EditorUtility.IsDirty(settings))
        {
            throw new InvalidOperationException("Do not overwrite unsaved PlayerSettings.");
        }
        var output = Path.Combine(Project, "P/NativeInput", label, mode, run);
        RejectLinks(output);
        if (File.Exists(output) || Directory.Exists(output))
        {
            throw new IOException("Never overwrite an existing native-input build output.");
        }
        Directory.CreateDirectory(output);
        var settingsPath = Path.Combine(Project, "ProjectSettings/ProjectSettings.asset");
        var originalBytes = File.ReadAllBytes(settingsPath);
        var originalJson = EditorJsonUtility.ToJson(settings);
        var originalDiskHash = Sha(settingsPath);
        BuildReport report = null;
        var actualBackend = "";
        var actualStripping = "";
        var actualCpp = "NotApplicable";
        try
        {
            PlayerSettings.companyName = Path.GetFileName((string)owner["Delivery"]);
            PlayerSettings.productName = Path.GetFileName(Project) + "-Native-" + label + "-" + mode + "-" + run;
            PlayerSettings.SetApiCompatibilityLevel(NamedBuildTarget.Standalone, ApiCompatibilityLevel.NET_Standard);
            PlayerSettings.SetScriptingBackend(NamedBuildTarget.Standalone, backend);
            PlayerSettings.SetManagedStrippingLevel(NamedBuildTarget.Standalone,
                backend == ScriptingImplementation.IL2CPP ? ManagedStrippingLevel.Low : ManagedStrippingLevel.Disabled);
            if (backend == ScriptingImplementation.IL2CPP)
            {
                PlayerSettings.SetIl2CppCompilerConfiguration(NamedBuildTarget.Standalone, Il2CppCompilerConfiguration.Release);
            }
            actualBackend = PlayerSettings.GetScriptingBackend(NamedBuildTarget.Standalone).ToString();
            actualStripping = PlayerSettings.GetManagedStrippingLevel(NamedBuildTarget.Standalone).ToString();
            if (backend == ScriptingImplementation.IL2CPP)
            {
                actualCpp = PlayerSettings.GetIl2CppCompilerConfiguration(NamedBuildTarget.Standalone).ToString();
            }
            if (actualBackend != backend.ToString() || actualStripping != (backend == ScriptingImplementation.IL2CPP ? "Low" : "Disabled") ||
                backend == ScriptingImplementation.IL2CPP && actualCpp != "Release")
            {
                throw new InvalidOperationException("Actual build settings do not match the required support configuration.");
            }
            report = BuildPipeline.BuildPlayer(new BuildPlayerOptions
            {
                scenes = new[] { ScenePath }, locationPathName = Path.Combine(output, "Stage9NativeInput.exe"),
                target = BuildTarget.StandaloneWindows64, options = BuildOptions.None
            });
        }
        finally
        {
            // Restore only this receiver's known object/file, including empty per-target maps materialized by getters.
            EditorJsonUtility.FromJsonOverwrite(originalJson, settings);
            if (EditorJsonUtility.ToJson(settings) != originalJson)
            {
                throw new InvalidOperationException("Exact in-memory PlayerSettings restoration failed; no accepted BuildReport will be emitted.");
            }
            if (Sha(settingsPath) != originalDiskHash)
            {
                File.WriteAllBytes(settingsPath, originalBytes);
            }
            EditorUtility.ClearDirty(settings);
            if (Sha(settingsPath) != originalDiskHash)
            {
                throw new IOException("Exact on-disk PlayerSettings restoration failed.");
            }
        }
        if (report == null)
        {
            throw new InvalidOperationException("BuildPipeline returned no BuildReport.");
        }
        var licenses = Path.Combine(output, "Licenses");
        Directory.CreateDirectory(licenses);
        File.Copy("Assets/YLFramework/LICENSE.md", Path.Combine(licenses, "YLFramework-LICENSE.md"), false);
        foreach (var package in packages.Where(p => p.name == "com.unity.ugui" || p.name == "com.unity.inputsystem" ||
            p.name == "com.unity.nuget.newtonsoft-json"))
        {
            foreach (var name in new[] { "LICENSE.md", "LICENSE", "Third Party Notices.md" })
            {
                var path = Path.Combine(package.resolvedPath, name);
                if (File.Exists(path))
                {
                    File.Copy(path, Path.Combine(licenses, package.name + "-" + name), false);
                }
            }
        }
        var receipt = new JObject
        {
            ["Schema"] = "Stage9NativeInput-v1", ["CaptureProtocol"] = "B06Capture-v1", ["RemainingProtocol"] = "B06Remaining-v1", ["PlayerActivationProtocol"] = "B06PlayerActivation-v1", ["Project"] = Project, ["Delivery"] = owner["Delivery"],
            ["Unity"] = Application.unityVersion, ["UguiVersion"] = owner["UguiVersion"],
            ["InputSystemVersion"] = inputPackage == null ? null : inputPackage.version,
            ["PackageSha256"] = owner["PackageSha256"], ["GenerationSha256"] = Sha(generationPath),
            ["BuildRun"] = run, ["Mode"] = mode, ["LegacyInput"] = legacy, ["NewInput"] = newer,
            ["Backend"] = actualBackend, ["Stripping"] = actualStripping, ["CppConfiguration"] = actualCpp,
            ["Result"] = report.summary.result.ToString(), ["Errors"] = report.summary.totalErrors,
            ["Warnings"] = report.summary.totalWarnings, ["Development"] = (report.summary.options & BuildOptions.Development) != 0,
            ["Options"] = report.summary.options.ToString(), ["BuildGuid"] = report.summary.guid.ToString(),
            ["Target"] = report.summary.platform.ToString(), ["Executable"] = "Stage9NativeInput.exe",
            ["SettingsRestored"] = true, ["PlayerSettingsBeforeSha256"] = originalDiskHash,
            ["PlayerSettingsAfterSha256"] = Sha(settingsPath), ["PlayerDefines"] = new JArray(playerAssembly.defines),
            ["Messages"] = JArray.FromObject(report.steps.SelectMany(s => s.messages).Where(m => m.type != LogType.Log)
                .Select(m => new { Type = m.type.ToString(), m.content })),
            ["PlayerFiles"] = new JArray(SafeFiles(output).OrderBy(p => p, StringComparer.Ordinal).Select(path => new JObject
            {
                ["Path"] = Path.GetRelativePath(output, path).Replace('\\', '/'),
                ["Length"] = new FileInfo(path).Length, ["Sha256"] = Sha(path)
            }))
        };
        WriteNew(Path.Combine(output, "BuildReport.json"), receipt);
        if (report.summary.result != BuildResult.Succeeded || report.summary.totalErrors != 0)
        {
            throw new InvalidOperationException("Actual native-input Player build failed; preserve the failed output and report.");
        }
        Debug.Log("STAGE9_NATIVE_INPUT_BUILD_PASS " + label + " " + mode + " " + run);
    }

    private static string[] SafeFiles(string directory)
    {
        RejectLinks(directory);
        return Directory.GetFiles(directory).Concat(Directory.GetDirectories(directory).SelectMany(SafeFiles)).ToArray();
    }
    private static string Sha(string path)
    {
        RejectLinks(Path.GetFullPath(path));
        using var hash = SHA256.Create();
        using var file = File.OpenRead(path);
        return BitConverter.ToString(hash.ComputeHash(file)).Replace("-", "").ToLowerInvariant();
    }
    private static void WriteNew(string path, JObject value)
    {
        RejectLinks(Path.GetFullPath(path));
        using var stream = new FileStream(path, FileMode.CreateNew, FileAccess.Write, FileShare.None);
        using var writer = new StreamWriter(stream, new UTF8Encoding(false));
        writer.Write(value.ToString(Formatting.Indented));
    }
    private static void RejectLinks(string path)
    {
        for (var cursor = path; !string.IsNullOrEmpty(cursor); cursor = Path.GetDirectoryName(cursor))
        {
            if ((File.Exists(cursor) || Directory.Exists(cursor)) &&
                (File.GetAttributes(cursor) & FileAttributes.ReparsePoint) != 0)
            {
                throw new IOException("Linked path rejected: " + cursor);
            }
        }
    }
    private static string Argument(string name)
    {
        var args = Environment.GetCommandLineArgs();
        var index = Array.IndexOf(args, name);
        if (index < 0 || index + 1 == args.Length || Array.LastIndexOf(args, name) != index)
        {
            throw new ArgumentException("Exactly one explicit argument required: " + name);
        }
        return args[index + 1];
    }
}
