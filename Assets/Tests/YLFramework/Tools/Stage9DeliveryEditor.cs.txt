public static class Stage9DeliveryEditor
{
    private const string Root = "Assets/Stage9";
    private static string Project => Path.GetFullPath(Application.dataPath + "/..");
    private static JObject Owner => JObject.Parse(File.ReadAllText(Path.Combine(Project, "stage9-owner.json")));
    private static string Delivery => (string)Owner["Delivery"];
    private static string Run => Argument("-s9-run", "initial");

    private static void Verify()
    {
        var owner = Owner;
        if ((string)owner["Purpose"] != "Stage9-independent-receiver" || (string)owner["Project"] != Project ||
            !Project.StartsWith(Path.GetFullPath(Delivery) + Path.DirectorySeparatorChar, StringComparison.Ordinal) ||
            Project.IndexOf("CGJ2024", StringComparison.OrdinalIgnoreCase) >= 0)
        {
            throw new InvalidOperationException("Wrong independent project identity.");
        }

        var expectedUnity = (string)owner["UnityVersion"];
        if ((expectedUnity != "2022.3.62f3c1" && expectedUnity != "6000.3.27f1") ||
            Application.unityVersion != expectedUnity || EditorApplication.isPlayingOrWillChangePlaymode ||
            EditorUtility.scriptCompilationFailed || EditorUserBuildSettings.activeBuildTarget != BuildTarget.StandaloneWindows64)
        {
            throw new InvalidOperationException("Wrong actual Unity state.");
        }

        for (var i = 0; i < EditorSceneManager.sceneCount; i++)
        {
            if (EditorSceneManager.GetSceneAt(i).isDirty)
            {
                throw new InvalidOperationException("Never save an unowned dirty scene.");
            }
        }
    }

    public static void Audit()
    {
        Verify();
        var files = JArray.Parse(File.ReadAllText(Path.Combine(Delivery, (string)Owner["Package"] + ".files.json")));
        var skippedEmptyFolders = new List<string>();
        foreach (var file in files)
        {
            var relative = (string)file["Path"];
            if ((bool)file["Folder"] && !File.Exists(Path.Combine(Project, relative + ".meta")) &&
                !files.Any(x => ((string)x["Path"]).StartsWith(relative + "/", StringComparison.Ordinal)))
            {
                // Unity的正式导入会忽略无载荷的空目录；只允许无后代、无使用者的目录身份缺席。
                skippedEmptyFolders.Add(relative);
                continue;
            }
            if (Sha(Path.Combine(Project, relative + ".meta")) != (string)file["MetaSha256"] ||
                (!(bool)file["Folder"] && Sha(Path.Combine(Project, relative)) != (string)file["Sha256"]))
            {
                throw new InvalidDataException("Imported product bytes differ from the actual package: " + relative);
            }
        }
        var duplicates = Directory.GetFiles(Application.dataPath, "*.meta", SearchOption.AllDirectories)
            .Select(path => new { Path = path, Guid = Regex.Match(File.ReadAllText(path), @"(?m)^guid:\s*([0-9a-f]{32})").Groups[1].Value })
            .Where(x => x.Guid.Length != 0).GroupBy(x => x.Guid).Where(g => g.Count() != 1).ToArray();
        if (duplicates.Length != 0)
        {
            throw new InvalidDataException("Duplicate Assets GUIDs.");
        }

        var prefabs = AssetDatabase.FindAssets("t:Prefab", new[] { "Assets/YLFramework" }).Select(AssetDatabase.GUIDToAssetPath).ToArray();
        foreach (var path in prefabs)
        {
            var prefab = AssetDatabase.LoadAssetAtPath<GameObject>(path);
            foreach (var transform in prefab.GetComponentsInChildren<Transform>(true))
            {
                if (GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(transform.gameObject) != 0)
                {
                    throw new InvalidDataException("Missing script: " + path);
                }

                foreach (var component in transform.GetComponents<Component>())
                {
                    if (!component)
                    {
                        throw new InvalidDataException("Missing component: " + path);
                    }

                    var serialized = new SerializedObject(component);
                    var property = serialized.GetIterator();
                    while (property.Next(true))
                    {
                        if (property.propertyType == SerializedPropertyType.ObjectReference && property.objectReferenceInstanceIDValue != 0 && !property.objectReferenceValue)
                        {
                            throw new InvalidDataException("Invalid prefab reference: " + path + " / " + property.propertyPath);
                        }
                    }
                }
            }
            foreach (var dependency in AssetDatabase.GetDependencies(path, true))
            {
                if (!dependency.StartsWith("Assets/YLFramework/", StringComparison.Ordinal) && !dependency.StartsWith("Packages/com.unity.", StringComparison.Ordinal) &&
                    !dependency.StartsWith("Resources/", StringComparison.Ordinal) && !dependency.StartsWith("Library/", StringComparison.Ordinal))
                {
                    throw new InvalidDataException("Unexpected product prefab dependency: " + dependency);
                }
            }
        }
        YLFramework.Editor.Core.UIPanelAssetMigration.ValidateAssets();
        var packages = UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages();
        if (packages.Any(p => !p.name.StartsWith("com.unity.modules.", StringComparison.Ordinal) &&
            !new[] { "com.unity.ugui", "com.unity.nuget.newtonsoft-json" }.Contains(p.name)))
        {
            throw new InvalidDataException("A package outside the declared bare-product closure entered the receiver.");
        }

        var references = typeof(ResourceSystem).Assembly.GetReferencedAssemblies().Select(x => x.Name).ToArray();
        if (references.Any(x => !Regex.IsMatch(x,
            @"^(?:mscorlib|netstandard|System(?:\.[A-Za-z0-9_]+)*|UnityEngine(?:\.[A-Za-z0-9_]+)*|UnityEditor(?:\.[A-Za-z0-9_]+)*|Unity\.ugui|Microsoft\.CSharp|Microsoft\.Win32\.Primitives|Newtonsoft\.Json|BouncyCastle\.Cryptography)$") || x.Contains("TestRunner") || x.Contains("Tests")))
        {
            throw new InvalidDataException("Product Editor-profile runtime reverse dependency: " + string.Join(", ", references));
        }

        var player = CompilationPipeline.GetAssemblies(AssembliesType.Player).Single(a => a.name == "YLFramework.Runtime");
        var playerReferences = player.compiledAssemblyReferences.Select(Path.GetFileNameWithoutExtension)
            .Concat(player.assemblyReferences.Select(a => a.name)).ToArray();
        if (playerReferences.Any(x => !Regex.IsMatch(x,
            @"^(?:mscorlib|netstandard|System(?:\.[A-Za-z0-9_]+)*|UnityEngine(?:\.[A-Za-z0-9_]+)*|Unity\.ugui|Microsoft\.CSharp|Microsoft\.Win32\.Primitives|Newtonsoft\.Json|BouncyCastle\.Cryptography)$") || x.Contains("TestRunner") || x.Contains("Tests")))
        {
            throw new InvalidDataException("Product Player-profile runtime reverse dependency: " + string.Join(", ", playerReferences));
        }

        var analyzer = "Assets/YLFramework/Runtime/Core/Analyzers/YLFramework.Analyzers.dll";
        var analyzerRecord = files.Single(file => (string)file["Path"] == analyzer);
        if ((bool)analyzerRecord["Folder"] || Sha(analyzer) != (string)analyzerRecord["Sha256"])
        {
            throw new InvalidDataException("Analyzer differs from this receiver's audited package artifact.");
        }

        var responseFiles = Directory.GetFiles("Library/Bee", "*.rsp", SearchOption.AllDirectories)
            .Where(x => File.ReadAllText(x).Contains("YLFramework.Analyzers.dll")).ToArray();
        if (responseFiles.Length == 0)
        {
            throw new InvalidDataException("Analyzer absent from actual Unity compiler inputs.");
        }

        var evidence = new JObject
        {
            ["Passed"] = true,
            ["Project"] = Project,
            ["Pid"] = System.Diagnostics.Process.GetCurrentProcess().Id,
            ["Unity"] = Application.unityVersion,
            ["ProductRecords"] = files.Count,
            ["MissingScripts"] = 0,
            ["DuplicateGuids"] = 0,
            ["Prefabs"] = new JArray(prefabs),
            ["RuntimeReferences"] = new JArray(references),
            ["ImporterSkippedEmptyFolders"] = new JArray(skippedEmptyFolders),
            ["PlayerCompilationReferences"] = new JArray(playerReferences),
            ["RenderPipeline"] = GraphicsSettings.currentRenderPipeline ? GraphicsSettings.currentRenderPipeline.GetType().FullName : "Built-in",
            ["Packages"] = JArray.FromObject(packages.Select(p => new { p.name, p.version, Source = p.source.ToString(), p.resolvedPath })),
            ["AnalyzerSha256"] = Sha(analyzer),
            ["AnalyzerExpectedSha256"] = (string)analyzerRecord["Sha256"],
            ["AnalyzerCompilerInputs"] = new JArray(responseFiles),
            ["Assemblies"] = JArray.FromObject(CompilationPipeline.GetAssemblies().Select(a => new { a.name, a.sourceFiles, a.compiledAssemblyReferences }))
        };
        Write("audit-" + Run + ".json", evidence);
        CleanBatchScene();
    }

    public static void Generate()
    {
        Verify();
        if (AssetDatabase.IsValidFolder(Root + "/Content"))
        {
            throw new IOException("Never adopt an existing fixture.");
        }

        EditorSettings.spritePackerMode = SpritePackerMode.SpriteAtlasV2;
        AssetDatabase.CreateFolder(Root, "Content");
        if (!AssetDatabase.IsValidFolder("Assets/StreamingAssets"))
        {
            AssetDatabase.CreateFolder("Assets", "StreamingAssets");
        }

        var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        var camera = new GameObject("Independent delivery camera", typeof(Camera), typeof(AudioListener));
        camera.GetComponent<Camera>().clearFlags = CameraClearFlags.SolidColor;
        camera.GetComponent<Camera>().backgroundColor = new Color(.025f, .04f, .07f);
        EditorSceneManager.SaveScene(scene, Root + "/Bootstrap.unity");
        var neutral = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        new GameObject("Neutral camera", typeof(Camera), typeof(AudioListener));
        EditorSceneManager.SaveScene(neutral, Root + "/Stage9Neutral.unity");
        scene = EditorSceneManager.OpenScene(Root + "/Bootstrap.unity", OpenSceneMode.Single);
        var config = new JObject { ["UnityVersion"] = Application.unityVersion, ["Projects"] = new JArray() };
        var publications = new JArray();
        foreach (var encrypted in new[] { false, true })
        {
            var id = encrypted ? "stage9-encrypted" : "stage9-none";
            var folder = Root + "/Content/" + id;
            AssetDatabase.CreateFolder(Root + "/Content", id);
            AssetDatabase.CreateFolder(folder, "Text");
            AssetDatabase.CreateFolder(folder, "Visual");
            Texture(folder + "/Visual/icon.png");
            Wave(folder + "/Text/Tone.wav");
            PanelPrefab(folder + "/Visual/Panel.prefab", new Vector2(-240, 0));
            PanelPrefab(folder + "/Visual/Modal.prefab", new Vector2(240, 0));
            if (!encrypted)
            {
                AssetDatabase.CreateFolder(folder, "Scenes");
                foreach (var name in new[] { "A", "B" })
                {
                    var target = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
                    new GameObject("Stage9 scene " + name, typeof(Camera));
                    EditorSceneManager.SaveScene(target, folder + "/Scenes/Stage9" + name + ".unity");
                }

                scene = EditorSceneManager.OpenScene(Root + "/Bootstrap.unity", OpenSceneMode.Single);
            }
            var project = ScriptableObject.CreateInstance<ResourceProject>();
            project.projectId = id;
            project.displayName = "Stage 9 independent " + id;
            project.sourceDirectories.Add(new ResourceSourceDirectory { guid = AssetDatabase.AssetPathToGUID(folder + "/Text"), addressPrefix = "text" });
            project.sourceDirectories.Add(new ResourceSourceDirectory { guid = AssetDatabase.AssetPathToGUID(folder + "/Visual"), addressPrefix = "ui" });
            project.compression = ResourceCompression.Uncompressed;
            project.contentEncryption = encrypted ? ResourceContentEncryption.Aes256GcmHkdfSegmentedV1 : ResourceContentEncryption.None;
            project.contentKeyId = encrypted ? "stage9-nonproduction" : null;
            project.groups.Add(Group("text", (folder + "/Text/Message.txt", "message"), (folder + "/Text/Tone.wav", "tone")));
            project.groups.Add(Group("ui", (folder + "/Visual/Panel.prefab", "panel"), (folder + "/Visual/Modal.prefab", "modal")));
            project.groups.Add(new ResourceGroup { id = "atlases", packing = ResourcePacking.Together });
            if (!encrypted)
            {
                project.sourceDirectories.Add(new ResourceSourceDirectory { guid = AssetDatabase.AssetPathToGUID(folder + "/Scenes"), addressPrefix = "scenes" });
                project.groups.Add(Group("scenes", (folder + "/Scenes/Stage9A.unity", "scene-a"), (folder + "/Scenes/Stage9B.unity", "scene-b")));
            }
            AssetDatabase.CreateAsset(project, folder + "/Project.asset");
            var atlases = ResourceAtlasGenerator.CreateConfiguration(project, folder, "Generated");
            atlases.rules.Add(new ResourceAtlasRule
            {
                id = "ui",
                groupId = "atlases",
                address = "atlas",
                grouping = ResourceAtlasGrouping.Together,
                sources = new Object[] { AssetDatabase.LoadAssetAtPath<Texture2D>(folder + "/Visual/icon.png") }
            });
            EditorUtility.SetDirty(atlases);
            AssetDatabase.SaveAssetIfDirty(atlases);
            var digests = new JArray();
            for (var version = 1; version <= 2; version++)
            {
                File.WriteAllText(folder + "/Text/Message.txt", "stage9-version-" + version, new UTF8Encoding(false));
                AssetDatabase.ImportAsset(folder + "/Text/Message.txt", ImportAssetOptions.ForceSynchronousImport);
                project = AssetDatabase.LoadAssetAtPath<ResourceProject>(folder + "/Project.asset");
                // Message首次导入后重新登记真实GUID；不伪造或拷贝来源工程身份。
                project.groups[0].selections[0].guid = AssetDatabase.AssetPathToGUID(folder + "/Text/Message.txt");
                EditorUtility.SetDirty(project);
                AssetDatabase.SaveAssetIfDirty(project);
                EditorSceneManager.SaveScene(scene, Root + "/Bootstrap.unity");
                var report = ResourceEditorOperations.BuildAndPublish(project, "stage9-app", provider: encrypted ? new Stage9DeliveryKeys() : null);
                if (report.BuildError != null || report.PublicationError != null || report.GenerationError != null || report.Release == null)
                {
                    throw new InvalidOperationException(report.Summary, report.BuildError ?? report.PublicationError ?? report.GenerationError);
                }

                var release = report.Release;
                var deployed = ResourceBuiltInDeployer.Deploy(ResourceBuiltInDeployer.Preflight(release));
                digests.Add(release.Catalog.CatalogDigest);
                publications.Add(new JObject
                {
                    ["Project"] = id,
                    ["Version"] = version,
                    ["Digest"] = release.Catalog.CatalogDigest,
                    ["Release"] = release.DirectoryPath,
                    ["Deployment"] = deployed.RelativeDirectory,
                    ["SourceDirectories"] = project.sourceDirectories.Count,
                    ["AutomaticAtlas"] = report.Build.AtlasGeneration != null
                });
            }
            ((JArray)config["Projects"]).Add(new JObject { ["Id"] = id, ["Digests"] = digests, ["Encrypted"] = encrypted });
        }
        // 真的使用随包样例的Create/Build/Deploy公开入口，不复制旧Acceptance场景。
        var sampleFolder = ResourceStartupExampleEditor.Create();
        var example = Object.FindObjectOfType<ResourceStartupExample>();
        var sampleRelease = ResourceStartupExampleEditor.Build(example);
        EditorSceneManager.SaveScene(example.gameObject.scene, sampleFolder + "/Example.unity");
        ResourceStartupExampleEditor.DeployBuiltIn(example);
        example.mode = ResourceLoadMode.Bundles;
        EditorSceneManager.SaveScene(example.gameObject.scene, sampleFolder + "/Example.unity");
        config["Sample"] = new JObject { ["Id"] = example.projectId, ["Digest"] = sampleRelease.Catalog.CatalogDigest, ["App"] = example.appCompatibility, ["Folder"] = sampleFolder };
        EditorSceneManager.OpenScene(Root + "/Bootstrap.unity", OpenSceneMode.Single);
        var probe = new GameObject("Independent delivery consumer").AddComponent<Stage9DeliveryProbe>();
        probe.fullTemplate = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/YLFramework/Prefabs/YLFramework.prefab");
        probe.colorPrefab = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/YLFramework/Prefabs/ColorSelector.prefab");
        probe.panelPrefab = AssetDatabase.LoadAssetAtPath<GameObject>(Root + "/Content/stage9-none/Visual/Panel.prefab");
        probe.modalPrefab = AssetDatabase.LoadAssetAtPath<GameObject>(Root + "/Content/stage9-none/Visual/Modal.prefab");
        EditorSceneManager.SaveScene(probe.gameObject.scene, Root + "/Bootstrap.unity");
        File.WriteAllText("Assets/StreamingAssets/stage9-config.json", config.ToString(Formatting.Indented));
        AssetDatabase.Refresh(ImportAssetOptions.ForceSynchronousImport);
        Write("publications-" + Run + ".json", publications);
        Write("configuration-" + Run + ".json", config);
        CleanBatchScene();
        Debug.Log("STAGE9_GENERATE_PASS");
    }

    public static void VerifyGeneration()
    {
        Verify();
        var configuration = JObject.Parse(File.ReadAllText("Assets/StreamingAssets/stage9-config.json"));
        var checkedVersions = new JArray();
        foreach (var item in (JArray)configuration["Projects"])
        {
            var id = (string)item["Id"];
            var project = AssetDatabase.LoadAssetAtPath<ResourceProject>(Root + "/Content/" + id + "/Project.asset");
            foreach (var digest in item["Digests"])
            {
                var release = ResourcePublisher.ReadVersion(project, BuildTarget.StandaloneWindows64, "stage9-app", (string)digest);
                var snapshot = ResourceLocalSnapshot.OpenBuiltIn(id, (string)digest, "StandaloneWindows64", "stage9-app");
                if (release.Catalog.CatalogDigest != (string)digest || snapshot.Document.CatalogDigest != (string)digest)
                {
                    throw new InvalidDataException("Incomplete independent release/deployment.");
                }

                checkedVersions.Add(new JObject { ["Id"] = id, ["Digest"] = digest, ["Release"] = release.DirectoryPath });
            }
        }
        var sample = configuration["Sample"];
        ResourceLocalSnapshot.OpenBuiltIn((string)sample["Id"], (string)sample["Digest"], "StandaloneWindows64", (string)sample["App"]);
        var scene = EditorSceneManager.OpenScene(Root + "/Bootstrap.unity", OpenSceneMode.Single);
        if (scene.GetRootGameObjects().SelectMany(x => x.GetComponentsInChildren<Stage9DeliveryProbe>(true)).Count() != 1)
        {
            throw new InvalidDataException("Independent bootstrap not saved.");
        }

        Write("generation-verified-" + Run + ".json", new JObject
        {
            ["Passed"] = true,
            ["Project"] = Project,
            ["CheckedVersions"] = checkedVersions,
            ["Sample"] = sample,
            ["Recovery"] = "Verify the same previously generated releases/deployments; no duplicate native builds or copied old scenes."
        });
        CleanBatchScene();
        Debug.Log("STAGE9_GENERATION_VERIFIED");
    }

    public static void DiagnoseReadDefaults()
    {
        Verify();
        var path = "ProjectSettings/ProjectSettings.asset";
        var before = File.ReadAllText(path);
        var beforeSha = Sha(path);
        var backend = PlayerSettings.GetScriptingBackend(BuildTargetGroup.Standalone);
        var stripping = PlayerSettings.GetManagedStrippingLevel(BuildTargetGroup.Standalone);
        AssetDatabase.SaveAssets();
        var exported = Path.Combine(Delivery, Path.GetFileName(Project) + "-defaults-probe-" + Run + ".unitypackage");
        YLFramework.Editor.Distribution.ExportUnityPackage.Export(exported);
        var after = File.ReadAllText(path);
        Write("default-getters-" + Run + ".json", new JObject
        {
            ["Project"] = Project,
            ["Pid"] = System.Diagnostics.Process.GetCurrentProcess().Id,
            ["BeforeSha256"] = beforeSha,
            ["AfterSha256"] = Sha(path),
            ["BeforeBackendEmpty"] = before.Contains("scriptingBackend: {}"),
            ["BeforeStrippingEmpty"] = before.Contains("managedStrippingLevel: {}"),
            ["Backend"] = backend.ToString(),
            ["Stripping"] = stripping.ToString(),
            ["FormalExportProbe"] = exported,
            ["AfterBackendEmpty"] = after.Contains("scriptingBackend: {}"),
            ["AfterStrippingEmpty"] = after.Contains("managedStrippingLevel: {}")
        });
        CleanBatchScene();
    }

    public static void ConfigureUrp()
    {
        Verify();
        var package = UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages()
            .SingleOrDefault(item => item.name == "com.unity.render-pipelines.universal");
        if (Application.unityVersion != "6000.3.27f1" || package == null || package.version != "17.3.0" ||
            UnityEngine.Rendering.GraphicsSettings.defaultRenderPipeline || QualitySettings.renderPipeline)
        {
            throw new InvalidOperationException("Only the exact new 6.3/URP17.3.0 Built-in predecessor may be configured.");
        }
        var folder = Root + "/Urp";
        if (Directory.Exists(folder) || File.Exists(folder + ".meta"))
        {
            throw new IOException("Never adopt or overwrite a prior URP configuration.");
        }
        var scene = EditorSceneManager.OpenScene(Root + "/Bootstrap.unity", OpenSceneMode.Single);
        var type = AppDomain.CurrentDomain.GetAssemblies()
            .Select(assembly => assembly.GetType("UnityEngine.Rendering.Universal.UniversalRenderPipelineAsset"))
            .Single(item => item != null);
        var factory = type.GetMethods(System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.NonPublic)
            .Single(method => method.Name == "CreateRendererAsset" && method.GetParameters().Length == 4 &&
                method.GetParameters()[0].ParameterType == typeof(string) && method.GetParameters()[1].ParameterType.IsEnum &&
                method.GetParameters()[2].ParameterType == typeof(bool) && method.GetParameters()[3].ParameterType == typeof(string));
        AssetDatabase.CreateFolder(Root, "Urp");
        var path = folder + "/Pipeline.asset";
        var renderer = (ScriptableObject)factory.Invoke(null, new object[]
        {
            path, Enum.Parse(factory.GetParameters()[1].ParameterType, "UniversalRenderer"), true, "Renderer"
        });
        var create = type.GetMethods(System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.Public)
            .Single(method => method.Name == "Create" && method.GetParameters().Length == 1 &&
                method.GetParameters()[0].ParameterType.IsInstanceOfType(renderer));
        var pipeline = (UnityEngine.Rendering.RenderPipelineAsset)create.Invoke(null, new object[] { renderer });
        AssetDatabase.CreateAsset(pipeline, path);
        var shader = Shader.Find("Universal Render Pipeline/Unlit");
        if (!shader)
        {
            throw new InvalidOperationException("Official URP Unlit shader is missing.");
        }
        var material = new Material(shader);
        material.SetColor("_BaseColor", Color.white);
        AssetDatabase.CreateAsset(material, folder + "/Display.mat");
        var consumer = Object.FindObjectOfType<Stage9DeliveryProbe>();
        consumer.urpMaterial = material;
        EditorUtility.SetDirty(consumer);
        EditorSceneManager.SaveScene(scene, Root + "/Bootstrap.unity");
        UnityEngine.Rendering.GraphicsSettings.defaultRenderPipeline = pipeline;
        QualitySettings.renderPipeline = pipeline;
        AssetDatabase.SaveAssetIfDirty(renderer);
        AssetDatabase.SaveAssetIfDirty(pipeline);
        var paths = new[] { path, folder + "/Pipeline_Renderer.asset", folder + "/Display.mat", Root + "/Bootstrap.unity" };
        Write("urp-configuration-" + Run + ".json", JObject.FromObject(new
        {
            Project, Unity = Application.unityVersion, Package = package.version,
            GraphicsPipeline = AssetDatabase.GetAssetPath(UnityEngine.Rendering.GraphicsSettings.defaultRenderPipeline),
            QualityPipeline = AssetDatabase.GetAssetPath(QualitySettings.renderPipeline),
            QualityLevel = QualitySettings.GetQualityLevel(),
            OfficialFactory = factory.ToString(),
            Files = paths.Select(item => new { Path = item, Guid = AssetDatabase.AssetPathToGUID(item), Sha256 = Sha(item), MetaSha256 = Sha(item + ".meta") }),
            EditorSha256 = Sha(Root + "/Editor/Stage9DeliveryEditor.cs"), ProbeSha256 = Sha(Root + "/Stage9DeliveryProbe.cs"),
            ConfigurationOnly = true, VisualPassed = false
        }));
        CleanBatchScene();
        Debug.Log("STAGE9_URP_CONFIGURED");
    }

    public static void BuildMono() => Build(ScriptingImplementation.Mono2x, "Mono");
    public static void BuildIl2Cpp() => Build(ScriptingImplementation.IL2CPP, "Il2Cpp");
    private static void Build(ScriptingImplementation backend, string name)
    {
        Verify();
        var output = Path.Combine(Project, "P", (name == "Mono" ? "m-" : "i-") + Run);
        if (Directory.Exists(output))
        {
            throw new IOException("Never overwrite an existing Player.");
        }

        foreach (var source in Directory.GetFiles("Assets/StreamingAssets", "*", SearchOption.AllDirectories))
        {
            var destination = Path.Combine(output, "Stage9Delivery_Data/StreamingAssets", Path.GetRelativePath("Assets/StreamingAssets", source));
            if (Path.DirectorySeparatorChar == '\\' && destination.Length >= 260)
            {
                throw new PathTooLongException("Use a shorter build Run id before copying native StreamingAssets: " + destination);
            }
        }

        Directory.CreateDirectory(output);
        EditorSceneManager.OpenScene(Root + "/Bootstrap.unity", OpenSceneMode.Single);
        PlayerSettings.companyName = Path.GetFileName(Delivery);
        PlayerSettings.productName = Path.GetFileName(Project) + "-" + name;
        PlayerSettings.SetApiCompatibilityLevel(BuildTargetGroup.Standalone, ApiCompatibilityLevel.NET_Standard);
        PlayerSettings.SetScriptingBackend(BuildTargetGroup.Standalone, backend);
        PlayerSettings.SetManagedStrippingLevel(BuildTargetGroup.Standalone, backend == ScriptingImplementation.IL2CPP ? ManagedStrippingLevel.Low : ManagedStrippingLevel.Disabled);
        if (backend == ScriptingImplementation.IL2CPP)
        {
            PlayerSettings.SetIl2CppCompilerConfiguration(BuildTargetGroup.Standalone, Il2CppCompilerConfiguration.Release);
        }

        var ownerSha = Sha(Path.Combine(Project, "stage9-owner.json"));
        var templatePaths = new[] { Root + "/Editor/Stage9DeliveryEditor.cs", Root + "/Stage9DeliveryProbe.cs", Root + "/Stage9DeliveryPanel.cs" };
        var templateHashes = JObject.FromObject(templatePaths.ToDictionary(Path.GetFileNameWithoutExtension, Sha));
        var report = BuildPipeline.BuildPlayer(new BuildPlayerOptions
        {
            scenes = new[] { Root + "/Bootstrap.unity", Root + "/Stage9Neutral.unity" },
            locationPathName = Path.Combine(output, "Stage9Delivery.exe"),
            target = BuildTarget.StandaloneWindows64,
            options = BuildOptions.None
        });
        if (ownerSha != Sha(Path.Combine(Project, "stage9-owner.json")) ||
            templatePaths.Any(path => Sha(path) != (string)templateHashes[Path.GetFileNameWithoutExtension(path)]))
        {
            throw new InvalidDataException("Build owner or templates changed during the actual native build; preserve the output without claiming it.");
        }
        var licenses = Path.Combine(output, "Licenses");
        Directory.CreateDirectory(licenses);
        foreach (var path in new[] { "Assets/YLFramework/LICENSE.md",
            "Assets/YLFramework/Plugins/BouncyCastle/SOURCE.txt" })
        {
            File.Copy(path, Path.Combine(licenses, path.Replace("Assets/YLFramework/", "").Replace('/', '-')));
        }

        foreach (var assembly in new[] { typeof(JsonConvert).Assembly, typeof(Button).Assembly })
        {
            var package = UnityEditor.PackageManager.PackageInfo.FindForAssembly(assembly);
            foreach (var file in new[] { "LICENSE.md", "Third Party Notices.md" })
            {
                if (File.Exists(Path.Combine(package.resolvedPath, file)))
                {
                    File.Copy(Path.Combine(package.resolvedPath, file), Path.Combine(licenses, package.name + "-" + file));
                }
            }
        }
        var receipt = new JObject
        {
            ["Schema"] = "Stage9NativeBuild-v2",
            ["OwnerSha256"] = ownerSha,
            ["TemplateSha256"] = templateHashes,
            ["Result"] = report.summary.result.ToString(),
            ["Errors"] = report.summary.totalErrors,
            ["Warnings"] = report.summary.totalWarnings,
            ["Options"] = report.summary.options.ToString(),
            ["Development"] = false,
            ["Backend"] = backend.ToString(),
            ["Stripping"] = PlayerSettings.GetManagedStrippingLevel(BuildTargetGroup.Standalone).ToString(),
            ["BuildGuid"] = report.summary.guid.ToString(),
            ["Unity"] = Application.unityVersion,
            ["Project"] = Project,
            ["Pid"] = System.Diagnostics.Process.GetCurrentProcess().Id,
            ["PackageSha256"] = Owner["PackageSha256"],
            ["Seconds"] = report.summary.totalTime.TotalSeconds,
            ["Messages"] = JArray.FromObject(report.steps.SelectMany(s => s.messages).Where(m => m.type != LogType.Log).Select(m => new { Type = m.type.ToString(), m.content })),
            ["Assemblies"] = JArray.FromObject(CompilationPipeline.GetAssemblies(AssembliesType.Player).Select(a => new { a.name, a.sourceFiles, a.compiledAssemblyReferences })),
            ["PlayerFiles"] = new JArray(Directory.GetFiles(output, "*", SearchOption.AllDirectories).OrderBy(x => x, StringComparer.Ordinal)
                .Select(x => new JObject { ["Path"] = x.Substring(output.Length + 1).Replace('\\', '/'), ["Sha256"] = Sha(x), ["Length"] = new FileInfo(x).Length }))
        };
        File.WriteAllText(Path.Combine(output, "BuildReport.json"), receipt.ToString(Formatting.Indented));
        Write("build-" + name + "-" + Run + ".json", receipt);
        if (report.summary.result != BuildResult.Succeeded || report.summary.totalErrors != 0)
        {
            throw new InvalidOperationException("Actual independent Player build failed.");
        }

        CleanBatchScene();
        Debug.Log("STAGE9_BUILD_PASS " + name);
    }

    private static void CleanBatchScene()
    {
        // 2022.3批处理退出时SceneTracker可能晚于MonoManager销毁；仅提前结束本工程自有的已保存场景。
        Selection.objects = Array.Empty<Object>();
        EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        if (Environment.GetCommandLineArgs().Contains("-quit"))
        {
            UnityEditor.SessionState.SetBool("Stage9.ReadyToExit", false);
            return;
        }
        UnityEditor.SessionState.SetBool("Stage9.ReadyToExit", true);
        EditorApplication.delayCall += ExitWhenSettled;
    }

    [InitializeOnLoadMethod]
    private static void ResumeExitAfterReload()
    {
        if (UnityEditor.SessionState.GetBool("Stage9.ReadyToExit", false))
        {
            EditorApplication.delayCall += ExitWhenSettled;
        }
    }
    private static void ExitWhenSettled()
    {
        if (EditorApplication.isCompiling || EditorApplication.isUpdating)
        {
            EditorApplication.delayCall += ExitWhenSettled;
            return;
        }
        UnityEditor.SessionState.SetBool("Stage9.ReadyToExit", false);
        EditorApplication.Exit(0);
    }

    private static void Texture(string path)
    {
        var texture = new Texture2D(16, 16, TextureFormat.RGBA32, false);
        texture.SetPixels(Enumerable.Repeat(new Color(.1f, .8f, .5f), 256).ToArray());
        texture.Apply();
        File.WriteAllBytes(path, texture.EncodeToPNG());
        Object.DestroyImmediate(texture);
        AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceSynchronousImport);
        var importer = (TextureImporter)AssetImporter.GetAtPath(path);
        importer.textureType = TextureImporterType.Sprite;
        importer.spriteImportMode = SpriteImportMode.Single;
        importer.mipmapEnabled = false;
        importer.textureCompression = TextureImporterCompression.Uncompressed;
        importer.SaveAndReimport();
    }
    private static void Wave(string path)
    {
        using (var writer = new BinaryWriter(File.Create(path)))
        {
            const int Rate = 22050;
            writer.Write(Encoding.ASCII.GetBytes("RIFF"));
            writer.Write(36 + Rate * 2);
            writer.Write(Encoding.ASCII.GetBytes("WAVEfmt "));
            writer.Write(16);
            writer.Write((short)1);
            writer.Write((short)1);
            writer.Write(Rate);
            writer.Write(Rate * 2);
            writer.Write((short)2);
            writer.Write((short)16);
            writer.Write(Encoding.ASCII.GetBytes("data"));
            writer.Write(Rate * 2);
            for (var i = 0; i < Rate; i++)
            {
                writer.Write((short)(Math.Sin(i * 2 * Math.PI * 220 / Rate) * 4096));
            }
        }
        AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceSynchronousImport);
    }
    private static string Sha(string path)
    {
        using var hash = SHA256.Create();
        using var file = File.OpenRead(path);
        return BitConverter.ToString(hash.ComputeHash(file)).Replace("-", "").ToLowerInvariant();
    }
    private static string Argument(string name, string fallback)
    {
        var args = Environment.GetCommandLineArgs();
        var index = Array.IndexOf(args, name);
        return index < 0 ? fallback : args[index + 1];
    }
    private static void Write(string name, JToken value)
    {
        var path = Path.Combine(Delivery, Path.GetFileName(Project) + "-" + name);
        if (File.Exists(path))
        {
            throw new IOException("Preserve existing evidence.");
        }
        File.WriteAllText(path, value.ToString(Formatting.Indented));
    }
