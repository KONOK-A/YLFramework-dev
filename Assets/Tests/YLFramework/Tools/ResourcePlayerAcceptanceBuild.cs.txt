// 只安装到新建、已登记身份的独立验收载体；不在原游戏或主工程执行。
public static class ResourcePlayerAcceptanceBuild
{
    private const string Root = "Assets/R810";
    private static string Carrier => Path.GetDirectoryName(Application.dataPath);
    private static JObject Owner => JObject.Parse(File.ReadAllText(Path.Combine(Carrier, "r810-owner.json")));
    private static string Evidence => (string)Owner["Evidence"];
    private static string ExpectedUnity => (string)Owner["UnityVersion"] ?? "2022.3.62f3c1";
    private static string ExpectedUgui => (string)Owner["UguiVersion"] ?? "1.0.0";
    private static string Argument(string name, string fallback)
    {
        var args = Environment.GetCommandLineArgs();
        var index = Array.IndexOf(args, name);
        return index >= 0 && index + 1 < args.Length ? args[index + 1] : fallback;
    }

    public static void Generate()
    {
        Verify();
        if (AssetDatabase.IsValidFolder(Root + "/Content"))
        {
            throw new IOException("Refusing an existing fixture.");
        }

        PlayerSettings.SetScriptingBackend(BuildTargetGroup.Standalone, ScriptingImplementation.Mono2x);
        PlayerSettings.SetApiCompatibilityLevel(BuildTargetGroup.Standalone, ApiCompatibilityLevel.NET_Standard);
        EditorSettings.spritePackerMode = SpritePackerMode.SpriteAtlasV2;
        var bootstrap = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        new GameObject("Acceptance Camera", typeof(Camera), typeof(AudioListener));
        EditorSceneManager.SaveScene(bootstrap, Root + "/Bootstrap.unity");
        AssetDatabase.CreateFolder(Root, "Content");
        // Deployer只管理YLFrameworkResources，载体显式创建自己的StreamingAssets父目录。
        if (!AssetDatabase.IsValidFolder("Assets/StreamingAssets"))
        {
            AssetDatabase.CreateFolder("Assets", "StreamingAssets");
        }

        var config = JObject.Parse(File.ReadAllText(Path.Combine(Evidence, "public-config.json")));
        config["UnityVersion"] = ExpectedUnity;
        config["UguiVersion"] = ExpectedUgui;
        config["None"] = new JArray();
        config["Encrypted"] = new JArray();
        var publications = new JArray();
        using var certificate = new System.Security.Cryptography.X509Certificates.X509Certificate2(
            Path.Combine(Evidence, "loopback.pfx"), "r810-loopback-only", System.Security.Cryptography.X509Certificates.X509KeyStorageFlags.EphemeralKeySet);
        using var rsa = System.Security.Cryptography.X509Certificates.RSACertificateExtensions.GetRSAPrivateKey(certificate);
        foreach (var id in new[] { "r810-a", "r810-b", "r810-e" })
        {
            var encrypted = id == "r810-e";
            var folder = Root + "/Content/" + id;
            AssetDatabase.CreateFolder(Root + "/Content", id);
            Texture(folder + "/shared.png", 16, false);
            Texture(folder + "/icon.png", 16, true);
            var material = new Material(Shader.Find("Unlit/Texture")) { mainTexture = AssetDatabase.LoadAssetAtPath<Texture2D>(folder + "/shared.png") };
            AssetDatabase.CreateAsset(material, folder + "/Shared.mat");
            foreach (var file in new[] { "Prefab.prefab", "Other.prefab" })
            {
                var cube = GameObject.CreatePrimitive(PrimitiveType.Cube);
                cube.GetComponent<Renderer>().sharedMaterial = material;
                PrefabUtility.SaveAsPrefabAsset(cube, folder + "/" + file);
                Object.DestroyImmediate(cube);
            }
            Wave(folder + "/Audio.wav");
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive);
            var marker = new GameObject("r810-scene");
            SceneManager.MoveGameObjectToScene(marker, scene);
            EditorSceneManager.SaveScene(scene, folder + "/" + id + "Scene.unity");
            EditorSceneManager.CloseScene(scene, true);
            SceneManager.SetActiveScene(bootstrap);
            var atlas = new SpriteAtlasAsset();
            atlas.Add(new Object[] { AssetDatabase.LoadAssetAtPath<Texture2D>(folder + "/icon.png") });
            atlas.SetIncludeInBuild(false);
            SpriteAtlasAsset.Save(atlas, folder + "/Atlas.spriteatlasv2");
            AssetDatabase.ImportAsset(folder + "/Atlas.spriteatlasv2", ImportAssetOptions.ForceSynchronousImport);
            var importer = (SpriteAtlasImporter)AssetImporter.GetAtPath(folder + "/Atlas.spriteatlasv2");
            importer.includeInBuild = false;
            importer.SaveAndReimport();
            if ((object)atlas is Object atlasObject)
            {
                Object.DestroyImmediate(atlasObject);
            }

            var project = ScriptableObject.CreateInstance<ResourceProject>();
            project.projectId = id;
            project.displayName = id;
            project.sourceDirectories.Add(new ResourceSourceDirectory { guid = AssetDatabase.AssetPathToGUID(folder), addressPrefix = "" });
            project.compression = ResourceCompression.Uncompressed;
            project.contentEncryption = encrypted ? ResourceContentEncryption.Aes256GcmHkdfSegmentedV1 : ResourceContentEncryption.None;
            project.contentKeyId = encrypted ? "r810-content" : null;
            AssetDatabase.CreateAsset(project, Root + "/" + id + ".asset");
            var digests = new JArray();
            for (var version = 1; version <= 2; version++)
            {
                File.WriteAllText(folder + "/Message.txt", "r810-version-" + version, new UTF8Encoding(false));
                AssetDatabase.ImportAsset(folder + "/Message.txt", ImportAssetOptions.ForceSynchronousImport);
                // 真实大文件使partial跨进程续传可观测；只作为正式TextAsset，不另造下载路径。
                var bulk = new byte[3 * 1024 * 1024];
                new System.Random(81000 + version).NextBytes(bulk);
                File.WriteAllBytes(folder + "/Bulk.bytes", bulk);
                AssetDatabase.ImportAsset(folder + "/Bulk.bytes", ImportAssetOptions.ForceSynchronousImport);
                // 正式部署会刷新AssetDatabase，下一轮重新取得持久配置，不复用已失效SO引用。
                project = AssetDatabase.LoadAssetAtPath<ResourceProject>(Root + "/" + id + ".asset");
                if (!project)
                {
                    throw new InvalidDataException("The owned persisted project is missing.");
                }

                project.groups = new List<ResourceGroup>
                {
                    Group(folder, "main", ("Message.txt", "message"), ("Prefab.prefab", "prefab"), ("Other.prefab", "other")),
                    Group(folder, "atlas", ("Atlas.spriteatlasv2", "atlas")), Group(folder, "audio", ("Audio.wav", "audio")),
                    Group(folder, "scene", (id + "Scene.unity", "scene")), Group(folder, "bulk", ("Bulk.bytes", "bulk"))
                };
                EditorUtility.SetDirty(project);
                AssetDatabase.SaveAssetIfDirty(project);
                ResourceEditorOperations.PrepareAtlases(project);
                var keys = new Keys();
                var build = ResourceBundleBuilder.BuildWithContentKey(project, BuildTarget.StandaloneWindows64, encrypted ? keys : null);
                var release = ResourcePublisher.PublishWithContentKey(build, "r810-app", encrypted ? keys : null);
                var deployed = ResourceBuiltInDeployer.Deploy(ResourceBuiltInDeployer.Preflight(release));
                digests.Add(release.Catalog.CatalogDigest);
                var serverRoot = Path.Combine(Evidence, "server", id);
                Directory.CreateDirectory(serverRoot);
                var target = Path.Combine(serverRoot, release.Catalog.CatalogDigest);
                ResourceEditorOperations.ExportRelease(release, target);
                var envelope = ResourceReleaseSigner.Create(release, "acceptance", version, "r810-test", rsa, DateTimeOffset.UtcNow.AddDays(2), encrypted);
                File.WriteAllBytes(Path.Combine(serverRoot, "release-v" + version + ".json"), envelope);
                if (version == 2)
                {
                    File.WriteAllBytes(Path.Combine(serverRoot, "release.json"), envelope);
                }
                // 同序不同digest的授权，保持合法签名，供序号冲突拒绝门禁。
                if (version == 1)
                {
                    File.WriteAllBytes(Path.Combine(serverRoot, "release-conflict.json"),
                    ResourceReleaseSigner.Create(release, "acceptance", 2, "r810-test", rsa, DateTimeOffset.UtcNow.AddDays(2), encrypted));
                }

                publications.Add(new JObject
                {
                    ["ProjectId"] = id,
                    ["Version"] = version,
                    ["Digest"] = release.Catalog.CatalogDigest,
                    ["BuildId"] = release.Catalog.BuildId,
                    ["ProjectGuid"] = AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(project)),
                    ["Builder"] = typeof(ResourceBundleBuilder).FullName,
                    ["Publisher"] = typeof(ResourcePublisher).FullName,
                    ["Deployment"] = deployed.RelativeDirectory,
                    ["Files"] = new JArray(Files(deployed.Snapshot.DirectoryPath)),
                    ["KeyCopies"] = keys.Copies
                });
                Debug.Log("R810_PUBLISHED " + id + " v" + version + " " + release.Catalog.CatalogDigest);
            }
            ((JArray)config[encrypted ? "Encrypted" : "None"]).Add(new JObject { ["Id"] = id, ["Digests"] = digests });
        }
        File.WriteAllText("Assets/StreamingAssets/r810-config.json", config.ToString(Formatting.Indented));
        AssetDatabase.Refresh(ImportAssetOptions.ForceSynchronousImport);
        Write("publications.json", publications);
        Verify();
    }

    public static void Build()
    {
        Verify();
        var args = Environment.GetCommandLineArgs();
        var name = Argument("-r810-build", "Original");
        var output = Path.Combine(Carrier, "Builds", name);
        if (Directory.Exists(output))
        {
            throw new IOException("Refusing to overwrite Player evidence.");
        }

        var backend = (ScriptingImplementation)Enum.Parse(typeof(ScriptingImplementation), Argument("-r810-backend", "Mono2x"));
        if (backend != ScriptingImplementation.Mono2x && backend != ScriptingImplementation.IL2CPP)
        {
            throw new InvalidOperationException("Unsupported acceptance backend.");
        }

        var profile = Argument("-r810-profile", "auto");
        if (profile != "auto" && profile != "development" && profile != "nondevelopment")
        {
            throw new InvalidOperationException("Unsupported acceptance development profile.");
        }

        var acceptance = name.StartsWith("C", StringComparison.Ordinal);
        var development = profile == "auto" ? !name.StartsWith("R", StringComparison.Ordinal) : profile == "development";
        if (acceptance && !development)
        {
            throw new InvalidOperationException("The acceptance pin is Development only; do not weaken it for a non-Development build.");
        }

        var candidate = args.Contains("-r810-il2cpp-candidate");
        if (candidate && (backend != ScriptingImplementation.IL2CPP || !development || !acceptance))
        {
            throw new InvalidOperationException("Candidate policy is isolated C-prefixed IL2CPP Development only.");
        }

        PlayerSettings.SetApiCompatibilityLevel(BuildTargetGroup.Standalone, ApiCompatibilityLevel.NET_Standard);
        PlayerSettings.SetScriptingBackend(BuildTargetGroup.Standalone, backend);
        PlayerSettings.SetManagedStrippingLevel(BuildTargetGroup.Standalone, backend == ScriptingImplementation.IL2CPP ? ManagedStrippingLevel.Low : ManagedStrippingLevel.Disabled);
        if (backend == ScriptingImplementation.IL2CPP)
        {
            PlayerSettings.SetIl2CppCompilerConfiguration(BuildTargetGroup.Standalone, Il2CppCompilerConfiguration.Release);
        }
        PlayerSettings.companyName = "YL-Resource-Acceptance";
        PlayerSettings.productName = "R810-" + name;

        var defines = new List<string>();
        if (acceptance)
        {
            defines.Add("YL_RESOURCE_ACCEPTANCE_TESTS");
        }

        if (candidate)
        {
            defines.Add("YL_RESOURCE_IL2CPP_CANDIDATE");
        }

        Directory.CreateDirectory(output);
        File.Copy(Path.Combine(Carrier, "source-identity.json"), Path.Combine(Evidence, "build-" + name + "-source.json"));
        var report = BuildPipeline.BuildPlayer(new BuildPlayerOptions
        {
            scenes = new[] { Root + "/Bootstrap.unity" },
            locationPathName = Path.Combine(output, "ResourceAcceptance.exe"),
            target = BuildTarget.StandaloneWindows64,
            targetGroup = BuildTargetGroup.Standalone,
            options = development ? BuildOptions.Development : BuildOptions.None,
            extraScriptingDefines = defines.ToArray()
        });
        Verify();
        if (backend == ScriptingImplementation.IL2CPP && (!File.Exists(Path.Combine(output, "GameAssembly.dll")) ||
            !File.Exists(Path.Combine(output, "ResourceAcceptance_Data/il2cpp_data/Metadata/global-metadata.dat"))))
        {
            throw new InvalidDataException("Actual IL2CPP native binary and metadata are missing.");
        }

        var streaming = Path.Combine(output, "ResourceAcceptance_Data/StreamingAssets");
        var sourceStreaming = Path.Combine(Carrier, "Assets/StreamingAssets");
        var expected = Files(sourceStreaming).Where(x => !((string)x["Path"]).EndsWith(".meta", StringComparison.Ordinal)).ToArray();
        var actual = Files(streaming).ToArray();
        if (!JToken.DeepEquals(new JArray(expected), new JArray(actual)))
        {
            throw new InvalidDataException("Packaged StreamingAssets bytes/path closure differ from C02 deployment.");
        }

        var licenses = Path.Combine(output, "Licenses");
        Directory.CreateDirectory(licenses);
        foreach (var item in new[] {
            ("Assets/YLFramework/LICENSE.md", "YLFramework-MIT.md"),
            ("Assets/YLFramework/Plugins/BouncyCastle/SOURCE.txt", "BouncyCastle-SOURCE.txt") })
        {
            File.Copy(item.Item1, Path.Combine(licenses, item.Item2));
        }

        var jsonPackage = UnityEditor.PackageManager.PackageInfo.FindForAssembly(typeof(JsonConvert).Assembly);
        if (jsonPackage == null || jsonPackage.name != "com.unity.nuget.newtonsoft-json")
        {
            throw new InvalidDataException("Json.NET provenance is not the approved UPM package.");
        }

        File.Copy(Path.Combine(jsonPackage.resolvedPath, "LICENSE.md"), Path.Combine(licenses, "Json.NET-UPM-LICENSE.md"));
        File.Copy(Path.Combine(jsonPackage.resolvedPath, "Third Party Notices.md"), Path.Combine(licenses, "Json.NET-UPM-THIRD-PARTY-NOTICES.md"));
        var receipt = new JObject
        {
            ["Result"] = report.summary.result.ToString(),
            ["Errors"] = report.summary.totalErrors,
            ["Warnings"] = report.summary.totalWarnings,
            ["Seconds"] = report.summary.totalTime.TotalSeconds,
            ["Unity"] = Application.unityVersion,
            ["UguiVersion"] = ExpectedUgui,
            ["Backend"] = PlayerSettings.GetScriptingBackend(BuildTargetGroup.Standalone).ToString(),
            ["Options"] = report.summary.options.ToString(),
            ["Development"] = development,
            ["DevelopmentProfile"] = profile,
            ["ManagedStripping"] = PlayerSettings.GetManagedStrippingLevel(BuildTargetGroup.Standalone).ToString(),
            ["CppConfiguration"] = backend == ScriptingImplementation.IL2CPP ? PlayerSettings.GetIl2CppCompilerConfiguration(BuildTargetGroup.Standalone).ToString() : null,
            ["CandidateValidation"] = candidate,
            ["BuildGuid"] = report.summary.guid.ToString(),
            ["ExtraDefines"] = new JArray(defines),
            ["SourceIdentitySha256"] = Sha(Path.Combine(Carrier, "source-identity.json")),
            ["Target"] = report.summary.platform.ToString(),
            ["StreamingPayloadFiles"] = new JArray(actual),
            ["PlayerFiles"] = new JArray(Files(output)),
            ["Assemblies"] = JArray.FromObject(CompilationPipeline.GetAssemblies(AssembliesType.Player)
                .Select(x => new { x.name, Sources = x.sourceFiles, References = x.assemblyReferences.Select(y => y.name), x.compiledAssemblyReferences }))
        };
        File.WriteAllText(Path.Combine(output, "BuildReport.json"), receipt.ToString(Formatting.Indented));
        if (report.summary.result != BuildResult.Succeeded || report.summary.totalErrors != 0)
        {
            throw new InvalidOperationException("Actual Player build failed.");
        }

        Debug.Log("R810_BUILD_SUCCEEDED " + output);
    }

    public static void Export()
    {
        Verify();
        YLFramework.Editor.Distribution.ExportUnityPackage.Export(Path.Combine(Evidence, "YLFramework-8.10.unitypackage"));
    }

    public static void Verify()
    {
        var owner = JObject.Parse(File.ReadAllText(Path.Combine(Carrier, "r810-owner.json")));
        if ((string)owner["Carrier"] != Carrier || (string)owner["Purpose"] != "Resource-8.10-owned-carrier" ||
            Path.GetFullPath(Carrier) == Path.GetFullPath((string)owner["Repository"]))
        {
            throw new InvalidOperationException("Not an owned isolated carrier.");
        }

        if ((ExpectedUnity != "2022.3.62f3c1" && ExpectedUnity != "6000.3.27f1") ||
            (ExpectedUgui != "1.0.0" && ExpectedUgui != "2.0.0") ||
            (ExpectedUnity == "6000.3.27f1") != (ExpectedUgui == "2.0.0") ||
            Application.unityVersion != ExpectedUnity || EditorUserBuildSettings.activeBuildTarget != BuildTarget.StandaloneWindows64 ||
            EditorApplication.isPlayingOrWillChangePlaymode || EditorUtility.scriptCompilationFailed)
        {
            throw new InvalidOperationException("Wrong actual Unity state.");
        }

        var ugui = UnityEditor.PackageManager.PackageInfo.FindForAssembly(typeof(Button).Assembly);
        if (ugui == null || ugui.version != ExpectedUgui)
        {
            throw new InvalidOperationException("The actual UGUI package does not match the carrier identity.");
        }

        var identity = JArray.Parse(File.ReadAllText(Path.Combine(Carrier, "source-identity.json")));
        foreach (var file in identity)
        {
            if (Sha(Path.Combine(Carrier, (string)file["Path"])) != (string)file["Sha256"])
            {
                throw new InvalidDataException("Source identity changed: " + file["Path"]);
            }
        }

        var formal = Directory.GetFiles(Path.Combine(Carrier, "Assets/YLFramework"), "*", SearchOption.AllDirectories).Select(x => x.Substring(Carrier.Length + 1).Replace('\\', '/'));
        if (!new HashSet<string>(formal, StringComparer.Ordinal).SetEquals(identity.Select(x => (string)x["Path"]).Where(x => x.StartsWith("Assets/YLFramework/", StringComparison.Ordinal))))
        {
            throw new InvalidDataException("The exact formal source/meta closure changed.");
        }
    }

    private static ResourceGroup Group(string folder, string id, params (string file, string address)[] assets) => new()
    {
        id = id,
        packing = ResourcePacking.PerAsset,
        selections = assets.Select(x => new ResourceSelection { guid = AssetDatabase.AssetPathToGUID(folder + "/" + x.file), address = x.address }).ToList()
    };
    private static IEnumerable<JObject> Files(string root) => Directory.GetFiles(root, "*", SearchOption.AllDirectories).OrderBy(x => x, StringComparer.Ordinal)
        .Select(x => new JObject { ["Path"] = x.Substring(root.Length + 1).Replace('\\', '/'), ["Length"] = new FileInfo(x).Length, ["Sha256"] = Sha(x) });
    private static string Sha(string path)
    {
        using var hash = SHA256.Create();
        using var file = File.OpenRead(path);
        return BitConverter.ToString(hash.ComputeHash(file)).Replace("-", "").ToLowerInvariant();
    }
    private static void Write(string name, object value) => File.WriteAllText(Path.Combine(Evidence, name), JsonConvert.SerializeObject(value, Formatting.Indented));
    private static void Texture(string path, int size, bool sprite)
    {
        var texture = new Texture2D(size, size, TextureFormat.RGBA32, false);
        texture.SetPixels(Enumerable.Repeat(Color.green, size * size).ToArray());
        texture.Apply();
        File.WriteAllBytes(path, texture.EncodeToPNG());
        Object.DestroyImmediate(texture);
        AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceSynchronousImport);
        var importer = (TextureImporter)AssetImporter.GetAtPath(path);
        importer.textureType = sprite ? TextureImporterType.Sprite : TextureImporterType.Default;
        importer.spriteImportMode = SpriteImportMode.Single;
        importer.mipmapEnabled = false;
        importer.textureCompression = TextureImporterCompression.Uncompressed;
        importer.npotScale = TextureImporterNPOTScale.None;
        importer.SaveAndReimport();
    }
    private static void Wave(string path)
    {
        using (var writer = new BinaryWriter(File.Create(path)))
        {
            const int Count = 22050;
            writer.Write(Encoding.ASCII.GetBytes("RIFF"));
            writer.Write(36 + Count * 2);
            writer.Write(Encoding.ASCII.GetBytes("WAVEfmt "));
            writer.Write(16);
            writer.Write((short)1);
            writer.Write((short)1);
            writer.Write(22050);
            writer.Write(44100);
            writer.Write((short)2);
            writer.Write((short)16);
            writer.Write(Encoding.ASCII.GetBytes("data"));
            writer.Write(Count * 2);
            for (var i = 0; i < Count; i++)
            {
                writer.Write((short)(Math.Sin(i * .03) * 1000));
            }
        }
        AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceSynchronousImport);
        var importer = (AudioImporter)AssetImporter.GetAtPath(path);
        var settings = importer.defaultSampleSettings;
        settings.loadType = AudioClipLoadType.DecompressOnLoad;
        settings.compressionFormat = AudioCompressionFormat.PCM;
        settings.sampleRateSetting = AudioSampleRateSetting.PreserveSampleRate;
        importer.defaultSampleSettings = settings;
        importer.SaveAndReimport();
    }
    private sealed class Keys : IResourceContentKeyProvider
    {
        internal int Copies;
        public bool TryCopyKey(string projectId, string keyId, byte[] destination)
        {
            var key = Convert.FromBase64String(Environment.GetEnvironmentVariable("YL_RESOURCE_CONTENT_KEY"));
            try
            {
                Buffer.BlockCopy(key, 0, destination, 0, 32);
                Copies++;
                return true;
            }
            finally
            {
                Array.Clear(key, 0, key.Length);
            }
        }
    }
