    private void Start()
    {
        try
        {
            Require(Application.platform == RuntimePlatform.WindowsPlayer && IntPtr.Size == 8 && !Debug.isDebugBuild,
                "Only a real Windows64 non-Development Player may run this consumer.");
            Require((expectedUnity == "2022.3.62f3c1" || expectedUnity == "6000.3.27f1") &&
                Application.unityVersion == expectedUnity && SystemInfo.graphicsDeviceType != GraphicsDeviceType.Null,
                "Wrong actual engine or missing graphics context.");
            _directory = Path.GetFullPath(Argument("--s9n-directory"));
            _token = Argument("--s9n-token");
            Require(Regex.IsMatch(_token, "^[0-9a-f]{32}$"), "A fresh IPC token is required.");
            _deadline = DateTime.Parse(Argument("--s9n-deadline"), CultureInfo.InvariantCulture,
                DateTimeStyles.RoundtripKind).ToUniversalTime();
            Require(_deadline > DateTime.UtcNow && (_deadline - DateTime.UtcNow).TotalSeconds <= 180,
                "Invalid external deadline.");
            Require(deliveryRoot == @"G:\Unity\U63-20261027-c63a1" &&
                receiverProject.StartsWith(deliveryRoot + "\\", StringComparison.Ordinal) &&
                receiverProject.IndexOf("CGJ2024", StringComparison.OrdinalIgnoreCase) < 0 &&
                _directory.StartsWith(Path.Combine(deliveryRoot, "Runs") + "\\", StringComparison.Ordinal),
                "Never run in Source or write outside this batch Runs root.");
            RejectLinks(_directory);
            Require(Directory.Exists(_directory) && fullTemplate && colorPrefab, "Missing owned IPC directory or formal prefab bindings.");
            Require(!FindObjectsByType<YLFrameworkRoot>(FindObjectsInactive.Include, FindObjectsSortMode.None).Any() &&
                !FindObjectsByType<EventSystem>(FindObjectsInactive.Include, FindObjectsSortMode.None).Any() &&
                ModuleManager.AllModules.Count == 0 && ModuleManager.AllBaseModules.Count == 0,
                "A foreign UI root, EventSystem or Module already exists; do not adopt it.");
            Application.runInBackground = true;
            Application.logMessageReceived += Observe;
        }
        catch (Exception error)
        {
            Fail(error);
            Finish(false);
        }
    }

    private void LateUpdate()
    {
        if (_ended)
        {
            return;
        }
        try
        {
            if (DateTime.UtcNow >= _deadline)
            {
                throw new TimeoutException("Owned native-input application deadline expired.");
            }
            if (_errors.Count != 0 && _failure == null)
            {
                throw new InvalidOperationException("Unexpected Unity error(s): " + string.Join("\n", _errors));
            }
            if (_cleanupStarted)
            {
                if (!_full && (_module == null || _module.State == ModuleState.Closed) &&
                    !FindObjectsByType<EventSystem>(FindObjectsInactive.Include, FindObjectsSortMode.None).Any())
                {
                    Finish(_failure == null);
                }
                return;
            }
            if (_state == 20)
            {
                for (var budget = 0; budget < 64; budget++)
                {
                    if (_captureRoutines.Count == 0)
                    {
                        BeginCleanup();
                        return;
                    }
                    var routine = _captureRoutines.Peek();
                    if (!routine.MoveNext())
                    {
                        _captureRoutines.Pop();
                    }
                    else if (routine.Current is IEnumerator child)
                    {
                        _captureRoutines.Push(child);
                    }
                    else
                    {
                        return;
                    }
                }
                throw new InvalidOperationException("Capture coroutine exceeded its per-frame budget.");
            }
            if (_state == 0)
            {
                var path = Path.Combine(_directory, "start.json");
                if (!File.Exists(path))
                {
                    return;
                }
                _identity = Read(path);
                var executable = new StringBuilder(32768);
                Require(GetProcessTimes(GetCurrentProcess(), out var created, out _, out _, out _) &&
                    GetModuleFileName(IntPtr.Zero, executable, executable.Capacity) > 0, "Cannot establish native startup identity.");
                Require((string)_identity["Schema"] == "Stage9NativeInput-v1" && (string)_identity["Token"] == _token &&
                    (uint)_identity["Pid"] == GetCurrentProcessId() &&
                    (long)_identity["StartTicks"] == DateTime.FromFileTimeUtc(created).Ticks &&
                    (string)_identity["Executable"] == executable.ToString() &&
                    (string)_identity["Project"] == receiverProject && (string)_identity["Unity"] == expectedUnity &&
                    (string)_identity["BuildGuid"] == Application.buildGUID &&
                    (string)_identity["DeadlineUtc"] == Argument("--s9n-deadline"), "Startup/build/IPC identity mismatch.");
                var mode = (string)_identity["Mode"];
                _remainingMode = (string)_identity["Suite"] == "Remaining";
                _captureMode = _remainingMode || (string)_identity["Suite"] == "Capture";
                Require((mode == "ExpectedLegacy" || mode == "ExpectedNew" || mode == "ExpectedBoth") &&
                    LegacyInput == (mode != "ExpectedNew") && NewInput == (mode != "ExpectedLegacy"),
                    "Actual compiled Legacy/New/Both macros do not match the requested mode.");
                Require(_directory == Path.Combine(deliveryRoot, "Runs", Path.GetFileName(receiverProject) +
                    "-native-" + mode + "-" + (string)_identity["Run"]), "Wrong isolated run directory.");
                _full = Instantiate(fullTemplate);
                _root = _full.GetComponentInChildren<UIRootContext>(true);
                Require(_root && _root.GetLayer(0) && _root.GetLayer(4) &&
                    _full.GetComponentInChildren<UITipsPresenter>(true), "The formal complete UI template is required.");
                _module = new Stage9NativeInputModule { PanelRoot = _root };
                _startup = ModuleManager.StartUpModule(_module);
                _state = 1;
            }
            else if (_state == 1 && _startup.IsDone)
            {
                Require(_startup.Exception == null && _module.State == ModuleState.Running,
                    "Owned Module startup failed: " + _startup.Exception);
                TryAuthorizedActivation();
                if (_captureMode)
                {
                    _captureRoutines.Push(_remainingMode ? RunRemaining() : RunCapture());
                    _state = 20;
                    return;
                }
                _initial = Color.cyan;
                _panel = _module.LoadPanel<ColorSelectorWindow>(colorPrefab,
                    ColorSelectorWindow.Arguments(_initial, OnColorChanged), overrides: new PanelOverrides { CacheOnClose = false });
                Require(_panel && _panel.Root == _root && _panel.CurrentColor == _initial, "Formal ColorSelectorWindow failed to open.");
                _selector = _panel.GetComponentInChildren<ColorSelector>(true);
                var buttons = _panel.GetComponentsInChildren<Button>(true);
                Require(_selector && buttons.Length == 1, "Expected the formal wheel and its one close Button.");
                _close = buttons[0];
                _close.onClick.AddListener(OnButtonClicked);
                _stage = 1;
                _state = 2;
            }
            else if (_state == 2)
            {
                if (!Application.isFocused || !HasOwnedForeground() || !EventSystem.current || !EventSystem.current.currentInputModule ||
                    !_panel.IsVisible || _panel.IsInputBlocked)
                {
                    return;
                }
                var expectedModule = (string)_identity["Mode"] == "ExpectedLegacy" ?
                    "UnityEngine.EventSystems.StandaloneInputModule" : "UnityEngine.InputSystem.UI.InputSystemUIInputModule";
                Require(ModuleName == expectedModule, "Production PanelManager/InputSystemBridge selected the wrong module.");
                PrepareStage();
            }
            else if (_state >= 3 && _state <= 5)
            {
                Require(DateTime.UtcNow < DateTime.Parse(_stageDeadline, CultureInfo.InvariantCulture,
                    DateTimeStyles.RoundtripKind).ToUniversalTime(), "Application gesture phase deadline expired.");
                ProcessPhase();
            }
            else if (_state == 6)
            {
                _stage = 2;
                _state = 2;
            }
        }
        catch (Exception error)
        {
            Fail(error);
            BeginCleanup();
            if (DateTime.UtcNow >= _deadline)
            {
                Finish(false);
            }
        }
    }

    private void PrepareStage()
    {
        Canvas.ForceUpdateCanvases();
        var target = _stage == 1 ? _selector.gameObject : _close.gameObject;
        var rect = (RectTransform)target.transform;
        var canvas = rect.GetComponentInParent<Canvas>().rootCanvas;
        var local = rect.rect.center;
        if (_stage == 1)
        {
            local.y += Mathf.Min(rect.rect.width, rect.rect.height) * 0.5f * 0.88f;
        }
        _point = RectTransformUtility.WorldToScreenPoint(
            canvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : canvas.worldCamera, rect.TransformPoint(local));
        Require(_point.x > 1 && _point.y > 1 && _point.x < Screen.width - 1 && _point.y < Screen.height - 1,
            "The actual rendered target is outside the Player client.");
        _witness = target.AddComponent<Stage9NativeInputWitness>();
        _witness.Observed = OnPointerObserved;
        _mouseOnUI = false;
        _stageDeadline = new DateTime(Math.Min(_deadline.Ticks, DateTime.UtcNow.AddSeconds(20).Ticks), DateTimeKind.Utc).ToString("O");
        _state = 3;
        var ready = Envelope(_stage, "move");
        ready["Kind"] = _stage == 1 ? "color" : "close-button";
        ready["X"] = _point.x;
        ready["Y"] = _point.y;
        ready["CoordinateOrigin"] = "Unity client bottom-left pixels; driver must match physical client size exactly";
        ready["ScreenWidth"] = Screen.width;
        ready["ScreenHeight"] = Screen.height;
        ready["ScreenDpi"] = Screen.dpi;
        ready["ScreenResolution"] = Screen.currentResolution.ToString();
        ready["ApplicationFocused"] = Application.isFocused;
        ready["LegacyInput"] = LegacyInput;
        ready["NewInput"] = NewInput;
        ready["Backend"] = Backend;
        ready["Module"] = ModuleName;
        ready["SendNavigationEvents"] = EventSystem.current.sendNavigationEvents;
        ready["SelectedObject"] = EventSystem.current.currentSelectedGameObject ? EventSystem.current.currentSelectedGameObject.name : null;
        ready["PointerQuery"] = "UIPointer.CheckMouseOnUI via production bridge; no probe-side Input System references";
        ready["Target"] = target.name;
        Write("ready-" + _stage + ".json", ready);
    }

    private void ProcessPhase()
    {
        var phase = _state == 3 ? "move" : _state == 4 ? "down" : "up";
        var path = Path.Combine(_directory, "command-" + _stage + "-" + phase + ".json");
        if (!File.Exists(path))
        {
            return;
        }
        var command = Read(path);
        foreach (var field in new[] { "Schema", "Token", "Run", "Pid", "StartTicks", "Project", "BuildGuid",
            "BuildRun", "Mode", "Unity", "DeadlineUtc", "BuildReportSha256" })
        {
            Require(JToken.DeepEquals(command[field], _identity[field]), "Command IPC identity mismatch: " + field);
        }
        Require((int)command["Stage"] == _stage && (string)command["Phase"] == phase &&
            (string)command["StageDeadlineUtc"] == _stageDeadline, "Wrong or stale gesture phase.");
        var window = new IntPtr((long)command["Hwnd"]);
        GetWindowThreadProcessId(window, out var pid);
        Require(window != IntPtr.Zero && IsWindow(window) && !IsIconic(window) && GetForegroundWindow() == window &&
            pid == (uint)(int)_identity["Pid"] && Application.isFocused && (_hwnd == 0 || _hwnd == window.ToInt64()),
            "Command HWND is not this exact foreground Player.");
        _hwnd = window.ToInt64();
        var mouse = UIPointer.CheckMouseOnUI();
        // Enter may precede witness attachment when the OS cursor was already over this Graphic.
        // Down/Up below still require the actual target callbacks, not a hover count or a synthetic raycast.
        if (phase == "move" && !mouse ||
            phase == "down" && _witness.Down == 0 || phase == "up" && _witness.Up == 0)
        {
            return;
        }
        Require(mouse || phase == "up" && _stage == 2 && _mouseOnUI,
            "Production mouse query did not hit the actual UI.");
        _mouseOnUI |= mouse;
        if (phase != "move")
        {
            Require(_witness.Down == 1 && _witness.Up == (phase == "up" ? 1 : 0), "Native pointer callbacks were duplicated or out of order.");
            Require(_colorChanges == 1 && _changed != _initial &&
                (_stage == 2 && phase == "up" || _panel && _panel.CurrentColor == _changed),
                "The real color gesture must change color and notify exactly once.");
            Require(_buttonClicks == (_stage == 2 && phase == "up" ? 1 : 0), "The standard Button must invoke exactly once, only after its real Up.");
        }
        if (phase == "up")
        {
            Require(_witness.Click == 1, "The module did not deliver exactly one completed UGUI click.");
        }
        var ack = Envelope(_stage, phase);
        ack["Hwnd"] = _hwnd;
        ack["MouseOnUI"] = _mouseOnUI;
        ack["MouseOnUIAtResponse"] = mouse;
        ack["PhaseObserved"] = true;
        ack["Observation"] = phase == "move" ? "production-mouse-query" : "actual-UGUI-callback";
        ack["Down"] = _witness.Down;
        ack["Up"] = _witness.Up;
        ack["Click"] = _witness.Click;
        ack["ColorChanges"] = _colorChanges;
        ack["ButtonClicks"] = _buttonClicks;
        Write("ack-" + _stage + "-" + phase + ".json", ack);
        if (phase == "up")
        {
            _cases.Add(new JObject { ["Stage"] = _stage, ["Kind"] = _stage == 1 ? "color" : "close-button",
                ["Down"] = _witness.Down, ["Up"] = _witness.Up, ["Click"] = _witness.Click,
                ["MouseOnUI"] = _mouseOnUI, ["Module"] = ModuleName });
            _witness.Observed = null;
            if (_stage == 2)
            {
                Require(!_panel || !_panel.IsLoaded, "The formal close Button did not close its own panel.");
                BeginCleanup();
            }
            else
            {
                _state = 6;
            }
        }
        else
        {
            _state++;
        }
    }

    private void OnPointerObserved(string kind, PointerEventData data)
    {
        try
        {
            Require(!_cleanupStarted && _state >= 3 && _state <= 5, "Unexpected native input outside an armed gesture.");
            if (kind != "enter")
            {
                Require(data.button == PointerEventData.InputButton.Left && Vector2.Distance(data.position, _point) <= 3,
                    "Pointer button/position does not match this native gesture.");
            }
            var mouse = UIPointer.CheckMouseOnUI();
            _mouseOnUI |= mouse;
            _callbacks.Add(new JObject
            {
                ["Stage"] = _stage, ["Kind"] = kind, ["Utc"] = DateTime.UtcNow.ToString("O"),
                ["Frame"] = Time.frameCount, ["PointerId"] = data.pointerId, ["Button"] = data.button.ToString(),
                ["X"] = data.position.x, ["Y"] = data.position.y, ["PointerDataType"] = data.GetType().FullName,
                ["Module"] = ModuleName, ["MouseOnUI"] = mouse, ["Focused"] = Application.isFocused,
                ["RaycastTarget"] = data.pointerCurrentRaycast.gameObject ? data.pointerCurrentRaycast.gameObject.name : ""
            });
        }
        catch (Exception error)
        {
            Fail(error);
        }
    }
    private void OnColorChanged(Color value)
    {
        _colorChanges++;
        _changed = value;
        _businessCallbacks.Add(new JObject { ["Kind"] = "color-changed", ["Stage"] = _stage,
            ["Utc"] = DateTime.UtcNow.ToString("O"), ["Count"] = _colorChanges,
            ["Color"] = new JArray(value.r, value.g, value.b, value.a) });
    }
    private void OnButtonClicked()
    {
        _buttonClicks++;
        _businessCallbacks.Add(new JObject { ["Kind"] = "button-onClick", ["Stage"] = _stage,
            ["Utc"] = DateTime.UtcNow.ToString("O"), ["Count"] = _buttonClicks });
    }

    private bool HasOwnedForeground()
    {
        var window = GetForegroundWindow();
        if (window == IntPtr.Zero || !IsWindow(window) || IsIconic(window))
        {
            return false;
        }
        GetWindowThreadProcessId(window, out var pid);
        return pid == (uint)(int)_identity["Pid"] && (_hwnd == 0 || window.ToInt64() == _hwnd);
    }

    private void TryAuthorizedActivation()
    {
        if (_activationChecked || (bool?)_identity["ActivationAuthorized"] != true)
        {
            return;
        }
        _activationChecked = true;
        var owned = IntPtr.Zero;
        var cursor = IntPtr.Zero;
        var count = 0;
        while ((cursor = FindWindowEx(IntPtr.Zero, cursor, "UnityWndClass", null)) != IntPtr.Zero)
        {
            GetWindowThreadProcessId(cursor, out var pid);
            if (pid == (uint)(int)_identity["Pid"] && IsWindowVisible(cursor) && !IsIconic(cursor))
            {
                owned = cursor;
                count++;
            }
        }
        Require(count == 1 && IsWindow(owned), "Exactly one own visible Player window required for ordinary activation.");
        var before = GetForegroundWindow();
        var requested = before != owned;
        var accepted = !requested || SetForegroundWindow(owned);
        var result = Envelope(0, "ordinary-activation");
        result["Hwnd"] = owned.ToInt64();
        result["BeforeForegroundHwnd"] = before.ToInt64();
        result["AfterForegroundHwnd"] = GetForegroundWindow().ToInt64();
        result["RequestMade"] = requested;
        result["RequestReturned"] = accepted;
        result["Accepted"] = accepted && GetForegroundWindow() == owned;
        result["Caller"] = "Exact owned Player process";
        Write("activation-request.json", result);
        Require((bool)result["Accepted"], "Windows declined the one authorized ordinary foreground request from this owned Player; no input requested.");
    }

    private static IEnumerator Frames(int count)
    {
        for (var i = 0; i < count; i++)
        {
            yield return null;
        }
    }

    private GameObject CaptureSource(string name)
    {
        var source = new GameObject(name, typeof(RectTransform));
        source.SetActive(false);
        ((RectTransform)source.transform).sizeDelta = new Vector2(600, 440);
        source.AddComponent<Stage9NativeCapturePanel>();
        _captureObjects.Add(source);
        return source;
    }

    private static GameObject Control(GameObject parent, string name, Vector2 size, params Type[] types)
    {
        var control = new GameObject(name, typeof(RectTransform), typeof(Image));
        control.transform.SetParent(parent.transform, false);
        ((RectTransform)control.transform).sizeDelta = size;
        foreach (var type in types)
        {
            control.AddComponent(type);
        }
        return control;
    }

    private Stage9NativeCapturePanel OpenCapture(GameObject source, UIRootContext root = null, bool modal = false)
    {
        var panel = _module.LoadPanel<Stage9NativeCapturePanel>(source, root: root ? root : _root,
            overrides: new PanelOverrides { InstanceMode = PanelInstanceMode.Multiple, CacheOnClose = true, Modal = modal, Layer = modal ? 1 : 0 });
        panel.gameObject.SetActive(true);
        return panel;
    }

    private static Vector2 CapturePoint(Transform target, Vector2 offset = default)
    {
        Canvas.ForceUpdateCanvases();
        var canvas = target.GetComponentInParent<Canvas>().rootCanvas;
        return RectTransformUtility.WorldToScreenPoint(canvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : canvas.worldCamera,
            target.TransformPoint(offset));
    }

    private void CapturePassed(string name, JObject details = null)
    {
        _cases.Add(new JObject { ["Name"] = name, ["Passed"] = true, ["Details"] = details ?? new JObject() });
        Debug.Log("STAGE9_CAPTURE_CASE_PASS " + name);
    }

    private IEnumerator CaptureInput(string name, string action, Vector2 point, int button = 0, Func<bool> observed = null, int key = 0)
    {
        while (!Application.isFocused || !HasOwnedForeground() || !EventSystem.current || !EventSystem.current.currentInputModule)
        {
            yield return null;
        }
        Require(point.x > 1 && point.y > 1 && point.x < Screen.width - 1 && point.y < Screen.height - 1,
            "Capture target is outside the actual Player client: " + name);
        _inputSteps++;
        _stageDeadline = new DateTime(Math.Min(_deadline.Ticks, DateTime.UtcNow.AddSeconds(20).Ticks), DateTimeKind.Utc).ToString("O");
        var ready = Envelope(_inputSteps, "input");
        ready["Case"] = name;
        ready["Action"] = action;
        ready["Button"] = button;
        ready["Key"] = key;
        ready["X"] = point.x;
        ready["Y"] = point.y;
        ready["ScreenWidth"] = Screen.width;
        ready["ScreenHeight"] = Screen.height;
        ready["ApplicationFocused"] = Application.isFocused;
        ready["Backend"] = Backend;
        ready["LegacyInput"] = LegacyInput;
        ready["NewInput"] = NewInput;
        ready["Module"] = ModuleName;
        Write("ready-" + _inputSteps + ".json", ready);
        var commandPath = Path.Combine(_directory, "command-" + _inputSteps + "-input.json");
        while (!File.Exists(commandPath))
        {
            Require(DateTime.UtcNow < DateTime.Parse(_stageDeadline, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind),
                "Capture input phase expired: " + name + "/" + action);
            yield return null;
        }
        var command = Read(commandPath);
        foreach (var field in new[] { "Schema", "Token", "Run", "Pid", "StartTicks", "Project", "BuildGuid", "BuildRun", "Mode", "Unity", "Suite", "DeadlineUtc", "BuildReportSha256" })
        {
            Require(JToken.DeepEquals(command[field], _identity[field]), "Capture command identity mismatch: " + field);
        }
        var window = new IntPtr((long)command["Hwnd"]);
        GetWindowThreadProcessId(window, out var pid);
        Require((int)command["Stage"] == _inputSteps && (string)command["Phase"] == "input" &&
            (string)command["Action"] == action && (int)command["Button"] == button && (string)command["StageDeadlineUtc"] == _stageDeadline &&
            (!_remainingMode || (int)command["Key"] == key) &&
            window != IntPtr.Zero && IsWindow(window) && !IsIconic(window) && GetForegroundWindow() == window &&
            pid == (uint)(int)_identity["Pid"] && Application.isFocused && (_hwnd == 0 || _hwnd == window.ToInt64()),
            "Capture command is not this exact foreground Player/stage.");
        _hwnd = window.ToInt64();
        yield return Frames(2);
        if (key != 0)
        {
            var state = Envelope(_inputSteps, "key-observed");
            state["Key"] = key;
            state["SelectedObject"] = EventSystem.current.currentSelectedGameObject ? EventSystem.current.currentSelectedGameObject.name : null;
            state["SendNavigationEvents"] = EventSystem.current.sendNavigationEvents;
#if !ENABLE_IL2CPP
            if (NewInput)
            {
                var keyboardType = Type.GetType("UnityEngine.InputSystem.Keyboard, Unity.InputSystem", true);
                var keyboard = keyboardType.GetProperty("current").GetValue(null);
                state["KeyboardPresent"] = keyboard != null;
                if (keyboard != null)
                {
                    var property = key switch { 13 => "enterKey", 27 => "escapeKey", 37 => "leftArrowKey", 38 => "upArrowKey", 39 => "rightArrowKey", 40 => "downArrowKey", 49 => "digit1Key", 50 => "digit2Key", 51 => "digit3Key", _ => throw new InvalidOperationException("Unknown diagnostic key.") };
                    var control = keyboardType.GetProperty(property).GetValue(keyboard);
                    state["NativeKeyboard"] = (bool)keyboardType.GetProperty("native").GetValue(keyboard);
                    state["ControlPressed"] = (bool)control.GetType().GetProperty("isPressed").GetValue(control);
                }
                var nativeModule = EventSystem.current.currentInputModule;
                var reference = nativeModule.GetType().GetProperty("move").GetValue(nativeModule);
                var moveAction = reference.GetType().GetProperty("action").GetValue(reference);
                state["MoveActionEnabled"] = (bool)moveAction.GetType().GetProperty("enabled").GetValue(moveAction);
                var value = moveAction.GetType().GetMethod("ReadValueAsObject", Type.EmptyTypes).Invoke(moveAction, null);
                if (value is Vector2 move)
                {
                    state["MoveX"] = move.x;
                    state["MoveY"] = move.y;
                }
            }
#endif
            Write("key-observed-" + _inputSteps + ".json", state);
        }
        while (observed != null && !observed())
        {
            Require(DateTime.UtcNow < DateTime.Parse(_stageDeadline, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind),
                "No actual native callback: " + name + "/" + action);
            yield return null;
        }
        var ack = Envelope(_inputSteps, "input");
        ack["Hwnd"] = _hwnd;
        ack["Action"] = action;
        ack["Case"] = name;
        ack["Key"] = key;
        ack["PhaseObserved"] = true;
        Write("ack-" + _inputSteps + "-input.json", ack);
    }

    private IEnumerator CaptureClick(string name, Vector2 point, Stage9NativeCaptureWitness witness, int button = 0)
    {
        var down = witness.Downs[button];
        var clicks = witness.Clicks[button];
        yield return CaptureInput(name, "move", point, button);
        yield return CaptureInput(name, "down", point, button, () => witness.Downs[button] > down);
        yield return CaptureInput(name, "up", point, button, () => witness.Clicks[button] > clicks);
        Require(witness.Clicks[button] == clicks + 1, "Fresh gesture did not click exactly once: " + name);
    }

    private void BeginCleanup()
    {
        if (_cleanupStarted)
        {
            return;
        }
        _cleanupStarted = true;
        while (_captureRoutines.Count != 0)
        {
            try
            {
                (_captureRoutines.Pop() as IDisposable)?.Dispose();
            }
            catch (Exception error)
            {
                Fail(error);
            }
        }
        if (_witness)
        {
            _witness.Observed = null;
        }
        if (_close)
        {
            _close.onClick.RemoveListener(OnButtonClicked);
        }
        try
        {
            if (_module != null && _module.State != ModuleState.Closed)
            {
                _module.Close();
            }
        }
        catch (Exception error)
        {
            Fail(error);
        }
        finally
        {
            if (_full)
            {
                Destroy(_full);
            }
            foreach (var owned in _captureObjects)
            {
                if (owned)
                {
                    Destroy(owned);
                }
            }
        }
    }

    private void Finish(bool passed)
    {
        if (_ended)
        {
            return;
        }
        _ended = true;
        var cleanup = _cleanupStarted && !_full && _module != null && _module.State == ModuleState.Closed &&
            !ModuleManager.AllModules.Values.Any(item => ReferenceEquals(item, _module)) &&
            !FindObjectsByType<EventSystem>(FindObjectsInactive.Include, FindObjectsSortMode.None).Any();
        passed &= cleanup && _failure == null && _errors.Count == 0 && _cases.Count == (_remainingMode ? (NewInput ? 5 : 4) : _captureMode ? 13 : 2);
        if (_identity != null)
        {
            var result = Envelope(_captureMode ? _inputSteps + 1 : 3, "complete");
            result["StageDeadlineUtc"] = _deadline.ToString("O");
            result["BusinessPassed"] = passed;
            result["OwnedCleanupCompleted"] = cleanup;
            result["Development"] = Debug.isDebugBuild;
            result["Backend"] = Backend;
            result["LegacyInput"] = LegacyInput;
            result["NewInput"] = NewInput;
            result["Graphics"] = SystemInfo.graphicsDeviceType.ToString();
            result["GraphicsDevice"] = SystemInfo.graphicsDeviceName;
            result["ScreenWidth"] = Screen.width;
            result["ScreenHeight"] = Screen.height;
            result["ColorChanges"] = _colorChanges;
            result["ButtonClicks"] = _buttonClicks;
            result["InitialColor"] = new JArray(_initial.r, _initial.g, _initial.b, _initial.a);
            result["ChangedColor"] = new JArray(_changed.r, _changed.g, _changed.b, _changed.a);
            result["Cases"] = _cases;
            result["CaptureProtocol"] = _captureMode ? "B06Capture-v1" : null;
            result["RemainingProtocol"] = _remainingMode ? "B06Remaining-v1" : null;
            result["InputSteps"] = _inputSteps;
            result["NativeCallbacks"] = _callbacks;
            result["BusinessCallbacks"] = _businessCallbacks;
            result["Errors"] = new JArray(_errors);
            result["Failure"] = _failure;
            result["Scope"] = _remainingMode ? "Real OS keyboard/text and held geometry; native mouse API removal/re-addition only, not physical hotplug" :
                _captureMode ? "Real OS capture/control/lifecycle/camera/root cases; keyboard/device/shared-actions remain separate" :
                "Two real OS left-click gestures only; native SendInput records are owned by the external driver";
            Write("result.json", result);
        }
        Debug.Log(passed ? "STAGE9_NATIVE_INPUT_BUSINESS_PASS" : "STAGE9_NATIVE_INPUT_FAILED");
        Application.Quit(passed ? 0 : 1);
    }
    private void Fail(Exception error)
    {
        _failure = _failure == null ? error.ToString() : _failure + "\n" + error;
        Debug.LogException(error);
    }
    private void Observe(string message, string trace, LogType type)
    {
        if (type != LogType.Error && type != LogType.Exception && type != LogType.Assert)
        {
            return;
        }
        var text = type + ": " + message + "\n" + trace;
        _errors.Add(text);
        if (_ended)
        {
            File.AppendAllText(Path.Combine(_directory, "late-errors.log"), text + "\n");
        }
    }
    private void OnDestroy() => Application.logMessageReceived -= Observe;

    private JObject Envelope(int stage, string phase)
    {
        var value = (JObject)_identity.DeepClone();
        value["Stage"] = stage;
        value["Phase"] = phase;
        value["StageDeadlineUtc"] = _stageDeadline;
        value["Utc"] = DateTime.UtcNow.ToString("O");
        return value;
    }
    private JObject Read(string path)
    {
        RejectLinks(path);
        using var stream = File.OpenText(path);
        using var reader = new JsonTextReader(stream) { DateParseHandling = DateParseHandling.None };
        return JObject.Load(reader);
    }
    private void Write(string name, JObject value)
    {
        var path = Path.Combine(_directory, name);
        RejectLinks(path);
        var pending = path + ".pending";
        using (var stream = new FileStream(pending, FileMode.CreateNew, FileAccess.Write, FileShare.None))
        using (var writer = new StreamWriter(stream, new UTF8Encoding(false)))
        {
            writer.Write(value.ToString(Formatting.Indented));
        }
        File.Move(pending, path);
    }
    private static void RejectLinks(string path)
    {
        for (var cursor = path; !string.IsNullOrEmpty(cursor); cursor = Path.GetDirectoryName(cursor))
        {
            if ((File.Exists(cursor) || Directory.Exists(cursor)) &&
                (File.GetAttributes(cursor) & FileAttributes.ReparsePoint) != 0)
            {
                throw new IOException("Linked runtime IPC path rejected: " + cursor);
            }
        }
    }
    private static string Argument(string name)
    {
        var args = Environment.GetCommandLineArgs();
        var index = Array.IndexOf(args, name);
        Require(index >= 0 && index + 1 < args.Length && Array.LastIndexOf(args, name) == index,
            "Exactly one argument required: " + name);
        return args[index + 1];
    }
    private static void Require(bool condition, string message)
    {
        if (!condition)
        {
            throw new InvalidOperationException(message);
        }
    }
}
