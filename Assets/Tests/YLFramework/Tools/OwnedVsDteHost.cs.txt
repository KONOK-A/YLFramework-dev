using System;
using System.Collections.Generic;
using System.IO;
using System.IO.Pipes;
using System.Linq;
using System.Runtime.InteropServices;
using System.Security.AccessControl;
using System.Security.Cryptography;
using System.Security.Principal;
using System.Threading.Tasks;
using System.Web.Script.Serialization;
using System.Windows.Forms;
using EnvDTE;
using EnvDTE80;
using Process = System.Diagnostics.Process;

internal static class OwnedVsDteHost
{
    private static readonly JavaScriptSerializer Json = new JavaScriptSerializer { MaxJsonLength = 8 * 1024 * 1024 };
    private static Dictionary<string, object> _config;
    private static DTE2 _dte;
    private static Control _dispatch;
    private static uint _pid;
    private static bool _owned;
    private static bool _stopping;
    private static DateTime _deadline;
    private static DateTime _expires;

    private static string Config(string key) => (string)_config[key];
    private static string[] Probes => new[]
    {
        Path.Combine(Config("Project"), "Assets", "AnalyzerConsumers", "IdeDiagnosticProbe.cs"),
        Path.Combine(Config("Project"), "Assets", "IdeDefaultProbe.cs")
    };

    [STAThread]
    private static int Main(string[] args)
    {
        if (args.Length == 1 && args[0] == "--self-test")
        {
            return SelfTest();
        }
        if (args.Length != 1)
        {
            throw new ArgumentException("An explicit owned host configuration is required.");
        }
        _config = Json.Deserialize<Dictionary<string, object>>(File.ReadAllText(args[0]));
        VerifyConfiguration(args[0]);
        var previousPids = new HashSet<int>(Process.GetProcessesByName("devenv").Select(process => process.Id));
        var requested = DateTime.UtcNow;
        _expires = requested.AddSeconds(Convert.ToInt32(_config["LifetimeSeconds"]));
        _dispatch = new Control();
        var handle = _dispatch.Handle;
        OleMessageFilter previous;
        CoRegisterMessageFilter(new MessageFilter(), out previous);
        try
        {
            _deadline = DateTime.UtcNow.AddMinutes(2);
            _dte = (DTE2)Activator.CreateInstance(Type.GetTypeFromProgID("VisualStudio.DTE.17.0", true));
            GetWindowThreadProcessId(_dte.MainWindow.HWnd, out _pid);
            var process = Process.GetProcessById((int)_pid);
            _owned = CanOwn((int)_pid, process.StartTime.ToUniversalTime(), process.MainModule.FileName,
                _dte.UserControl, previousPids, requested, DateTime.UtcNow, Config("VsExecutable"));
            if (!_owned)
            {
                throw new InvalidOperationException("Refuse an existing, user-controlled, stale or unexpected VS instance.");
            }
            if (_dte.Solution.IsOpen)
            {
                throw new InvalidOperationException("A newly created VS unexpectedly opened a solution; no documents will be closed by the host.");
            }
            _dte.SuppressUI = true;
            _dte.MainWindow.Visible = false;
            WriteNew("ready.json", Info());
            Console.WriteLine("READY OWNED VS PID=" + _pid);
            Task.Run((Action)Serve);
            using (var timer = new Timer { Interval = 1000 })
            {
                timer.Tick += (_, __) =>
                {
                    if (DateTime.UtcNow >= _expires)
                    {
                        _stopping = true;
                        System.Windows.Forms.Application.ExitThread();
                    }
                };
                timer.Start();
                System.Windows.Forms.Application.Run();
            }
            return 0;
        }
        catch (Exception error)
        {
            Console.Error.WriteLine(error);
            return 1;
        }
        finally
        {
            if (_dte != null)
            {
                try
                {
                    _deadline = DateTime.UtcNow.AddSeconds(20);
                    var documents = _owned ? _dte.Documents.Cast<Document>().ToArray() : Array.Empty<Document>();
                    var safe = _owned && !_dte.UserControl &&
                        (!_dte.Solution.IsOpen || Same(_dte.Solution.FullName, Config("Solution"))) &&
                        documents.All(document => document.Saved || Probes.Contains(document.FullName, StringComparer.OrdinalIgnoreCase));
                    if (safe)
                    {
                        foreach (var document in documents.Where(document => Probes.Contains(document.FullName, StringComparer.OrdinalIgnoreCase)))
                        {
                            document.Close(vsSaveChanges.vsSaveChangesNo);
                        }
                        _dte.Solution.Close(false);
                        _dte.Quit();
                    }
                    WriteNew("cleanup.json", new { Owned = _owned, SafeToQuit = safe, QuitRequested = safe, Pid = _pid });
                }
                catch (COMException error)
                {
                    Console.Error.WriteLine("Owned VS cleanup could not be confirmed: " + error.Message);
                }
                finally
                {
                    Marshal.FinalReleaseComObject(_dte);
                }
            }
            OleMessageFilter ignored;
            CoRegisterMessageFilter(previous, out ignored);
            _dispatch.Dispose();
        }
    }

    private static bool CanOwn(int pid, DateTime started, string executable, bool userControl, HashSet<int> previous,
        DateTime requested, DateTime now, string expected) =>
        pid > 0 && !previous.Contains(pid) && !userControl && Same(executable, expected) &&
        started >= requested.AddSeconds(-2) && started <= now.AddSeconds(2);

    private static int SelfTest()
    {
        var now = new DateTime(2026, 10, 27, 0, 0, 0, DateTimeKind.Utc);
        var exe = Path.Combine(Path.GetTempPath(), "owned-vs.exe");
        var empty = new HashSet<int>();
        var checks = new[]
        {
            CanOwn(12, now, exe, false, empty, now, now, exe),
            !CanOwn(0, now, exe, false, empty, now, now, exe),
            !CanOwn(12, now, exe, false, new HashSet<int> { 12 }, now, now, exe),
            !CanOwn(12, now.AddMinutes(-1), exe, false, empty, now, now, exe),
            !CanOwn(12, now.AddMinutes(1), exe, false, empty, now, now, exe),
            !CanOwn(12, now, exe + ".other", false, empty, now, now, exe),
            !CanOwn(12, now, exe, true, empty, now, now, exe)
        };
        Console.WriteLine(Json.Serialize(new { Passed = checks.All(value => value), Cases = checks.Length, IdeLaunched = false }));
        return checks.All(value => value) ? 0 : 1;
    }

    private static bool Same(string first, string second) =>
        string.Equals(Path.GetFullPath(first), Path.GetFullPath(second), StringComparison.OrdinalIgnoreCase);

    private static void VerifyConfiguration(string path)
    {
        if (Config("Purpose") != "Stage9-owned-vs-host" || !Same(Path.GetDirectoryName(path), Config("Root")) ||
            !Same(Config("Root"), Path.Combine(Config("AnalyzerRoot"), Path.GetFileName(Config("Root")))) ||
            Convert.ToInt32(_config["LifetimeSeconds"]) < 60 || Convert.ToInt32(_config["LifetimeSeconds"]) > 3600)
        {
            throw new IOException("Invalid owned VS configuration.");
        }
        var owner = Json.Deserialize<Dictionary<string, object>>(File.ReadAllText(Path.Combine(Config("AnalyzerRoot"), "owner.json")));
        if ((string)owner["Purpose"] != "Stage9-owned-analyzer-inputs" || !Same((string)owner["Project"], Config("Project")) ||
            (string)owner["PackageSha256"] != Config("PackageSha256") || (string)owner["Unity"] != Config("Unity") ||
            !Same(Path.GetDirectoryName(Config("Solution")), Config("Project")) || Hash(Config("Solution")) != Config("SolutionSha256"))
        {
            throw new IOException("VS solution and Analyzer ownership differ.");
        }
        if (File.Exists(Path.Combine(Config("Root"), "ready.json")) || File.Exists(Path.Combine(Config("Root"), "cleanup.json")))
        {
            throw new IOException("Never reuse an IDE process session.");
        }
    }

    private static string Hash(string path)
    {
        using (var sha = SHA256.Create())
        using (var input = File.OpenRead(path))
        {
            return BitConverter.ToString(sha.ComputeHash(input)).Replace("-", "").ToLowerInvariant();
        }
    }

    private static void WriteNew(string name, object value)
    {
        using (var output = new StreamWriter(new FileStream(Path.Combine(Config("Root"), name), FileMode.CreateNew)))
        {
            output.Write(Json.Serialize(value));
        }
    }

    private static object Info() => new
    {
        Pid = _pid, StartTimeUtc = Process.GetProcessById((int)_pid).StartTime.ToUniversalTime().ToString("O"),
        Version = _dte.Version, Executable = _dte.FullName, UserControl = _dte.UserControl,
        Solution = _dte.Solution.FullName, IsOpen = _dte.Solution.IsOpen, ProjectRoot = Config("Project"),
        Owned = _owned, HostPid = Process.GetCurrentProcess().Id, BuildInvokedByHarness = false
    };

    private static void Serve()
    {
        while (!_stopping)
        {
            var security = new PipeSecurity();
            security.AddAccessRule(new PipeAccessRule(WindowsIdentity.GetCurrent().User, PipeAccessRights.FullControl, AccessControlType.Allow));
            using (var pipe = new NamedPipeServerStream(Config("Pipe"), PipeDirection.InOut, 1, PipeTransmissionMode.Byte, PipeOptions.None, 65536, 65536, security))
            {
                pipe.WaitForConnection();
                using (var reader = new StreamReader(pipe))
                using (var writer = new StreamWriter(pipe) { AutoFlush = true })
                {
                    var request = reader.ReadLine();
                    var completion = new TaskCompletionSource<string>();
                    _dispatch.BeginInvoke((Action)(() =>
                    {
                        _deadline = DateTime.UtcNow.AddSeconds(90);
                        try
                        {
                            var data = Json.Deserialize<Dictionary<string, object>>(request);
                            if (!data.ContainsKey("nonce") || (string)data["nonce"] != Config("Nonce"))
                            {
                                throw new InvalidOperationException("The request is not bound to this owned host.");
                            }
                            completion.SetResult(Json.Serialize(new { Ok = true, Info = Info(), Result = Execute(data) }));
                        }
                        catch (Exception error)
                        {
                            completion.SetResult(Json.Serialize(new { Ok = false, Error = error.ToString() }));
                        }
                    }));
                    writer.WriteLine(completion.Task.GetAwaiter().GetResult());
                }
            }
            if (_stopping)
            {
                _dispatch.BeginInvoke((Action)System.Windows.Forms.Application.ExitThread);
            }
        }
    }

    private static object Execute(Dictionary<string, object> request)
    {
        if (!_owned || _dte.UserControl)
        {
            throw new InvalidOperationException("VS process ownership changed; no further action is permitted.");
        }
        var action = (string)request["action"];
        if (action == "info") { return Info(); }
        if (action == "load")
        {
            if (_dte.Solution.IsOpen && !Same(_dte.Solution.FullName, Config("Solution")))
            {
                throw new IOException("Never replace another solution.");
            }
            if (!_dte.Solution.IsOpen) { _dte.Solution.Open(Config("Solution")); }
            return Info();
        }
        if (action == "snapshot") { return Snapshot(); }
        if (action == "stop") { _stopping = true; return new { Stopping = true, OwnedPid = _pid }; }
        if (action == "open" || action == "buffer" || action == "reset")
        {
            if (!_dte.Solution.IsOpen || !Same(_dte.Solution.FullName, Config("Solution")))
            {
                throw new IOException("Only the declared generated solution may be edited.");
            }
            var path = Path.GetFullPath(Path.Combine(Config("Project"), (string)request["file"]));
            if (!Probes.Contains(path, StringComparer.OrdinalIgnoreCase)) { throw new IOException("Only the two owned probes may be opened or edited."); }
            foreach (var document in _dte.Documents.Cast<Document>().Where(document => Same(document.FullName, path)).ToArray())
            {
                if (action == "reset") { document.Close(vsSaveChanges.vsSaveChangesNo); }
            }
            var window = _dte.ItemOperations.OpenFile(path, Constants.vsViewKindTextView);
            var doc = window.Document;
            if (doc.ProjectItem == null || doc.ProjectItem.ContainingProject == null || doc.ProjectItem.ContainingProject.Name == "Miscellaneous Files")
            {
                throw new InvalidOperationException("Probe is not attached to a loaded C# project.");
            }
            if (action == "buffer")
            {
                var text = (TextDocument)doc.Object("TextDocument");
                var edit = text.StartPoint.CreateEditPoint();
                edit.Delete(text.EndPoint);
                edit.Insert((string)request["content"]);
            }
            return new { File = doc.FullName, Project = doc.ProjectItem.ContainingProject.UniqueName, Saved = doc.Saved,
                Lines = ((TextDocument)doc.Object("TextDocument")).EndPoint.Line };
        }
        throw new InvalidOperationException("Unsupported owned host action.");
    }

    private static object Snapshot()
    {
        var rows = new List<object>();
        var items = _dte.ToolWindows.ErrorList.ErrorItems;
        for (var index = 1; index <= items.Count; index++)
        {
            var item = items.Item(index);
            if (string.IsNullOrEmpty(item.FileName) || !Probes.Contains(Path.GetFullPath(item.FileName), StringComparer.OrdinalIgnoreCase)) { continue; }
            rows.Add(new { File = item.FileName, Project = item.Project, Line = item.Line, Column = item.Column,
                Level = (int)item.ErrorLevel, Message = item.Description });
        }
        var docs = _dte.Documents.Cast<Document>().Select(document => new { File = document.FullName,
            Saved = document.Saved, Project = document.ProjectItem?.ContainingProject?.UniqueName }).ToArray();
        return new { Info = Info(), ErrorItems = rows, Documents = docs, CapturedUtc = DateTime.UtcNow.ToString("O"),
            DiagnosticIdFieldAvailable = false, AnalysisCompletionKnown = false };
    }

    [DllImport("user32.dll")] private static extern uint GetWindowThreadProcessId(IntPtr hwnd, out uint processId);
    [DllImport("ole32.dll")] private static extern int CoRegisterMessageFilter(OleMessageFilter value, out OleMessageFilter previous);
    [ComImport, Guid("00000016-0000-0000-C000-000000000046"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
    private interface OleMessageFilter
    {
        [PreserveSig] int HandleInComingCall(int callType, IntPtr caller, int tickCount, IntPtr info);
        [PreserveSig] int RetryRejectedCall(IntPtr callee, int tickCount, int rejectType);
        [PreserveSig] int MessagePending(IntPtr callee, int tickCount, int pendingType);
    }
    private sealed class MessageFilter : OleMessageFilter
    {
        public int HandleInComingCall(int callType, IntPtr caller, int tickCount, IntPtr info) => 0;
        public int RetryRejectedCall(IntPtr callee, int tickCount, int rejectType) => rejectType == 2 && DateTime.UtcNow < _deadline ? 200 : -1;
        public int MessagePending(IntPtr callee, int tickCount, int pendingType) => 2;
    }
}
