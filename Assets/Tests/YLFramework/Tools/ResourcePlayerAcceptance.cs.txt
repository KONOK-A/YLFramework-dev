// 仅由独立Player验收载体编译。消费正式API，不复制加载器或事务实现。
public sealed class ResourcePlayerAcceptance : MonoBehaviour
{
#if ENABLE_IL2CPP && !UNITY_EDITOR
    private const string Backend = "IL2CPP";
#else
    private const string Backend = "Mono2x";
#endif
    [System.Runtime.InteropServices.DllImport("kernel32.dll", ExactSpelling = true)]
    private static extern uint GetCurrentProcessId();
    [System.Runtime.InteropServices.DllImport("kernel32.dll", CharSet = System.Runtime.InteropServices.CharSet.Unicode, ExactSpelling = true)]
    private static extern IntPtr GetModuleHandleW(string name);
    [System.Runtime.InteropServices.DllImport("kernel32.dll", CharSet = System.Runtime.InteropServices.CharSet.Unicode, ExactSpelling = true)]
    private static extern uint GetModuleFileNameW(IntPtr module, System.Text.StringBuilder path, uint size);
    private static int ProcessId => checked((int)GetCurrentProcessId());
    private static bool AcceptanceTestsCompiled
    {
        get
        {
#if YL_RESOURCE_ACCEPTANCE_TESTS
            return true;
#else
            return false;
#endif
        }
    }
    private static bool CandidateCompiled
    {
        get
        {
#if YL_RESOURCE_IL2CPP_CANDIDATE
            return true;
#else
            return false;
#endif
        }
    }
    private JObject _config;
    private JObject _result;
    private readonly Dictionary<string, string> _args = new(StringComparer.Ordinal);
    private ResourceStartup _startup;
    private ResourceSystem _resources;
    private ResourceConfiguration _configuration;
    private string _output;
    private string _store;
    private string _command;
    private string _mode;

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        var host = new GameObject("Resource 8.10 Acceptance");
        DontDestroyOnLoad(host);
        host.AddComponent<ResourcePlayerAcceptance>();
    }

    private IEnumerator Start()
    {
        Application.runInBackground = true;
        var cli = Environment.GetCommandLineArgs();
        for (var i = 0; i + 1 < cli.Length; i++)
        {
            if (cli[i].StartsWith("--r810-", StringComparison.Ordinal))
            {
                _args[cli[i].Substring(7)] = cli[++i];
            }
        }

        _output = _args["out"];
        _store = _args["store"];
        _command = _args["command"];
        _mode = Get("mode", "none");
        _result = new JObject
        {
            ["Pid"] = ProcessId,
            ["Unity"] = Application.unityVersion,
            ["Platform"] = Application.platform.ToString(),
#if ENABLE_IL2CPP && !UNITY_EDITOR
            ["RuntimeAssembly"] = typeof(ResourceSystem).Assembly.FullName,
            ["RuntimeAssemblyIdentityKind"] = "AssemblyName-NativeBuildIdentityInBuildReport",
#else
            ["RuntimeAssembly"] = typeof(ResourceSystem).Assembly.ManifestModule.ModuleVersionId.ToString(),
#endif
            ["BuildGuid"] = Application.buildGUID,
            ["Development"] = Debug.isDebugBuild,
            ["Backend"] = Backend,
            ["LoopbackPinCompiled"] = typeof(ResourceRemoteSource).GetMethod("TrustLoopbackCertificate", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic) != null,
            ["AcceptanceTestsCompiled"] = AcceptanceTestsCompiled,
            ["CandidateCompiled"] = CandidateCompiled,
            // 只记录音频设备上下文；本验收未播放 Clip，不能据此声称 DSP 已播放。
            ["AudioPlayback"] = "ClipLoadedOnly",
            ["Command"] = _command,
            ["Mode"] = _mode,
            ["Store"] = _store,
            ["Graphics"] = SystemInfo.graphicsDeviceType.ToString(),
            ["GraphicsDevice"] = SystemInfo.graphicsDeviceName,
            ["GraphicsVersion"] = SystemInfo.graphicsDeviceVersion,
            ["Screen"] = new JObject { ["Width"] = Screen.width, ["Height"] = Screen.height },
            ["AudioContext"] = new JObject
            {
                ["SampleRate"] = AudioSettings.outputSampleRate,
                ["SpeakerMode"] = AudioSettings.speakerMode.ToString(),
                ["DriverCapabilities"] = AudioSettings.driverCapabilities.ToString(),
                ["DspTime"] = AudioSettings.dspTime
            },
            ["Steps"] = new JArray()
        };
        _config = JObject.Parse(File.ReadAllText(Path.Combine(Application.streamingAssetsPath, "r810-config.json")));
        var stack = new Stack<IEnumerator>();
        stack.Push(Run());
        Exception failure = null;
        while (stack.Count != 0)
        {
            object current = null;
            bool more = false;
            try
            {
                more = stack.Peek().MoveNext();
                if (more)
                {
                    current = stack.Peek().Current;
                }
            }
            catch (Exception error)
            {
                failure = error;
                break;
            }
            if (!more)
            {
                (stack.Pop() as IDisposable)?.Dispose();
                continue;
            }
            if (current is IEnumerator child && current is not CustomYieldInstruction)
            {
                stack.Push(child);
            }
            else
            {
                yield return current;
            }
        }
        while (stack.Count != 0)
        {
            try
            {
                (stack.Pop() as IDisposable)?.Dispose();
            }
            catch (Exception cleanup)
            {
                failure = failure == null ? cleanup : new AggregateException(failure, cleanup);
            }
        }

        if (failure != null)
        {
            _result["Failure"] = failure.ToString();
            Debug.LogException(failure);
        }
        _result["Passed"] = failure == null;
        _result["EndedUtc"] = DateTime.UtcNow.ToString("O");
        Save();
        Application.Quit(failure == null ? 0 : 1);
    }

    private IEnumerator Run()
    {
        var expectedUnity = (string)_config["UnityVersion"] ?? "2022.3.62f3c1";
        var expectedUgui = (string)_config["UguiVersion"] ?? "1.0.0";
        Require((expectedUnity is "2022.3.62f3c1" or "6000.3.27f1") &&
            (expectedUgui is "1.0.0" or "2.0.0") &&
            (expectedUnity == "6000.3.27f1") == (expectedUgui == "2.0.0") &&
            Application.unityVersion == expectedUnity && Application.platform == RuntimePlatform.WindowsPlayer && IntPtr.Size == 8,
            "The actual Player must be the expected Unity / Windows64.");
        Require(Get("backend", Backend) == Backend, "Actual compiled backend differs from the requested backend.");
        _result["BeforeStoreExists"] = Directory.Exists(_store);
        _result["BeforeDisk"] = Disk();
        var projects = Selected().ToArray();
        var mounts = projects.Select(project =>
        {
            var id = (string)project["Id"];
            var builtin = Snapshot(project, 1);
            ResourceRemoteSource remote = null;
            if (_command.StartsWith("remote", StringComparison.Ordinal) || Get("remote", "false") == "true")
            {
                var key = _config["SigningKey"];
                remote = new ResourceRemoteSource(Get("endpoint", (string)_config["Endpoint"]) + id + "/", "acceptance",
                    new[] { new ResourceSigningKey("r810-test", (string)key["Modulus"], (string)key["Exponent"]) },
                    timeoutSeconds: 15, idleTimeoutSeconds: 5, attempts: 2);
                // 测试专用编译边界内的精确loopback pin，正常发布编译中不存在此方法。
                if (Get("trust", "pin") is "pin" or "wrong")
                {
                    var pin = typeof(ResourceRemoteSource).GetMethod("TrustLoopbackCertificate", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic);
                    Require(pin != null, "The isolated certificate test surface must be compiled explicitly.");
                    var certificate = Convert.FromBase64String((string)_config["Certificate"]);
                    if (Get("trust", "pin") == "wrong")
                    {
                        certificate[certificate.Length - 1] ^= 1;
                    }

                    pin.Invoke(remote, new object[] { certificate });
                }
            }
            return new ResourceProjectMount(id, "StandaloneWindows64", "r810-app", builtin, remote: remote);
        }).ToArray();
        ResourceDecryptionOptions decryption = _mode == "encrypted" ? new ResourceDecryptionOptions(new Keys(Get("key", "valid")), 128L * 1024 * 1024) : null;
        _configuration = ResourceConfiguration.Create(ResourceLoadMode.Bundles, mounts, _store, decryption: decryption);
        _startup = new ResourceStartup(_configuration, bindCore: true, audioProjects: new[] { mounts[0].ProjectId });
        using (var start = _startup.StartAsync())
        {
            yield return start;
            if (Get("key", "valid") is "missing" or "wrong")
            {
                var expected = Get("key", "valid") == "missing" ? ResourceError.ContentKeyUnavailable : ResourceError.ContentAuthenticationFailed;
                Require(start.Error?.Kind == expected, "Missing/wrong key must be rejected without fallback: " + start.Error);
                Step("key-rejected", new
                {
                    Error = start.Error.Kind.ToString()
                });
                yield return Stop();
                yield break;
            }
            start.TakeResult();
        }
        _resources = _startup.Resources;
        _result["Mounted"] = Mounted();
        if (_args.TryGetValue("expect", out var expectedVersion))
        {
            AssertVersion(int.Parse(expectedVersion));
        }

        if (_command == "offline")
        {
            Require(!(bool)_result["BeforeStoreExists"], "Offline first launch requires a missing Store.");
            yield return Consume();
            Require(!Directory.Exists(_store), "Built-in startup alone must not create a Store.");
        }
        else if (_command == "probe")
        {
            yield return Consume();
        }
        else if (_command == "rollback")
        {
            using var rollback = _resources.RollbackAsync();
            yield return rollback;
            rollback.TakeResult();
            AssertVersion(int.Parse(Get("target", "1")));
            yield return Consume();
        }
        else if (_command == "activate")
        {
            var prepared = _resources.GetPreparedUpdates().Single();
            yield return Activate(prepared);
        }
        else if (_command.StartsWith("remote", StringComparison.Ordinal))
        {
            yield return Remote();
        }
        else
        {
            var target = int.Parse(Get("target", "2"));
            UpdatePlan plan;
            using (var check = _resources.CheckUpdatesAsync(projects.Select(x => Snapshot(x, target))))
            {
                yield return check;
                plan = check.TakeResult();
            }
            var before = Mounted().ToString(Formatting.None);
            if (_command == "cancel")
            {
                using var canceled = new CancellationTokenSource();
                canceled.Cancel();
                using var prepare = _resources.PrepareUpdateAsync(plan, canceled.Token);
                yield return prepare;
                Require(prepare.Error?.Kind == ResourceError.Canceled, "Canceled Prepare must be Canceled.");
            }
            else
            {
                PreparedUpdate prepared;
                using (var prepare = _resources.PrepareUpdateAsync(plan))
                {
                    yield return prepare;
                    prepared = prepare.TakeResult();
                }
                Require(before == Mounted().ToString(Formatting.None), "Prepare must not activate.");
                Step("prepared", new
                {
                    Count = _resources.GetPreparedUpdates().Count
                });
                if (_command != "prepare")
                {
                    yield return Activate(prepared);
                }
            }
        }
        _result["AfterMounted"] = Mounted();
        _result["AfterDisk"] = Disk();
        yield return Stop();
    }

    private IEnumerator Activate(PreparedUpdate prepared)
    {
        if (_command == "busy")
        {
            using var load = _resources.AcquireAsync<TextAsset>(new ResourceAddress((string)Selected().First()["Id"], "message"));
            yield return load;
            using (var held = load.TakeResult())
            {
                using var blocked = _resources.ActivatePreparedUpdateAsync(prepared);
                yield return blocked;
                Require(blocked.Error?.Kind == ResourceError.Busy, "A live lease must block the whole transaction.");
                using var old = _resources.AcquireAsync<TextAsset>(held.Address);
                yield return old;
                using var oldLease = old.TakeResult();
                Require(oldLease.Asset == held.Asset, "Busy must reopen old acquisition.");
                Step("busy-keeps-old", null);
            }
        }
        if (_args.TryGetValue("checkpoint", out var checkpoint))
        {
            var field = typeof(ArtifactStore).GetField("Checkpoint", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic);
            field.SetValue(_configuration.ArtifactStore, (Action<string>)(point =>
            {
                if (point != checkpoint)
                {
                    return;
                }

                _result["Checkpoint"] = point;
                _result["CheckpointDisk"] = Disk();
                _result["CheckpointMounted"] = Mounted();
                Save();
                if (Get("fault", "kill") == "throw")
                {
                    throw new IOException("r810 owned checkpoint exception");
                }

                File.WriteAllText(_output + ".checkpoint", ProcessId.ToString());
                using var gate = new ManualResetEvent(false);
                gate.WaitOne(); // 外部控制器只强停此已登记的自有Player。
            }));
        }
        using (var activate = _resources.ActivatePreparedUpdateAsync(prepared))
        {
            yield return activate;
            if (Get("fault", "") == "throw")
            {
                Require(activate.Error != null, "A pre-decision injected exception must not claim target activation.");
                AssertVersion(int.Parse(Get("expect", "1")));
                Step("injected-exception-recovered", new
                {
                    Error = activate.Error.ToString()
                });
                yield break;
            }
            activate.TakeResult();
        }
        AssertVersion(int.Parse(Get("target", "2")));
        if (_args.ContainsKey("checkpoint"))
        {
            typeof(ArtifactStore).GetField("Checkpoint", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic).SetValue(_configuration.ArtifactStore, null);
        }

        using (var repeat = _resources.ActivatePreparedUpdateAsync(prepared))
        {
            yield return repeat;
            repeat.TakeResult();
        }
        Step("activate-idempotent", null);
        yield return Consume();
    }

    private IEnumerator Remote()
    {
        var before = Mounted().ToString(Formatting.None);
        using var check = _resources.CheckRemoteUpdatesAsync();
        yield return check;
        if (_command is "remote-reject" or "remote-original-gate")
        {
            Require(check.Error != null, "The negative release must be rejected.");
            if (_args.TryGetValue("error", out var expectedError))
            {
                Require(check.Error.Kind.ToString() == expectedError, "Remote rejection kind mismatch: " + check.Error);
            }

            if (_args.TryGetValue("reason", out var expectedReason))
            {
                Require(check.Error.ToString().Contains(expectedReason), "Remote rejection reason mismatch: " + check.Error);
            }

            if (_command == "remote-original-gate")
            {
                Require(check.Error.Kind == ResourceError.Unsupported, "The original backend must explicitly reject Player.");
            }

            _result["Rejection"] = check.Error.ToString();
            Require(before == Mounted().ToString(Formatting.None), "Remote check failure must keep old memory.");
            yield return Consume();
            yield break;
        }
        var plan = check.TakeResult();
        using var prepare = _resources.PrepareUpdateAsync(plan);
        if (_command is "remote-partial" or "remote-kill-partial")
        {
            var until = Time.realtimeSinceStartup + 60;
            string part = null;
            while (!prepare.IsDone)
            {
                part = Directory.Exists(_store) ? Directory.GetFiles(_store, "*.part", SearchOption.AllDirectories)
                    .FirstOrDefault(x => new FileInfo(x).Length >= 1024 * 1024 && new FileInfo(x).Length < plan.MissingBytes) : null;
                if (part != null)
                {
                    break;
                }

                Require(Time.realtimeSinceStartup < until, "A real download partial was not produced.");
                yield return null;
            }
            Require(part != null && !prepare.IsDone, "Partial must be an incomplete real GET, not manufactured bytes.");
            _result["PartialAtCancel"] = new JObject { ["Path"] = part, ["Length"] = new FileInfo(part).Length };
            if (_command == "remote-kill-partial")
            {
                _result["CheckpointDisk"] = Disk();
                Save();
                File.WriteAllText(_output + ".checkpoint", ProcessId.ToString());
                using var gate = new ManualResetEvent(false);
                gate.WaitOne();
            }
            prepare.Cancel();
            yield return prepare;
            Require(prepare.Error?.Kind == ResourceError.Canceled, "Partial cancellation must be Canceled.");
            Require(before == Mounted().ToString(Formatting.None), "Partial must not activate.");
            Step("actual-partial-canceled", null);
            yield break;
        }
        yield return prepare;
        if (_command == "remote-corrupt")
        {
            Require(prepare.Error != null, "Corrupt body must not be prepared.");
            if (_args.TryGetValue("error", out var expectedError))
            {
                Require(prepare.Error.Kind.ToString() == expectedError, "Transfer rejection kind mismatch: " + prepare.Error);
            }

            _result["Rejection"] = prepare.Error.ToString();
            Require(before == Mounted().ToString(Formatting.None), "Corrupt transfer must keep old memory.");
            yield return Consume();
            yield break;
        }
        var prepared = prepare.TakeResult();
        if (_command == "remote-prepare")
        {
            Step("remote-prepared-without-activation", new
            {
                Count = _resources.GetPreparedUpdates().Count
            });
            yield break;
        }
        yield return Activate(prepared);
    }

    private IEnumerator Consume()
    {
        foreach (var project in Selected())
        {
            var id = (string)project["Id"];
            var digest = _resources.GetActiveCatalogDigest(id);
            var version = Enumerable.Range(1, 2).Single(x => (string)project["Digests"][x - 1] == digest);
            using (var read = _startup.CoreProvider.ReadTextAsync(id, "message"))
            {
                yield return read;
                Require(read.TakeResult() == "r810-version-" + version, "Core must consume the selected exact generation.");
            }
            using (var prefab = _resources.AcquireAsync<GameObject>(new ResourceAddress(id, "prefab")))
            using (var other = _resources.AcquireAsync<GameObject>(new ResourceAddress(id, "other")))
            {
                yield return prefab;
                yield return other;
                using var first = prefab.TakeResult();
                using var second = other.TakeResult();
                var a = first.Asset.GetComponent<Renderer>().sharedMaterial;
                var b = second.Asset.GetComponent<Renderer>().sharedMaterial;
                Require(a && a == b && a.mainTexture, "Prefab shared material and texture must come from the same Bundle graph.");
                using var request = _startup.CoreProvider.AcquirePrefabAsync(id, "prefab");
                yield return request;
                using var core = request.TakeResult();
                using var instance = core.Instantiate(null, SceneManager.GetActiveScene());
                Require(instance.Instance && instance.Instance.GetComponent<Renderer>().sharedMaterial == a, "Core instance must share the official Prefab graph.");
                instance.Discard();
            }
            using (var atlas = _resources.AcquireAsync<SpriteAtlas>(new ResourceAddress(id, "atlas")))
            {
                yield return atlas;
                using var lease = atlas.TakeResult();
                var clone = lease.Asset.GetSprite("icon");
                Require(clone && clone.texture && clone.rect.width == 16, "The packaged Atlas must materialize its Sprite and shared Texture.");
                Destroy(clone);
                yield return null;
            }
            using (var audio = _resources.AcquireAsync<AudioClip>(new ResourceAddress(id, "audio")))
            {
                yield return audio;
                using var lease = audio.TakeResult();
                Require(lease.Asset.samples > 0 && lease.Asset.frequency == 22050, "Packaged Audio must have real samples.");
                if (id == (string)Selected().First()["Id"])
                {
                    using var clip = _startup.AudioProviders[id].Load("audio");
                    Require(clip.Clip == lease.Asset, "Audio provider must share the same resource graph.");
                }
            }
            using (var scene = _resources.LoadSceneAsync(new ResourceAddress(id, "scene"), LoadSceneMode.Additive))
            {
                yield return scene;
                using var lease = scene.TakeResult();
                Require(lease.Scene.isLoaded && lease.Scene.GetRootGameObjects().Any(x => x.name == "r810-scene"), "Packaged Scene must actually load.");
                using var unload = _resources.UnloadSceneAsync(lease);
                yield return unload;
                unload.TakeResult();
            }
            yield return null;
            Step("consume", new
            {
                Project = id,
                Version = version,
                Digest = digest,
                Content = _resources.GetContentDiagnosticsSnapshot()
            });
        }
    }

    private IEnumerator Stop()
    {
        if (_startup == null)
        {
            yield break;
        }

        if (_resources != null)
        {
            _resources.ClearInstanceCaches();
            var until = Time.realtimeSinceStartup + 30;
            while (_resources.GetDiagnosticsReport().InstanceCaches.Count != 0 || _resources.GetContentDiagnosticsSnapshot().WorkCount != 0)
            {
                Require(Time.realtimeSinceStartup < until, "Owned responsibilities did not drain.");
                yield return null;
            }
            _result["BeforeStop"] = JObject.FromObject(_resources.GetDiagnosticsReport());
        }
        using (var stop = _startup.StopAsync())
        {
            yield return stop;
            stop.TakeResult();
        }
        yield return null;
        var content = _resources?.GetContentDiagnosticsSnapshot() ?? default;
        _result["AfterStopContent"] = JObject.FromObject(content);
        if (_resources != null)
        {
            var terminal = _resources.GetDiagnosticsReport();
            _result["AfterStopDiagnostics"] = JObject.FromObject(terminal);
            Require(terminal.Summary.LeaseCount == 0 && terminal.Summary.ProducerCount == 0 && terminal.Summary.WaiterCount == 0 &&
                terminal.Summary.RunningBundleCount == 0 && terminal.Summary.QueuedBundleCount == 0 &&
                terminal.Summary.NativeBundleLoadCount == terminal.Summary.NativeBundleUnloadCount &&
                terminal.Assets.Count == 0 && terminal.Bundles.Count == 0 && terminal.InstanceCaches.Count == 0 && terminal.Scenes.Count == 0,
                "Terminal shutdown must drain real leases, graphs, scenes and native work.");
        }
        var cng = typeof(ResourceSystem).Assembly.GetType("YLFramework.Resource.CngSegmentAead", true);
        var counters = new JObject();
        foreach (var name in new[] { "ActiveAlgorithms", "ActiveKeys", "ActiveKeyObjectBytes", "ReleaseFailures", "RetainedReleaseOwners" })
        {
            counters[name] = Convert.ToInt64(cng.GetProperty(name, System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Static).GetValue(null));
        }

        _result["AfterStopProcessCng"] = counters;
        Require(counters.Properties().All(x => (long)x.Value == 0), "Process CNG counters must be real zero even after failed startup.");
        var module = GetModuleHandleW("bcrypt.dll");
        if (module != IntPtr.Zero)
        {
            var path = new System.Text.StringBuilder(32768);
            Require(GetModuleFileNameW(module, path, (uint)path.Capacity) > 0, "The loaded CNG dependency must be identifiable.");
            _result["CngNativeModule"] = path.ToString();
        }
        _result["AfterStopBundles"] = AssetBundle.GetAllLoadedAssetBundles().Count();
        Require(ResourceSystem.Current == null && CoreResourceAccess.Provider == null && AssetBundle.GetAllLoadedAssetBundles().Count() == 0,
            "Normal shutdown must return authority, Core binding and native Bundles.");
        Require(content.LiveInputs == 0 && content.WorkCount == 0 && content.AllocatedBufferBytes == 0 && content.ReservedBufferBytes == 0 &&
            content.RetainedCleanupOwners == 0 && content.ProcessCngProviderCount == 0 && content.ProcessCngKeyCount == 0 && content.ProcessCngReleaseFailures == 0,
            "Normal shutdown must return all content, budget, in-flight and CNG responsibility.");
        Step("normal-shutdown-clean", null);
    }

    private IEnumerable<JObject> Selected() => ((JArray)_config[_mode == "encrypted" ? "Encrypted" : "None"]).Cast<JObject>();
    private ResourceLocalSnapshot Snapshot(JObject project, int version) => ResourceLocalSnapshot.OpenBuiltIn((string)project["Id"],
        (string)project["Digests"][version - 1], "StandaloneWindows64", "r810-app");
    private JArray Mounted() => new(Selected().Select(x => new JObject { ["ProjectId"] = (string)x["Id"], ["Digest"] = _resources.GetActiveCatalogDigest((string)x["Id"]) }));
    private void AssertVersion(int version)
    {
        foreach (var project in Selected())
        {
            Require(_resources.GetActiveCatalogDigest((string)project["Id"]) == (string)project["Digests"][version - 1], "No mixed generation is allowed.");
        }
    }
    private JObject Disk() => new(Directory.Exists(_store) ? Directory.GetFiles(_store, "*.json", SearchOption.AllDirectories)
        .Where(x => new[] { "active.json", "last-known-good.json", "commit-journal.json", "release-floors.json" }.Contains(Path.GetFileName(x)))
        .Select(x => new JProperty(x.Substring(_store.Length + 1).Replace('\\', '/'), DiskRecord(x))) : Array.Empty<JProperty>());
    private static JToken DiskRecord(string path)
    {
        var bytes = File.ReadAllBytes(path);
        try
        {
            return JObject.Parse(System.Text.Encoding.UTF8.GetString(bytes));
        }
        catch (JsonException error)
        {
            return new JObject { ["Malformed"] = error.GetType().Name, ["OriginalBase64"] = Convert.ToBase64String(bytes) };
        }
    }
    private string Get(string key, string fallback) => _args.TryGetValue(key, out var value) ? value : fallback;
    private static void Require(bool condition, string message)
    {
        if (!condition)
        {
            throw new InvalidOperationException(message);
        }
    }
    private void Step(string name, object data)
    {
        ((JArray)_result["Steps"]).Add(new JObject { ["Name"] = name, ["Data"] = data == null ? JValue.CreateNull() : JToken.FromObject(data) });
        Debug.Log("R810 " + name);
    }
    private void Save() => File.WriteAllText(_output, _result.ToString(Formatting.Indented));

    private sealed class Keys : IResourceContentKeyProvider
    {
        private readonly string _mode;
        internal Keys(string mode) => this._mode = mode;
        public bool TryCopyKey(string projectId, string keyId, byte[] destination)
        {
            if (_mode == "missing")
            {
                return false;
            }

            var key = Convert.FromBase64String(Environment.GetEnvironmentVariable("YL_RESOURCE_CONTENT_KEY"));
            try
            {
                if (_mode == "wrong")
                {
                    key[0] ^= 1;
                }
                Buffer.BlockCopy(key, 0, destination, 0, 32);
                return true;
            }
            finally
            {
                Array.Clear(key, 0, key.Length);
            }
        }
    }
}
