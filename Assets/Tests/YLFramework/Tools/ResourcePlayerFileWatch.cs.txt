using System;
using System.IO;
using System.Text;
using System.Text.Json;
using System.Threading;

// 仅用于外部控制器的有界目录观察，不声称进程归因或捕获全部瞬时写入。
public sealed class ResourcePlayerFileWatch : IDisposable
{
    private readonly FileSystemWatcher _watcher;
    private readonly StreamWriter _output;
    private readonly object _gate = new();
    private readonly string _run;
    private readonly string _store;
    private int _errors;
    private int _plaintext;
    private int _misses;
    private int _controls;
    private bool _isDisposed;
    public int Errors => Volatile.Read(ref _errors);
    public int Plaintext => Volatile.Read(ref _plaintext);
    public int Misses => Volatile.Read(ref _misses);
    public int ControlEntriesNotProbed => Volatile.Read(ref _controls);
    public ResourcePlayerFileWatch(string root, string run, string store)
    {
        this._run = run + Path.DirectorySeparatorChar;
        this._store = store + Path.DirectorySeparatorChar;
        _output = new StreamWriter(new FileStream(Path.Combine(run, "writes.jsonl"), FileMode.CreateNew, FileAccess.Write, FileShare.Read), new UTF8Encoding(false)) { AutoFlush = true };
        _watcher = new FileSystemWatcher(root) { IncludeSubdirectories = true, InternalBufferSize = 65536, NotifyFilter = NotifyFilters.FileName | NotifyFilters.Size | NotifyFilters.LastWrite };
        _watcher.Created += (_, e) => Observe(e);
        _watcher.Changed += (_, e) => Observe(e);
        _watcher.Renamed += (_, e) => Observe(e);
        _watcher.Error += (_, _) => Interlocked.Increment(ref _errors);
        _watcher.EnableRaisingEvents = true;
    }
    private void Observe(FileSystemEventArgs change)
    {
        var path = change.FullPath;
        if ((!path.StartsWith(_run, StringComparison.OrdinalIgnoreCase) && !path.StartsWith(_store, StringComparison.OrdinalIgnoreCase)) || path.EndsWith("writes.jsonl", StringComparison.Ordinal))
        {
            return;
        }

        string header = null;
        string failure = null;
        // 即使FILE_SHARE_DELETE，打开ReplaceFile的短命旧文件也会造成ERROR_UNABLE_TO_REMOVE_REPLACED。
        // 只记录Store控制JSON/next/系统替换临时槽，不读取；Bundle、partial和普通TEMP仍探测原始头。
        var fileName = Path.GetFileName(path);
        var control = path.StartsWith(_store, StringComparison.OrdinalIgnoreCase) &&
            (fileName.EndsWith(".json", StringComparison.OrdinalIgnoreCase) || fileName.EndsWith(".json.next", StringComparison.OrdinalIgnoreCase) || fileName.Contains(".json~RF", StringComparison.OrdinalIgnoreCase));
        if (control)
        {
            failure = "ControlEntryNotProbed";
            Interlocked.Increment(ref _controls);
        }
        else if (!Directory.Exists(path))
        {
            try
            {
                using var file = new FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite | FileShare.Delete);
                var bytes = new byte[8];
                var count = file.Read(bytes, 0, bytes.Length);
                header = Convert.ToHexString(bytes.AsSpan(0, count));
                if (count < 8)
                {
                    Interlocked.Increment(ref _misses);
                }

                if (count == 8 && Encoding.ASCII.GetString(bytes) == "UnityFS\0")
                {
                    Interlocked.Increment(ref _plaintext);
                }
            }
            catch (Exception error) when (error is IOException || error is UnauthorizedAccessException)
            {
                failure = error.GetType().Name;
                Interlocked.Increment(ref _misses);
            }
        }

        lock (_gate)
        {
            if (!_isDisposed)
            {
                _output.WriteLine(JsonSerializer.Serialize(new
                {
                    Path = path,
                    Kind = change.ChangeType.ToString(),
                    Header = header,
                    Failure = failure,
                    Utc = DateTime.UtcNow
                }));
            }
        }
    }
    public void Dispose()
    {
        if (_isDisposed)
        {
            return;
        }

        _watcher.EnableRaisingEvents = false;
        _watcher.Dispose();
        lock (_gate)
        {
            _isDisposed = true;
            _output.Dispose();
        }
    }
}
