using System;
using System.IO;
using System.Linq;
using System.Security.Cryptography;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;
using UnityEditor;
using UnityEditor.Build.Reporting;
using UnityEditor.Compilation;
using UnityEditor.SceneManagement;
using UnityEngine;

// 仅安装到未导入产品的空白工程；构建和运行均不引用 YL。
public static class Stage9SceneLoopEditor
{
    public static void BuildSceneLoopMono() => Build(ScriptingImplementation.Mono2x);
    public static void BuildSceneLoopIl2Cpp() => Build(ScriptingImplementation.IL2CPP);

    private static void Build(ScriptingImplementation backend)
    {
        var project = Path.GetFullPath(Application.dataPath + "/..");
        var owner = JObject.Parse(File.ReadAllText(Path.Combine(project, "stage9-owner.json")));
        var delivery = Path.GetFullPath((string)owner["Delivery"]);
        var expected = (string)owner["UnityVersion"];
        if ((string)owner["Purpose"] != "Stage9-independent-receiver" || (string)owner["Project"] != project ||
            !project.StartsWith(delivery + Path.DirectorySeparatorChar, StringComparison.Ordinal) ||
            (expected != "2022.3.62f3c1" && expected != "6000.3.27f1") || expected != Application.unityVersion ||
            Directory.Exists("Assets/YLFramework") || EditorApplication.isPlayingOrWillChangePlaymode ||
            EditorUtility.scriptCompilationFailed || EditorUserBuildSettings.activeBuildTarget != BuildTarget.StandaloneWindows64)
        {
            throw new InvalidOperationException("Wrong empty control project or engine state.");
        }

        for (var i = 0; i < EditorSceneManager.sceneCount; i++)
        {
            if (EditorSceneManager.GetSceneAt(i).isDirty)
            {
                throw new InvalidOperationException("Never save an unowned dirty scene.");
            }
        }

        var assemblies = CompilationPipeline.GetAssemblies(AssembliesType.Player);
        if (assemblies.Any(a => !System.Text.RegularExpressions.Regex.IsMatch(a.name,
            @"^(?:Assembly-CSharp|UnityEngine(?:\.[A-Za-z0-9_]+)*|Unity(?:\.[A-Za-z0-9_]+)+)$") ||
            a.name.Contains("Tests") || a.name.Contains("TestRunner")))
        {
            throw new InvalidOperationException("Product or host entered the pure control compilation.");
        }

        var args = Environment.GetCommandLineArgs();
        var run = args[Array.IndexOf(args, "-s9-run") + 1];
        var output = Path.Combine(project, "P", (backend == ScriptingImplementation.Mono2x ? "loop-m-" : "loop-i-") + run);
        var folder = "Assets/Stage9/SceneLoop-" + run;
        if (Directory.Exists(output) || AssetDatabase.IsValidFolder(folder))
        {
            throw new IOException("Never overwrite an existing control build or scene input.");
        }

        AssetDatabase.CreateFolder("Assets/Stage9", "SceneLoop-" + run);
        var scenes = new[] { folder + "/SceneLoopA.unity", folder + "/SceneLoopB.unity" };
        foreach (var path in scenes)
        {
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            var camera = new GameObject("Pure Unity control camera", typeof(Camera));
            camera.GetComponent<Camera>().clearFlags = CameraClearFlags.SolidColor;
            EditorSceneManager.SaveScene(scene, path);
        }

        PlayerSettings.companyName = Path.GetFileName(delivery);
        PlayerSettings.productName = Path.GetFileName(project) + "-SceneLoop-" + backend;
        PlayerSettings.SetScriptingBackend(BuildTargetGroup.Standalone, backend);
        PlayerSettings.SetManagedStrippingLevel(BuildTargetGroup.Standalone,
            backend == ScriptingImplementation.IL2CPP ? ManagedStrippingLevel.Low : ManagedStrippingLevel.Disabled);
        Directory.CreateDirectory(output);
        var report = BuildPipeline.BuildPlayer(new BuildPlayerOptions
        {
            scenes = scenes,
            locationPathName = Path.Combine(output, "Stage9SceneLoop.exe"),
            target = BuildTarget.StandaloneWindows64,
            options = BuildOptions.None
        });
        var receipt = new JObject
        {
            ["Result"] = report.summary.result.ToString(),
            ["Errors"] = report.summary.totalErrors,
            ["Warnings"] = report.summary.totalWarnings,
            ["Unity"] = Application.unityVersion,
            ["Backend"] = backend.ToString(),
            ["Development"] = (report.summary.options & BuildOptions.Development) != 0,
            ["Stripping"] = PlayerSettings.GetManagedStrippingLevel(BuildTargetGroup.Standalone).ToString(),
            ["BuildGuid"] = report.summary.guid.ToString(),
            ["PackageSha256"] = owner["PackageSha256"],
            ["ProductInstalled"] = false,
            ["PlayerAssemblies"] = JArray.FromObject(assemblies.Select(a => new { a.name, a.sourceFiles, a.compiledAssemblyReferences })),
            ["Project"] = project,
            ["Seconds"] = report.summary.totalTime.TotalSeconds,
            ["Messages"] = JArray.FromObject(report.steps.SelectMany(s => s.messages).Where(m => m.type != LogType.Log).Select(m => new { Type = m.type.ToString(), m.content })),
            ["PlayerFiles"] = new JArray(Directory.GetFiles(output, "*", SearchOption.AllDirectories).OrderBy(x => x, StringComparer.Ordinal)
                .Select(x => new JObject { ["Path"] = Path.GetRelativePath(output, x).Replace('\\', '/'), ["Sha256"] = Sha(x), ["Length"] = new FileInfo(x).Length }))
        };
        File.WriteAllText(Path.Combine(output, "BuildReport.json"), receipt.ToString(Formatting.Indented));
        if (report.summary.result != BuildResult.Succeeded || report.summary.totalErrors != 0)
        {
            throw new InvalidOperationException("Pure Unity control build failed.");
        }

        EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        Debug.Log("STAGE9_SCENE_LOOP_BUILD_PASS " + backend);
    }

    private static string Sha(string path)
    {
        using var sha = SHA256.Create();
        using var stream = File.OpenRead(path);
        return BitConverter.ToString(sha.ComputeHash(stream)).Replace("-", "").ToLowerInvariant();
    }
}
