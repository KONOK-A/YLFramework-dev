#if UNITY_EDITOR && YL_DOMAIN_PLAYER_PROBE
using System;
using UnityEditor;
using UnityEngine;
using YLFramework.Buff;
using YLFramework.Mission;
using YLFramework.Skill;

namespace YLFramework.DomainFixture
{
    public static class DomainFixtureAuthoring
    {
        [MenuItem("Tools/YLFramework/Domain Fixture/Create Receiver Assets")]
        public static void Create()
        {
            const string Path = "Assets/Resources/DomainFixture";
            if (AssetDatabase.IsValidFolder(Path))
            {
                throw new InvalidOperationException("Receiver fixture already exists; refusing to overwrite user data.");
            }

            if (!AssetDatabase.IsValidFolder("Assets/Resources"))
            {
                AssetDatabase.CreateFolder("Assets", "Resources");
            }

            AssetDatabase.CreateFolder("Assets/Resources", "DomainFixture");
            var skills = ScriptableObject.CreateInstance<SkillsConfig>();
            skills.name = "Skills";
            var skill = ScriptableObject.CreateInstance<SkillConfigData>();
            skill.id = 100;
            skill.duration = 0.02f;
            skill.cooldownSeconds = 0.02f;
            skill.skillClassFullName = "YLFramework.DomainFixture.ReceiverSkill";
            skill.parent = skills;
            skills.skills.Add(skill);
            AssetDatabase.CreateAsset(skills, Path + "/Skills.asset");
            AssetDatabase.AddObjectToAsset(skill, skills);
            var buffs = ScriptableObject.CreateInstance<BuffsConfig>();
            buffs.name = "Buffs";
            var buff = ScriptableObject.CreateInstance<BuffConfigData>();
            buff.id = 100;
            buff.duration = 0.02f;
            buff.buffClassFullName = "YLFramework.DomainFixture.ReceiverBuff";
            buff.parent = buffs;
            buffs.buffs.Add(buff);
            AssetDatabase.CreateAsset(buffs, Path + "/Buffs.asset");
            AssetDatabase.AddObjectToAsset(buff, buffs);
            var missions = ScriptableObject.CreateInstance<MissionsConfig>();
            missions.name = "Missions";
            var mission = ScriptableObject.CreateInstance<MissionConfigData>();
            mission.id = 100;
            mission.missionTypeClassFullName = "YLFramework.DomainFixture.ReceiverMission";
            mission.conditions.Add(new MissionCondition { paramName = "amount", compareType = CompareType.GreaterEqual, target = 1 });
            mission.rewards.Add(new RewardInfo { id = 1, count = 1 });
            missions.parameters.Add(new MissionParameter { name = "amount", type = MissionParamType.Float });
            mission.parent = missions;
            missions.missions.Add(mission);
            AssetDatabase.CreateAsset(missions, Path + "/Missions.asset");
            AssetDatabase.AddObjectToAsset(mission, missions);
            EditorUtility.SetDirty(skills);
            EditorUtility.SetDirty(buffs);
            EditorUtility.SetDirty(missions);
            AssetDatabase.SaveAssetIfDirty(skills);
            AssetDatabase.SaveAssetIfDirty(buffs);
            AssetDatabase.SaveAssetIfDirty(missions);
        }
    }
}
#endif
