#if YL_DOMAIN_PLAYER_PROBE
using System;
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using YLFramework.Buff;
using YLFramework.Logging;
using YLFramework.Mission;
using YLFramework.RedDots;
using YLFramework.Resource;
using YLFramework.Skill;
using ResourceLocation = YLFramework.Resource.ResourceLocation;

namespace YLFramework.DomainFixture
{
    // 这些类不被new/typeof引用；创建来自配置类名，构造器由本夹具专用link.xml保留。
    public sealed class ReceiverSkill : SkillBase
    {
    }
    public sealed class ReceiverBuff : BuffBase
    {
    }
    public sealed class ReceiverMission : MissionBase
    {
    }

    public sealed class DomainPlayerProbe : MonoBehaviour, IEquippableSkill, IAddableBuff
    {
        public Dictionary<int, SkillBase> Skills { get; } = new();
        public List<BuffBase> Buffs { get; } = new();
        public bool CanRelease() => true;
        public bool CanAdd(int id) => true;
        public void OnAddBuff(BuffBase buff)
        {
        }
        public void OnRemoveBuff(BuffBase buff)
        {
        }
        public static bool? HasPassed
        {
            get; private set;
        }

        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
        private static void Boot() => new GameObject("YL.DomainFixture").AddComponent<DomainPlayerProbe>();

        private IEnumerator Start()
        {
            var original = Debug.unityLogger.logHandler;
            Exception failure = null;
            var routine = Run();
            while (true)
            {
                object current = null;
                bool next = false;
                try
                {
                    next = routine.MoveNext();
                    if (next)
                    {
                        current = routine.Current;
                    }
                }
                catch (Exception error)
                {
                    failure = error;
                }
                if (failure != null || !next)
                {
                    break;
                }

                yield return current;
            }
            try
            {
                SkillManager.CloseHost(this);
                BuffManager.CloseHost(this);
                MissionManager.Shutdown();
            }
            finally
            {
                LoggerManager.Disable();
                ReddotManager.Reset();
                (routine as IDisposable)?.Dispose();
            }
            if (!ReferenceEquals(Debug.unityLogger.logHandler, original))
            {
                failure ??= new InvalidOperationException("Logger handler not restored.");
            }

            HasPassed = failure == null;
            if (failure == null)
            {
                Debug.Log("YL_DOMAIN_PLAYER_PASS");
            }
            else
            {
                Debug.LogException(failure);
            }
#if !UNITY_EDITOR
            Application.Quit(failure == null ? 0 : 1);
#endif
        }

        private IEnumerator Run()
        {
            Require(!LoggerManager.IsEnabled, "Logger must be passive before explicit Enable.");
            var session = LoggerManager.Enable(new LoggerOptions { WriteToFile = true, Directory = System.IO.Path.Combine(Application.temporaryCachePath, "YL.DomainFixture." + Guid.NewGuid().ToString("N")) });
            session.LogFormat(LogType.Log, null, "domain-log");
            session.LogFormat(LogType.Warning, null, "domain-warning");
            session.LogFormat(LogType.Error, null, "domain-error");
            session.LogFormat(LogType.Assert, null, "domain-assert");
            session.LogException(new InvalidOperationException("domain-exception"), null);
            using (ReddotManager.RegisterPath("Domain/Inbox"))
            {
                Require(ReddotManager.ShouldBeDisplayed("Domain") && !ReddotManager.ShouldBeDisplayed("Dom"), "Reddot segment matching failed.");
            }

            var entries = new[]
            {
                new ResourceCatalogEntry("skills", new ResourceAddress("domain-fixture", "skills"), new ResourceLocation(ResourceSourceId.Resources, "DomainFixture/Skills", typeof(SkillsConfig))),
                new ResourceCatalogEntry("buffs", new ResourceAddress("domain-fixture", "buffs"), new ResourceLocation(ResourceSourceId.Resources, "DomainFixture/Buffs", typeof(BuffsConfig))),
                new ResourceCatalogEntry("missions", new ResourceAddress("domain-fixture", "missions"), new ResourceLocation(ResourceSourceId.Resources, "DomainFixture/Missions", typeof(MissionsConfig)))
            };
            using var initialized = ResourceSystem.InitializeAsync(new ResourceConfiguration(new[] { new ResourceCatalog("domain-fixture", entries) }));
            var resources = initialized.TakeResult();
            SkillManager.LoadConfig(resources, new ResourceAddress("domain-fixture", "skills"));
            BuffManager.LoadConfig(resources, new ResourceAddress("domain-fixture", "buffs"));
            MissionManager.LoadConfig(resources, new ResourceAddress("domain-fixture", "missions"));
            var skill = SkillManager.AddSkills(this, 100);
            var buff = BuffManager.AddBuff(this, 100);
            Require(skill.GetType().FullName.EndsWith("ReceiverSkill") && buff.GetType().FullName.EndsWith("ReceiverBuff"), "Reflected subtype was replaced.");
            Require(skill.Release() == ReleaseSkillStatus.Success, "Skill release failed.");
            var mission = MissionManager.GetMission("Missions", 100);
            Require(mission.GetType().FullName.EndsWith("ReceiverMission"), "Mission subtype missing.");
            mission.State = MissionState.InProgress;
            MissionManager.SetFloat("Missions", "amount", 1);
            Require(mission.State == MissionState.Completed, "Mission did not complete.");
            yield return new WaitForSecondsRealtime(0.2f);
            Require(skill.IsCooldown && !skill.IsReleasing, "Skill Update/cooldown did not finish.");
            Require(Buffs.Count == 0, "Timed Buff did not expire.");
            SkillManager.Shutdown();
            BuffManager.Shutdown();
            MissionManager.Shutdown();
            Require(resources.GetDiagnosticsSnapshot().RecordCount == 0, "Domain root leases were not returned.");
            LoggerManager.Disable();
            using (var file = new System.IO.FileStream(session.LogFilePath, System.IO.FileMode.Open, System.IO.FileAccess.ReadWrite, System.IO.FileShare.None))
            {
                Require(file.Length > 0, "Logger file empty/locked.");
            }
        }
        private static void Require(bool condition, string message)
        {
            if (!condition)
            {
                throw new InvalidOperationException(message);
            }
        }
    }
}
#endif
