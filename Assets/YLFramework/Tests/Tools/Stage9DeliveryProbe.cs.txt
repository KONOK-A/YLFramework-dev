    [DllImport("kernel32.dll", ExactSpelling = true)] private static extern uint GetCurrentProcessId();

    private IEnumerator Start()
    {
        Application.runInBackground = true;
        _started = Time.realtimeSinceStartup;
        _output = Argument("--s9-out", null);
        if (_output == null)
        {
            yield break;
        }

        _command = Argument("--s9-command", "smoke");
        _store = Argument("--s9-store", null);
        _expected = int.Parse(Argument("--s9-expect", "1"));
        _result = new JObject
        {
            ["Pid"] = GetCurrentProcessId(),
            ["Unity"] = Application.unityVersion,
            ["Platform"] = Application.platform.ToString(),
            ["Development"] = Debug.isDebugBuild,
            ["Graphics"] = SystemInfo.graphicsDeviceType.ToString(),
            ["Command"] = _command,
            ["Store"] = _store,
            ["BuildGuid"] = Application.buildGUID,
            ["Steps"] = new JArray(),
            ["RuntimeAssembly"] = typeof(ResourceSystem).Assembly.FullName,
#if ENABLE_IL2CPP && !UNITY_EDITOR
            ["Backend"] = "IL2CPP",
#else
            ["Backend"] = "Mono2x",
#endif
#if ENABLE_LEGACY_INPUT_MANAGER
            ["LegacyInput"] = true,
#else
            ["LegacyInput"] = false,
#endif
#if ENABLE_INPUT_SYSTEM
            ["NewInput"] = true
#else
            ["NewInput"] = false
#endif
        };
        Application.logMessageReceived += Observe;
        _config = JObject.Parse(File.ReadAllText(Path.Combine(Application.streamingAssetsPath, "stage9-config.json")));
        var stack = new Stack<IEnumerator>();
        stack.Push(Run());
        Exception failure = null;
        while (stack.Count != 0)
        {
            object current = null;
            bool more = false;
            try
            {
                more = stack.Peek().MoveNext();
                if (more)
                {
                    current = stack.Peek().Current;
                }
            }
            catch (Exception error)
            {
                failure = error;
                break;
            }
            if (!more)
            {
                (stack.Pop() as IDisposable)?.Dispose();
                continue;
            }
            if (current is IEnumerator child && current is not CustomYieldInstruction)
            {
                stack.Push(child);
            }
            else
            {
                yield return current;
            }
        }
        while (stack.Count != 0)
        {
            try
            {
                (stack.Pop() as IDisposable)?.Dispose();
            }
            catch (Exception cleanup)
            {
                failure = failure == null ? cleanup : new AggregateException(failure, cleanup);
            }
        }

        if (failure != null)
        {
            _result["Failure"] = failure.ToString();
            Debug.LogException(failure);
        }
        if (failure == null && _errors.Count != 0)
        {
            failure = new InvalidOperationException("Unexpected Unity errors: " + string.Join("\n", _errors));
            _result["Failure"] = failure.ToString();
        }
        _result["Errors"] = new JArray(_errors);
        _result["BusinessPassed"] = failure == null;
        _result["ExitValidationPending"] = true;
        _result["EndedUtc"] = DateTime.UtcNow.ToString("O");
        File.WriteAllText(_output, _result.ToString(Formatting.Indented));
        _hasEnded = true;
        Debug.Log(failure == null ? "STAGE9_PLAYER_PASS" : "STAGE9_PLAYER_FAILED");
        Application.Quit(failure == null ? 0 : 1);
    }

    private void Update()
    {
        if (_output == null || _hasEnded || Time.realtimeSinceStartup - _started < 150)
        {
            return;
        }

        _hasEnded = true;
        _result["BusinessPassed"] = false;
        _result["ExitValidationPending"] = true;
        _result["Failure"] = "Owned consumer watchdog expired; no gate is claimed.";
        File.WriteAllText(_output, _result.ToString(Formatting.Indented));
        Application.Quit(9);
    }
    private void Observe(string message, string trace, LogType type)
    {
        if (type == LogType.Error || type == LogType.Exception || type == LogType.Assert)
        {
            var text = type + ": " + message + "\n" + trace;
            _errors.Add(text);
            // 保持监听至进程结束，不能让Application.Quit之后的清理错误漏过最终控制器。
            if (_hasEnded && _output != null)
            {
                File.AppendAllText(_output + ".late-errors.log", text + "\n");
            }
        }
    }
    private IEnumerator Run()
    {
        var expectedUnity = (string)_config["UnityVersion"];
        Require((expectedUnity == "2022.3.62f3c1" || expectedUnity == "6000.3.27f1") &&
            Application.unityVersion == expectedUnity && Application.platform == RuntimePlatform.WindowsPlayer && IntPtr.Size == 8 && !Debug.isDebugBuild,
            "Real Windows64 non-Development Player required.");
        var assemblies = AppDomain.CurrentDomain.GetAssemblies().Select(a => a.GetName().Name).ToArray();
        Require(!assemblies.Any(name => !System.Text.RegularExpressions.Regex.IsMatch(name,
            @"^(?:Assembly-CSharp|YLFramework\.Runtime|YLFramework\.ResourceStartup\.Sample|mscorlib|netstandard|System(?:\.[A-Za-z0-9_]+)*|Mono(?:\.[A-Za-z0-9_]+)*|Microsoft\.CSharp|Microsoft\.Win32\.Primitives|UnityEngine(?:\.[A-Za-z0-9_]+)*|Unity(?:\.[A-Za-z0-9_]+)+|Newtonsoft\.Json|BouncyCastle\.Cryptography)$") || name.Contains("Tests") || name.Contains("TestRunner")),
            "Only the explicit native Player product, consumer and platform assembly closure may load.");
        _result["LoadedAssemblies"] = new JArray(assemblies);
        if (_command == "urp")
        {
            yield return null;
            yield return new WaitForEndOfFrame();
            Require(GraphicsSettings.currentRenderPipeline && RenderPipelineManager.currentPipeline != null &&
                GraphicsSettings.currentRenderPipeline.GetType().FullName == "UnityEngine.Rendering.Universal.UniversalRenderPipelineAsset" &&
                RenderPipelineManager.currentPipeline.GetType().FullName == "UnityEngine.Rendering.Universal.UniversalRenderPipeline",
                "Actual URP asset and live pipeline required; a Built-in frame is not URP evidence.");
        }
        if (_command == "scene-loop")
        {
            DontDestroyOnLoad(gameObject);
        }

        yield return Foundation();
        if (_command == "sample")
        {
            yield return Sample();
            yield return FinalCleanup();
            yield break;
        }
        var encrypted = _command == "encrypted" || _command == "missing-key";
        var selected = ((JArray)_config["Projects"]).Cast<JObject>().Single(p => (bool)p["Encrypted"] == encrypted);
        var id = (string)selected["Id"];
        var mount = new ResourceProjectMount(id, "StandaloneWindows64", "stage9-app", Snapshot(selected, 1), new[] { typeof(Stage9DeliveryPanel) });
        var configuration = ResourceConfiguration.Create(ResourceLoadMode.Bundles, new[] { mount }, _store,
            decryption: encrypted ? new ResourceDecryptionOptions(new Stage9DeliveryKeys(_command == "missing-key")) : null);
        _startup = new ResourceStartup(configuration, bindCore: true, audioProjects: new[] { id });
        var storeExisted = Directory.Exists(_store);
        using (var start = _startup.StartAsync())
        {
            yield return start;
            if (_command == "missing-key")
            {
                Require(start.Error?.Kind == ResourceError.ContentKeyUnavailable, "Missing content key must reject without fallback: " + start.Error);
                Step("missing-key-rejected", new
                {
                    Error = start.Error.Kind.ToString()
                });
                using var stop = _startup.StopAsync();
                yield return stop;
                stop.TakeResult();
                Require(ResourceSystem.Current == null && CoreResourceAccess.Provider == null, "Rejected startup must release authority and Core binding.");
                yield return FinalCleanup();
                yield break;
            }
            start.TakeResult();
        }
        _resources = _startup.Resources;
        Require(_resources.GetActiveCatalogDigest(id) == (string)selected["Digests"][_expected - 1], "Restart selected the wrong committed version.");
        Step("package-first-start", new
        {
            Project = id,
            Version = _expected,
            StoreExisted = storeExisted,
            Digest = _resources.GetActiveCatalogDigest(id)
        });
        if (_command == "scene-loop")
        {
            yield return SceneLoop(id);
            yield return StopResources();
            yield return FinalCleanup();
            yield break;
        }

        if (_command == "update")
        {
            using var check = _resources.CheckUpdatesAsync(new[] { Snapshot(selected, 2) });
            yield return check;
            var plan = check.TakeResult();
            using var prepare = _resources.PrepareUpdateAsync(plan);
            yield return prepare;
            var prepared = prepare.TakeResult();
            Require(_resources.GetActiveCatalogDigest(id) == (string)selected["Digests"][0], "Prepare must not activate.");
            using var activate = _resources.ActivatePreparedUpdateAsync(prepared);
            yield return activate;
            activate.TakeResult();
            _expected = 2;
            Step("update-activated", new
            {
                Digest = _resources.GetActiveCatalogDigest(id)
            });
        }
        else if (_command == "rollback")
        {
            using var rollback = _resources.RollbackAsync();
            yield return rollback;
            rollback.TakeResult();
            _expected = 1;
            Step("rollback-activated", new
            {
                Digest = _resources.GetActiveCatalogDigest(id)
            });
        }
        Require(_resources.GetActiveCatalogDigest(id) == (string)selected["Digests"][_expected - 1], "Actual active version mismatch.");
        using (var read = _startup.CoreProvider.ReadTextAsync(id, "message"))
        {
            yield return read;
            Require(read.TakeResult() == "stage9-version-" + _expected, "Core text did not consume the selected version.");
        }
        using (var request = _resources.AcquireAsync<SpriteAtlas>(new ResourceAddress(id, "atlas")))
        {
            yield return request;
            using var lease = request.TakeResult();
            var sprite = lease.Asset.GetSprite("icon");
            Require(sprite && sprite.texture && sprite.rect.width == 16, "Newly generated automatic Atlas did not materialize.");
            if (_command == "urp")
            {
                yield return UrpVisual(sprite, id);
            }
            Destroy(sprite);
        }
        if (_command == "urp")
        {
            yield return StopResources();
            yield return FinalCleanup();
            yield break;
        }
        using (var sound = AudioKit.PlaySoundAsync(new AudioClipAddress(_startup.AudioProviders[id], "tone"), new AudioSoundOptions { Loop = true }))
        {
            var until = Time.realtimeSinceStartup + 10;
            while (!sound.HasStarted && !sound.IsDone && Time.realtimeSinceStartup < until)
            {
                yield return null;
            }

            Require(sound.HasStarted, "Resource-backed Audio did not start: " + sound.Error);
            yield return new WaitForSecondsRealtime(.15f);
            var sources = FindObjectsOfType<AudioSource>().Where(s => s.isPlaying).ToArray();
            Require(sources.Length > 0, "Real native audio Source not playing.");
            var samples = new float[256];
            sources[0].GetOutputData(samples, 0);
            Step("audio-resource-playback", new
            {
                Sources = sources.Length,
                Peak = samples.Max(v => Math.Abs(v)),
                Clip = sources[0].clip.name,
                SourceVolume = sources[0].volume,
                sources[0].mute,
                ListenerVolume = AudioListener.volume,
                ListenerPaused = AudioListener.pause,
                AudioSettings.dspTime
            });
        }
        yield return UserInterface(id);
        yield return StopResources();
        SaveRoundTrip();
        if (_command == "smoke")
        {
            Require(!storeExisted && !Directory.Exists(_store), "Built-in first launch must not rely on or manufacture a persistent Store.");
        }

        yield return FinalCleanup();
    }

    private IEnumerator UrpVisual(Sprite sprite, string project)
    {
        Require(urpMaterial && urpMaterial.shader && urpMaterial.shader.isSupported &&
            urpMaterial.shader.name == "Universal Render Pipeline/Unlit", "A serialized supported URP material is required.");
        _result["UrpVisual"] = new JObject
        {
            ["Asset"] = GraphicsSettings.currentRenderPipeline.GetType().FullName,
            ["Pipeline"] = RenderPipelineManager.currentPipeline.GetType().FullName,
            ["NormalEndOfFrame"] = true,
            ["CameraRenderCalled"] = false,
            ["InputInjected"] = false,
            ["Passed"] = false,
            ["Samples"] = new JArray()
        };
        var camera = FindObjectsOfType<Camera>().Single(item => item.enabled && item.gameObject.activeInHierarchy);
        var previousPosition = camera.transform.position;
        var previousRotation = camera.transform.rotation;
        var previousOrthographic = camera.orthographic;
        var previousSize = camera.orthographicSize;
        var previousFlags = camera.clearFlags;
        var previousBackground = camera.backgroundColor;
        var quad = GameObject.CreatePrimitive(PrimitiveType.Quad);
        var material = new Material(urpMaterial);
        try
        {
            camera.transform.SetPositionAndRotation(new Vector3(0, 0, -10), Quaternion.identity);
            camera.orthographic = true;
            camera.orthographicSize = 4;
            camera.clearFlags = CameraClearFlags.SolidColor;
            camera.backgroundColor = Color.black;
            quad.name = "Owned URP resource Atlas display";
            quad.transform.position = new Vector3(-3, 2, 0);
            quad.transform.localScale = Vector3.one;
            var uv = sprite.uv;
            material.SetTexture("_BaseMap", sprite.texture);
            material.SetTextureOffset("_BaseMap", new Vector2(uv.Min(item => item.x), uv.Min(item => item.y)));
            material.SetTextureScale("_BaseMap", new Vector2(uv.Max(item => item.x) - uv.Min(item => item.x), uv.Max(item => item.y) - uv.Min(item => item.y)));
            quad.GetComponent<MeshRenderer>().sharedMaterial = material;
            var worldPoint = (Vector2)camera.WorldToScreenPoint(quad.transform.position);
            var atlasColor = new Color(.1f, .8f, .5f, 1);
            yield return UrpPixels("resource-atlas", new[] { worldPoint }, atlasColor);

            _full = Instantiate(fullTemplate);
            yield return null;
            var root = _full.GetComponentInChildren<UIRootContext>(true);
            Require(root && root.GetLayer(4), "Formal complete UI root and layers required.");
            _module = new Stage9DeliveryModule { ProjectName = project, PanelRoot = root };
            yield return ModuleManager.StartUpModule(_module);
            var color = _module.LoadPanel<ColorSelectorWindow>(colorPrefab, ColorSelectorWindow.Arguments(Color.cyan));
            Require(color && color.CurrentColor == Color.cyan, "Formal ColorSelector bindings required.");
            yield return new WaitForSecondsRealtime(.25f);
            Canvas.ForceUpdateCanvases();
            var selector = color.GetComponentInChildren<ColorSelector>(true);
            Require(selector, "Formal ColorSelector graphic missing.");
            var rect = (RectTransform)selector.transform;
            var canvas = selector.GetComponentInParent<Canvas>();
            var local = rect.rect.center + Vector2.right * (.44f * Mathf.Min(rect.rect.width, rect.rect.height));
            var uiPoint = RectTransformUtility.WorldToScreenPoint(canvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : canvas.worldCamera, rect.TransformPoint(local));
            yield return UrpPixels("formal-color-ring", new[] { uiPoint }, Color.red);
            color.Close();
            yield return null;

            var finished = false;
            ActionKit.ScreenTransition.FadeIn().Color(Color.blue).Duration(.1f).StartGlobal(() => finished = true);
            var deadline = Time.realtimeSinceStartup + 10;
            while (!finished && Time.realtimeSinceStartup < deadline)
            {
                yield return null;
            }
            Require(finished, "Formal screen fade-in action did not complete.");
            yield return UrpPixels("formal-screen-fade-in", new[]
            {
                new Vector2(Screen.width * .5f, Screen.height * .5f), new Vector2(10, 10), new Vector2(Screen.width - 11, Screen.height - 11)
            }, Color.blue);
            finished = false;
            ActionKit.ScreenTransition.FadeOut().Color(Color.blue).Duration(.1f).StartGlobal(() => finished = true);
            deadline = Time.realtimeSinceStartup + 10;
            while (!finished && Time.realtimeSinceStartup < deadline)
            {
                yield return null;
            }
            Require(finished, "Formal screen fade-out action did not complete.");
            yield return UrpPixels("formal-screen-fade-out", new[] { worldPoint }, atlasColor);
            _module.Close();
            Require(_module.State == ModuleState.Closed && ModuleManager.AllModules.Count == 0, "URP consumer must release its Module.");
            _result["UrpVisual"]["Passed"] = true;
        }
        finally
        {
            Destroy(quad);
            Destroy(material);
            camera.transform.SetPositionAndRotation(previousPosition, previousRotation);
            camera.orthographic = previousOrthographic;
            camera.orthographicSize = previousSize;
            camera.clearFlags = previousFlags;
            camera.backgroundColor = previousBackground;
        }
    }

    private IEnumerator UrpPixels(string name, Vector2[] points, Color expected)
    {
        yield return null;
        yield return new WaitForEndOfFrame();
        var frame = ScreenCapture.CaptureScreenshotAsTexture();
        Require(frame && frame.width == Screen.width && frame.height == Screen.height, "Actual normal-frame backbuffer capture required.");
        try
        {
            var path = Path.Combine(Path.GetDirectoryName(_output), name + ".png");
            Require(!File.Exists(path), "Never replace an earlier URP frame.");
            File.WriteAllBytes(path, frame.EncodeToPNG());
            var pixels = new JArray();
            var maximum = 0f;
            foreach (var point in points)
            {
                var x = Mathf.FloorToInt(point.x);
                var y = Mathf.FloorToInt(point.y);
                Require(x >= 1 && y >= 1 && x < frame.width - 1 && y < frame.height - 1, "Expected visual target must lie inside the captured frame.");
                var actual = frame.GetPixel(x, y);
                maximum = Mathf.Max(maximum, Mathf.Abs(actual.r - expected.r), Mathf.Abs(actual.g - expected.g), Mathf.Abs(actual.b - expected.b));
                pixels.Add(new JObject { ["X"] = x, ["Y"] = y, ["R"] = actual.r, ["G"] = actual.g, ["B"] = actual.b });
            }
            using var sha = SHA256.Create();
            var row = new JObject
            {
                ["Name"] = name, ["Frame"] = Time.frameCount, ["Width"] = frame.width, ["Height"] = frame.height,
                ["Expected"] = new JArray(expected.r, expected.g, expected.b), ["Pixels"] = pixels,
                ["MaximumChannelError"] = maximum, ["Passed"] = maximum <= .08f, ["Png"] = name + ".png",
                ["Sha256"] = BitConverter.ToString(sha.ComputeHash(File.ReadAllBytes(path))).Replace("-", "").ToLowerInvariant()
            };
            ((JArray)_result["UrpVisual"]["Samples"]).Add(row);
            Require(maximum <= .08f, "URP visible-pixel mismatch: " + row.ToString(Formatting.None));
        }
        finally
        {
            Destroy(frame);
        }
    }

    private IEnumerator SceneLoop(string project)
    {
        _module = new Stage9DeliveryModule { ProjectName = project };
        yield return ModuleManager.StartUpModule(_module);
        var controller = _module.LoadController<SceneController>();
        for (var round = 0; round < 20; round++)
        {
            Step("scene-loop-load-b", new { Round = round });
            using var toB = controller.LoadSceneAsync("scene-b", LoadSceneMode.Single);
            yield return toB;
            var b = toB.TakeResult();
            Require(b.HasSceneSideEffects && b.Scene.isLoaded && b.Scene.GetRootGameObjects().Any(x => x.name == "Stage9 scene B"),
                "SceneController must return the actual B identity.");
            var record = _resources.GetSceneRecords().Single(x => x.IsLoaded && x.Scene == b.Scene);

            Step("scene-loop-return-a", new { Round = round });
            using var toA = _resources.LoadSceneAsync(new ResourceAddress(project, "scene-a"), LoadSceneMode.Additive);
            yield return toA;
            using var a = toA.TakeResult();
            Require(a.IsValid && a.Scene.GetRootGameObjects().Any(x => x.name == "Stage9 scene A") && SceneManager.SetActiveScene(a.Scene),
                "Resource must return and activate the actual A lease.");

            Step("scene-loop-unload-b", new { Round = round });
            using var unload = _resources.UnloadSceneAsync(record);
            yield return unload;
            Require(unload.TakeResult() && !b.Scene.isLoaded && !record.IsLoaded && SceneManager.sceneCount == 1,
                "Explicit Resource unload must settle B's record without consuming A.");
            yield return Resources.UnloadUnusedAssets();
            Step("scene-loop-round-complete", new { Round = round + 1 });
        }

        // 最后一个 Bundle 场景由中立内置场景替换，不能重建带 probe 的 Bootstrap。
        yield return SceneManager.LoadSceneAsync("Stage9Neutral", LoadSceneMode.Single);
        var deadline = Time.realtimeSinceStartup + 10;
        while (_resources.GetSceneRecords().Count != 0)
        {
            Require(Time.realtimeSinceStartup < deadline, "Scene records did not drain after returning to neutral.");
            yield return null;
        }

        _module.Close();
        Require(_module.State == ModuleState.Closed && ModuleManager.AllModules.Count == 0, "Scene module must close cleanly.");
        _result["SceneLoop"] = new JObject { ["CompletedRounds"] = 20, ["UnloadCount"] = 20, ["RemainingSceneRecords"] = 0 };
    }

    private void SaveRoundTrip()
    {
        var runRoot = Path.GetDirectoryName(Path.GetFullPath(_output));
        var saveRoot = Path.Combine(runRoot, "save-roundtrip");
        Require(Directory.Exists(runRoot) && !Directory.Exists(saveRoot) && !File.Exists(saveRoot),
            "Save evidence must use a new directory owned by this Player run.");
        var previousRoot = SaveManager.StorageRoot;
        try
        {
            SaveManager.SetStorageRoot(saveRoot);
            Require(SaveManager.GetSaveItemCount() == 0, "A new owned Save root must start empty.");
            var item = SaveManager.CreateSaveItem();
            Require(SaveManager.RenameSaveItem(item, "namespace-roundtrip"), "Save index rename failed.");
            SaveManager.CurrentSaveId = item.SaveId;
            var original = new Stage9DeliverySaveData { Value = 73, Name = "namespace-roundtrip" };
            SaveManager.SaveObject(original, item);
            SaveManager.Close();
            original.Value = -1;
            original.Name = "memory-only";

            var items = SaveManager.GetAllSaveItems();
            Require(items.Count == 1 && items[0].SaveId == item.SaveId && items[0].DisplayName == "namespace-roundtrip" &&
                SaveManager.CurrentSaveId == item.SaveId, "Save index did not reload its slot metadata from disk.");
            var read = SaveManager.ReadObject<Stage9DeliverySaveData>(nameof(Stage9DeliverySaveData), item.SaveId);
            Require(read.Status == SaveReadStatus.Success && !read.IsCached && read.Value != null &&
                !ReferenceEquals(original, read.Value) && read.Value.Value == 73 && read.Value.Name == "namespace-roundtrip",
                "Save DTO did not reload its persisted values after closing the owner.");
            SaveManager.Close();
            var files = Directory.GetFiles(Path.Combine(saveRoot, "SaveData"), "*.json", SearchOption.AllDirectories);
            Require(files.Length >= 2, "The Save index and object must both exist on disk.");
            Step("save-index-object-roundtrip", new
            {
                AfterCloseReloaded = true,
                OwnedRunDirectory = true,
                SaveRoot = saveRoot,
                item.SaveId,
                read.Value.Value,
                read.Value.Name,
                PersistedFiles = files.Length,
                Files = files.Select(path => path.Substring(saveRoot.Length + 1).Replace('\\', '/')).ToArray()
            });
        }
        finally
        {
            SaveManager.Close();
            SaveManager.SetStorageRoot(previousRoot);
        }
    }

    private IEnumerator Foundation()
    {
        var dictionary = new SerializedDictionary<string, int>();
        dictionary.Add("one", 1);
        Require(dictionary["one"] == 1, "SerializedDictionary read/write.");
        var callbacks = 0;
        var property = new BindableProperty<int>(1);
        var subscription = property.Register(_ => callbacks++);
        property.Value = 2;
        subscription.Unregister();
        property.Value = 3;
        Require(callbacks == 1, "Binding exact subscription release.");
        var signal = new EasyEvent();
        var events = 0;
        var token = signal.Register(() => events++);
        signal.Trigger();
        token.Unregister();
        signal.Trigger();
        Require(events == 1, "Events unsubscribe.");
        using (var pool = new ObjectPool<List<int>>(() => new List<int>(), reset: x => x.Clear()))
        {
            var first = pool.Allocate();
            first.Add(9);
            pool.Recycle(first);
            var again = pool.Allocate();
            Require(ReferenceEquals(first, again) && again.Count == 0, "ObjectPool recycle/reset.");
            pool.Recycle(again);
        }
        var actionDone = false;
        var timerDone = false;

    private IEnumerator Click(RectTransform target)
    {
        Canvas.ForceUpdateCanvases();
        yield return null;
        var canvas = target.GetComponentInParent<Canvas>().rootCanvas;
        var point = RectTransformUtility.WorldToScreenPoint(canvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : canvas.worldCamera, target.TransformPoint(target.rect.center));
        Require(point.x > 0 && point.y > 0 && point.x < Screen.width && point.y < Screen.height, "UGUI target outside the rendered Player client.");
        var hits = new List<RaycastResult>();
        var data = new PointerEventData(EventSystem.current)
        {
            position = point,
            pressPosition = point,
            pointerId = -1,
            button = PointerEventData.InputButton.Left,
            eligibleForClick = true
        };
        EventSystem.current.RaycastAll(data, hits);
        Require(hits.Count != 0, "Actual Canvas raycaster found no UI at the rendered target.");
        data.pointerCurrentRaycast = hits[0];
        data.pointerPressRaycast = hits[0];
        Step("ugui-input-" + (++_clickIndex), new
        {
            ScreenX = point.x,
            ScreenY = point.y,
            Module = EventSystem.current.currentInputModule.GetType().FullName,
            Hits = hits.Select(x => x.gameObject.name).ToArray(),
            Canvas = canvas.renderMode.ToString(),
            Evidence = "Public EventSystem raycast and UGUI pointer handlers, not OS/native physical input"
        });
        ScreenCapture.CaptureScreenshot(Path.Combine(Path.GetDirectoryName(_output), "before-input-" + _clickIndex + ".png"));
        yield return new WaitForEndOfFrame();
        // 只向真实最上层命中对象派发，不跳过Mask命中去直接点击下层Button；不操作桌面全局输入。
        var hit = hits[0].gameObject;
        var pressed = ExecuteEvents.ExecuteHierarchy(hit, data, ExecuteEvents.pointerDownHandler);
        var click = ExecuteEvents.GetEventHandler<IPointerClickHandler>(hit);
        data.pointerPress = pressed;
        try
        {
            if (pressed)
            {
                ExecuteEvents.Execute(pressed, data, ExecuteEvents.pointerUpHandler);
            }

            if (click && click == pressed)
            {
                ExecuteEvents.Execute(click, data, ExecuteEvents.pointerClickHandler);
            }
        }
        finally
        {
            data.pointerPress = null;
        }
        yield return null;
    }

    private IEnumerator StopResources()
    {
        _resources.ClearInstanceCaches();
        using (var stop = _startup.StopAsync())
        {
            yield return stop;
            stop.TakeResult();
        }
        yield return null;
        var diagnostics = _resources.GetDiagnosticsReport();
        var content = _resources.GetContentDiagnosticsSnapshot();
        Require(diagnostics.Summary.LeaseCount == 0 && diagnostics.Assets.Count == 0 && diagnostics.Bundles.Count == 0 && diagnostics.InstanceCaches.Count == 0 && diagnostics.Scenes.Count == 0,
            "All UI/asset/instance resource responsibilities must drain.");
        Require(diagnostics.Summary.NativeBundleLoadCount == diagnostics.Summary.NativeBundleUnloadCount && content.WorkCount == 0 && content.LiveInputs == 0 && content.ProcessCngKeyCount == 0 && content.ProcessCngProviderCount == 0,
            "Real native Bundle and CNG owners must release.");
        Require(ResourceSystem.Current == null && CoreResourceAccess.Provider == null && !AssetBundle.GetAllLoadedAssetBundles().Any(), "Shutdown must release authority, Core binding and native Bundle instances.");
        Step("resource-stop-clean", new
        {
            Diagnostics = diagnostics,
            Content = content
        });
    }

    private IEnumerator FinalCleanup()

    private IEnumerator LoggerCoexistence()
    {
        Require(!LoggerManager.IsEnabled && !FindObjectsOfType<LoggerManager>().Any(), "Logging must not create an implicit session or owner.");
        var native = Debug.unityLogger.logHandler;
        var previous = new Stage9OwnedLogHandler(native);
        var foreign = new Stage9OwnedLogHandler(native);
        var directory = Path.Combine(Path.GetDirectoryName(_output), "yl-logger-files");
        Require(!Directory.Exists(directory) && !File.Exists(directory), "Logger files must use a new run-owned directory.");
        LoggerSession first = null;
        LoggerSession second = null;
        Debug.unityLogger.logHandler = previous;
        try
        {
            first = LoggerManager.Enable(new LoggerOptions { WriteToFile = true, Directory = directory });
            Require(ReferenceEquals(Debug.unityLogger.logHandler, first) && ReferenceEquals(LoggerManager.Enable(), first), "Explicit logging must own exactly one current session.");
            var observed = 0;
            Action<LogType, string> witness = (kind, text) =>
            {
                if (kind == LogType.Log && text == "STAGE9_LOGGER_FILE_WITNESS") observed++;
            };
            first.OnLogEvent += witness;
            try
            {
                Debug.Log("STAGE9_LOGGER_FILE_WITNESS");
            }
            finally
            {
                first.OnLogEvent -= witness;
            }
            Require(observed == 1 && previous.Witnesses == 1, "YL logging must forward the actual Debug call to its event and actual predecessor exactly once.");
            var firstPath = first.LogFilePath;
            LoggerManager.Disable();
            yield return null;
            Require(first.IsDisposed && !LoggerManager.IsEnabled && ReferenceEquals(Debug.unityLogger.logHandler, previous), "Normal session close must dispose and return the previous handler.");
            string firstText;
            using (var file = File.Open(firstPath, FileMode.Open, FileAccess.Read, FileShare.None))
            using (var reader = new StreamReader(file))
            {
                firstText = reader.ReadToEnd();
            }
            Require(firstText.Contains("STAGE9_LOGGER_FILE_WITNESS"), "The first writer must persist and release its file.");
            File.WriteAllText(Path.Combine(Path.GetDirectoryName(_output), "yl-logger-first.log"), firstText);

            second = LoggerManager.Enable(new LoggerOptions { WriteToFile = true, Directory = directory });
            Require(!ReferenceEquals(first, second) && ReferenceEquals(Debug.unityLogger.logHandler, second), "Reinitialization must not reuse a disposed session.");
            Debug.Log("STAGE9_LOGGER_SECOND_WRITER_WITNESS");
            Require(previous.Witnesses == 2, "The new session must still forward to the actual predecessor.");
            var secondPath = second.LogFilePath;
            Debug.unityLogger.logHandler = foreign;
            LoggerManager.Disable();
            yield return null;
            Require(second.IsDisposed && !LoggerManager.IsEnabled && ReferenceEquals(Debug.unityLogger.logHandler, foreign), "Session close must not overwrite a later handler owner.");
            Debug.Log("STAGE9_LOGGER_FOREIGN_WITNESS");
            Require(foreign.Witnesses == 1, "Later handler ownership not usable.");
            string secondText;
            using (var file = File.Open(secondPath, FileMode.Open, FileAccess.Read, FileShare.None))
            using (var reader = new StreamReader(file))
            {
                secondText = reader.ReadToEnd();
            }
            Require(secondText.Contains("STAGE9_LOGGER_SECOND_WRITER_WITNESS"), "The second writer must persist and release its own file.");
            File.WriteAllText(Path.Combine(Path.GetDirectoryName(_output), "yl-logger-second.log"), secondText);
            Require(!FindObjectsOfType<LoggerManager>().Any(), "Normal logging cleanup must not respawn a host.");
            Step("yl-logger-owner-writer-coexistence", new
            {
                EventWitnesses = observed,
                PreviousHandlerWitnesses = previous.Witnesses,
                LaterOwnerWitnesses = foreign.Witnesses,
                FileExclusiveAfterBothClosures = true,
                Scope = "Explicit YL sessions, native predecessor forwarding and later-owner preservation."
            });
        }
        finally
        {
            if (first != null && ReferenceEquals(LoggerManager.Current, first) || second != null && ReferenceEquals(LoggerManager.Current, second)) LoggerManager.Disable();
            if (ReferenceEquals(Debug.unityLogger.logHandler, previous) || ReferenceEquals(Debug.unityLogger.logHandler, foreign))
            {
                Debug.unityLogger.logHandler = native;
            }
        }
        Require(ReferenceEquals(Debug.unityLogger.logHandler, native), "The consumer must release only its own handler wrappers.");
    }

    private static string Argument(string name, string fallback)
    {
        var args = Environment.GetCommandLineArgs();
        var index = Array.IndexOf(args, name);
        return index < 0 ? fallback : args[index + 1];
    }
    private static void Require(bool condition, string message)
    {
        if (!condition)
        {
            throw new InvalidOperationException(message);
        }
    }

    {
        ((JArray)_result["Steps"]).Add(new JObject { ["Name"] = name, ["Data"] = value == null ? JValue.CreateNull() : JToken.FromObject(value) });
        Debug.Log("STAGE9 " + name);
    }

public sealed class Stage9OwnedLogHandler : ILogHandler
{
    private readonly ILogHandler _next;
    public int Witnesses;
    public Stage9OwnedLogHandler(ILogHandler next)
    {
        this._next = next;
    }
    public void LogException(Exception exception, UnityEngine.Object context) => _next.LogException(exception, context);
    public void LogFormat(LogType type, UnityEngine.Object context, string format, params object[] args)
    {
        if (string.Format(format, args).StartsWith("STAGE9_LOGGER_", StringComparison.Ordinal))
        {
            Witnesses++;
        }
        _next.LogFormat(type, context, format, args);
    }

public sealed class Stage9DeliveryKeys : IResourceContentKeyProvider
{
    private readonly bool _missing;
    public Stage9DeliveryKeys(bool missing = false)
    {
        this._missing = missing;
    }
    public bool TryCopyKey(string projectId, string keyId, byte[] destination)
    {
        if (_missing || projectId != "stage9-encrypted" || keyId != "stage9-nonproduction")
        {
            return false;
        }
        // 固定的公开非生产夹具输入；不输出、序列化或交付任何生产秘密。
        using var hash = SHA256.Create();
        var key = hash.ComputeHash(Encoding.ASCII.GetBytes("Stage9 public nonproduction acceptance fixture only"));
        try
        {
            Buffer.BlockCopy(key, 0, destination, 0, key.Length);
            return true;
        }
        finally
        {
            Array.Clear(key, 0, key.Length);
        }
    }
