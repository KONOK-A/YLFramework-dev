using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Runtime.InteropServices;
using UnityEngine;
using UnityEngine.SceneManagement;

// 仅调用 Unity API 的双场景对照，不经过 YL SceneController 或 Resource。
public sealed class Stage9SceneLoopProbe : MonoBehaviour
{
    [Serializable]
    private sealed class Evidence
    {
        public bool Passed;
        public int Pid;
        public string Unity;
        public string Backend;
        public bool Development;
        public string BuildGuid;
        public string Graphics;
        public int CompletedRounds;
        public int UnloadCount;
        public string LastOperation;
        public List<string> Errors = new List<string>();
    }

    private Evidence _result;
    private string _output;
    private bool _ended;
    private float _started;

    [DllImport("kernel32.dll", ExactSpelling = true)]
    private static extern uint GetCurrentProcessId();

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.BeforeSceneLoad)]
    private static void Initialize()
    {
        if (Argument("--s9-loop-out") == null)
        {
            return;
        }

        var host = new GameObject("Owned pure Unity scene loop");
        DontDestroyOnLoad(host);
        host.AddComponent<Stage9SceneLoopProbe>();
    }

    private IEnumerator Start()
    {
        _output = Argument("--s9-loop-out");
        if (File.Exists(_output))
        {
            throw new IOException("Never overwrite another scene-loop result.");
        }

        Application.runInBackground = true;
        _started = Time.realtimeSinceStartup;
        _result = new Evidence
        {
            Pid = (int)GetCurrentProcessId(),
            Unity = Application.unityVersion,
            Development = Debug.isDebugBuild,
            BuildGuid = Application.buildGUID,
            Graphics = SystemInfo.graphicsDeviceType.ToString(),
#if ENABLE_IL2CPP
            Backend = "IL2CPP"
#else
            Backend = "Mono2x"
#endif
        };
        Application.logMessageReceived += Observe;
        var run = Run();
        while (true)
        {
            bool more;
            object current;
            try
            {
                more = run.MoveNext();
                current = more ? run.Current : null;
            }
            catch (Exception error)
            {
                _result.Errors.Add(error.ToString());
                End(false);
                more = false;
                current = null;
            }

            if (!more)
            {
                break;
            }

            yield return current;
        }

        if (!_ended)
        {
            End(_result.Errors.Count == 0);
        }
    }

    private IEnumerator Run()
    {
        var expected = Argument("--s9-loop-unity");
        if ((expected != "2022.3.62f3c1" && expected != "6000.3.27f1") ||
            Application.unityVersion != expected || Application.platform != RuntimePlatform.WindowsPlayer ||
            IntPtr.Size != 8 || Debug.isDebugBuild)
        {
            throw new InvalidOperationException("Expected exact Windows64 non-Development control Player.");
        }

        if (AppDomain.CurrentDomain.GetAssemblies().Any(a => !System.Text.RegularExpressions.Regex.IsMatch(a.GetName().Name,
            @"^(?:Assembly-CSharp|mscorlib|netstandard|System(?:\.[A-Za-z0-9_]+)*|Mono(?:\.[A-Za-z0-9_]+)*|Microsoft\.CSharp|Microsoft\.Win32\.Primitives|UnityEngine(?:\.[A-Za-z0-9_]+)*|Unity(?:\.[A-Za-z0-9_]+)+|Newtonsoft\.Json)$") ||
            a.GetName().Name.Contains("Tests") || a.GetName().Name.Contains("TestRunner")))
        {
            throw new InvalidOperationException("Pure Unity control loaded product or host assemblies.");
        }

        for (var round = 0; round < 20; round++)
        {
            Progress(round + ": load B Single");
            yield return SceneManager.LoadSceneAsync("SceneLoopB", LoadSceneMode.Single);
            if (!SceneManager.GetSceneByName("SceneLoopB").isLoaded || SceneManager.GetSceneByName("SceneLoopA").isLoaded)
            {
                throw new InvalidOperationException("Single load did not replace A with B.");
            }

            Progress(round + ": return A Additive");
            yield return SceneManager.LoadSceneAsync("SceneLoopA", LoadSceneMode.Additive);
            var scene = SceneManager.GetSceneByName("SceneLoopA");
            if (!scene.isLoaded || !SceneManager.SetActiveScene(scene))
            {
                throw new InvalidOperationException("Additive return did not activate A.");
            }

            Progress(round + ": unload B");
            yield return SceneManager.UnloadSceneAsync("SceneLoopB");
            if (SceneManager.GetSceneByName("SceneLoopB").isLoaded || SceneManager.sceneCount != 1)
            {
                throw new InvalidOperationException("B remained loaded after explicit unload.");
            }

            _result.UnloadCount++;
            Progress(round + ": unload unused assets");
            yield return Resources.UnloadUnusedAssets();
            _result.CompletedRounds++;
            Progress(round + ": complete");
        }
    }

    private void Progress(string operation)
    {
        _result.LastOperation = operation;
        File.WriteAllText(_output, JsonUtility.ToJson(_result, true));
        Debug.Log("STAGE9_SCENE_LOOP_PROGRESS " + operation);
    }

    private void End(bool passed)
    {
        _ended = true;
        _result.Passed = passed;
        Progress(passed ? "complete" : "failed");
        Debug.Log(passed ? "STAGE9_SCENE_LOOP_PASS" : "STAGE9_SCENE_LOOP_FAILED");
        Application.Quit(passed ? 0 : 1);
    }

    private void Update()
    {
        if (!_ended && _result != null && Time.realtimeSinceStartup - _started > 180)
        {
            _result.Errors.Add("Owned scene-loop watchdog expired.");
            End(false);
        }
    }

    private void Observe(string message, string trace, LogType type)
    {
        if (type == LogType.Error || type == LogType.Exception || type == LogType.Assert)
        {
            _result.Errors.Add(type + ": " + message + "\n" + trace);
            File.WriteAllText(_output, JsonUtility.ToJson(_result, true));
        }
    }

    private static string Argument(string name)
    {
        var args = Environment.GetCommandLineArgs();
        var index = Array.IndexOf(args, name);
        return index >= 0 && index + 1 < args.Length ? args[index + 1] : null;
    }
}
