using System;
using System.IO;
using System.Linq;
using Newtonsoft.Json;
using UnityEditor;
using UnityEditor.Build.Reporting;
using UnityEditor.Compilation;
using UnityEditor.SceneManagement;
using UnityEngine;

namespace YLFramework.DomainFixture
{
    [InitializeOnLoad]
    public static class DomainCarrierBootstrap
    {
        private const string ExitKey = "YL.DomainCarrier.ExitPending";
        private static double _stableAfter;
        static DomainCarrierBootstrap()
        {
            if (SessionState.GetBool(ExitKey, false)) ScheduleExit();
        }
        private static void Owned()
        {
            if (!File.Exists("domain-carrier.json") || !Path.GetFullPath(".").Replace('\\', '/').Contains("/Logs/YLFramework.Domain/Carriers/")) throw new InvalidOperationException("Not a C-owned independent carrier.");
        }
        public static void ConfigureTests()
        {
            Owned();
            EditorSettings.enterPlayModeOptionsEnabled = true;
            EditorSettings.enterPlayModeOptions = EnterPlayModeOptions.DisableDomainReload;
            EditorSettings.serializationMode = SerializationMode.ForceText;
            PlayerSettings.SetApiCompatibilityLevel(BuildTargetGroup.Standalone, ApiCompatibilityLevel.NET_Standard);
            PlayerSettings.SetScriptingDefineSymbolsForGroup(BuildTargetGroup.Standalone,
                PlayerSettings.GetScriptingDefineSymbolsForGroup(BuildTargetGroup.Standalone).Split(';').Where(symbol => symbol != "YL_DOMAIN_PLAYER_PROBE").ToArray());
            SessionState.SetBool(ExitKey, true); ScheduleExit();
        }
        public static void ConfigurePlayer()
        {
            Owned();
            PlayerSettings.SetScriptingDefineSymbolsForGroup(BuildTargetGroup.Standalone, new[] { "YL_DOMAIN_PLAYER_PROBE" });
            PlayerSettings.SetApiCompatibilityLevel(BuildTargetGroup.Standalone, ApiCompatibilityLevel.NET_Standard);
            SessionState.SetBool(ExitKey, true); ScheduleExit();
        }
        public static void ConfigureExcel()
        {
            Owned();
            if (!File.Exists("B-input-commit.txt") || File.ReadAllText("B-input-commit.txt").Trim() != "6a9e2508a42333e55c82e6f29eea136643af4467") throw new InvalidOperationException("Missing pinned B snapshot.");
            PlayerSettings.SetScriptingDefineSymbolsForGroup(BuildTargetGroup.Standalone, new[] { "YL_DOMAIN_EXCEL" });
            SessionState.SetBool(ExitKey, true); ScheduleExit();
        }
        private static void ScheduleExit()
        {
            _stableAfter = EditorApplication.timeSinceStartup + 2;
            EditorApplication.update -= ExitWhenStable; EditorApplication.update += ExitWhenStable;
        }
        private static void ExitWhenStable()
        {
            if (EditorApplication.isCompiling || EditorApplication.isUpdating) { _stableAfter = EditorApplication.timeSinceStartup + 2; return; }
            if (EditorApplication.timeSinceStartup < _stableAfter) return;
            SessionState.EraseBool(ExitKey); EditorApplication.update -= ExitWhenStable; EditorApplication.Exit(0);
        }
        public static void PreparePlayer()
        {
            Owned();
            if (!PlayerSettings.GetScriptingDefineSymbolsForGroup(BuildTargetGroup.Standalone).Split(';').Contains("YL_DOMAIN_PLAYER_PROBE")) throw new InvalidOperationException("Missing explicit consumer symbol.");
            var author = AppDomain.CurrentDomain.GetAssemblies().Select(assembly => assembly.GetType("YLFramework.DomainFixture.DomainFixtureAuthoring")).FirstOrDefault(type => type != null);
            if (author == null) throw new InvalidOperationException("Fixture authoring not compiled.");
            author.GetMethod("Create").Invoke(null, null);
            if (!AssetDatabase.IsValidFolder("Assets/DomainScene")) AssetDatabase.CreateFolder("Assets", "DomainScene");
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            new GameObject("Domain Camera", typeof(Camera), typeof(AudioListener));
            EditorSceneManager.SaveScene(scene, "Assets/DomainScene/Bootstrap.unity");
        }
        public static void BuildMono() => Build(ScriptingImplementation.Mono2x);
        public static void BuildIl2Cpp() => Build(ScriptingImplementation.IL2CPP);
        private static void Build(ScriptingImplementation backend)
        {
            Owned();
            var packages = UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages();
            if (packages.Any(package => !package.name.StartsWith("com.unity.modules.", StringComparison.Ordinal) &&
                !new[] { "com.unity.ugui", "com.unity.nuget.newtonsoft-json", "com.unity.test-framework", "com.unity.ext.nunit" }.Contains(package.name))) throw new InvalidOperationException("Carrier contains a package outside its declared product/test closure.");
            var roots = CompilationPipeline.GetAssemblies(AssembliesType.Player);
            var runtime = roots.Single(assembly => assembly.name == "YLFramework.Runtime");
            var edges = runtime.assemblyReferences.Select(assembly => assembly.name).Concat(runtime.compiledAssemblyReferences.Select(Path.GetFileNameWithoutExtension)).Distinct().ToArray();
            if (edges.Any(name => !System.Text.RegularExpressions.Regex.IsMatch(name,
                @"^(?:mscorlib|netstandard|System(?:\.[A-Za-z0-9_]+)*|UnityEngine(?:\.[A-Za-z0-9_]+)*|Unity\.ugui|Microsoft\.CSharp|Microsoft\.Win32\.Primitives|Newtonsoft\.Json|BouncyCastle\.Cryptography)$") ||
                name.Contains("TestRunner") || name.Contains("Editor"))) throw new InvalidOperationException("Invalid Player reference closure.");
            PlayerSettings.SetScriptingBackend(BuildTargetGroup.Standalone, backend);
            PlayerSettings.SetManagedStrippingLevel(BuildTargetGroup.Standalone, backend == ScriptingImplementation.IL2CPP ? ManagedStrippingLevel.Low : ManagedStrippingLevel.Disabled);
            if (backend == ScriptingImplementation.IL2CPP) PlayerSettings.SetIl2CppCompilerConfiguration(BuildTargetGroup.Standalone, Il2CppCompilerConfiguration.Release);
            var output = Path.GetFullPath("Builds/" + (backend == ScriptingImplementation.IL2CPP ? "IL2CPP" : "Mono"));
            Directory.CreateDirectory(output);
            var report = BuildPipeline.BuildPlayer(new BuildPlayerOptions
            {
                scenes = new[] { "Assets/DomainScene/Bootstrap.unity" }, target = BuildTarget.StandaloneWindows64,
                targetGroup = BuildTargetGroup.Standalone, locationPathName = Path.Combine(output, "DomainProbe.exe"), options = BuildOptions.None
            });
            var record = new
            {
                unity = Application.unityVersion, backend = PlayerSettings.GetScriptingBackend(BuildTargetGroup.Standalone).ToString(),
                stripping = PlayerSettings.GetManagedStrippingLevel(BuildTargetGroup.Standalone).ToString(),
                result = report.summary.result.ToString(), errors = report.summary.totalErrors, warnings = report.summary.totalWarnings,
                options = report.summary.options.ToString(), guid = report.summary.guid.ToString(), platform = report.summary.platform.ToString(),
                runtimeReferences = edges, packages = packages.Select(package => new { package.name, package.version }).ToArray()
            };
            File.WriteAllText(Path.Combine(output, "BuildReport.json"), JsonConvert.SerializeObject(record, Formatting.Indented));
            if (report.summary.result != BuildResult.Succeeded || report.summary.totalErrors != 0 || report.summary.options != BuildOptions.None) throw new InvalidOperationException("Domain Player build failed.");
            Debug.Log("YL_DOMAIN_BUILD_PASS " + backend);
        }
    }
}
