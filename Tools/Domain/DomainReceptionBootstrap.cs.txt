        private static string Full(string path) => Path.GetFullPath(path).TrimEnd('\\', '/');
        private static bool Same(string left, string right) => string.Equals(Full(left), Full(right), StringComparison.OrdinalIgnoreCase);
        private static string Relative(string root, string path)
        {
            var prefix = Full(root) + Path.DirectorySeparatorChar;
            var full = Path.GetFullPath(path);
            if (!full.StartsWith(prefix, StringComparison.OrdinalIgnoreCase)) throw new IOException("Path escapes owned root: " + path);
            return full.Substring(prefix.Length).Replace('\\', '/');
        }
        private static void RejectLink(string path)
        {
            for (var cursor = Path.GetFullPath(path); cursor != null; cursor = Path.GetDirectoryName(cursor))
                if ((File.Exists(cursor) || Directory.Exists(cursor)) && (File.GetAttributes(cursor) & FileAttributes.ReparsePoint) != 0)
                    throw new IOException("Linked path rejected: " + cursor);
        }
        private static string Sha(string path)
        {
            RejectLink(path);
            using var algorithm = SHA256.Create();
            using var file = File.OpenRead(path);
            return BitConverter.ToString(algorithm.ComputeHash(file)).Replace("-", "").ToLowerInvariant();
        }
        private static JObject Read(string path)
        {
            RejectLink(path);
            return JObject.Parse(File.ReadAllText(path, Encoding.UTF8));
        }
        private static void Write(string path, string text)
        {
            RejectLink(path);
            if (!Directory.Exists(Path.GetDirectoryName(path))) throw new DirectoryNotFoundException("Missing explicit output parent.");
            using var file = new FileStream(path, FileMode.CreateNew, FileAccess.Write, FileShare.Read);
            var bytes = new UTF8Encoding(false).GetBytes(text);
            file.Write(bytes, 0, bytes.Length);
        }
        private static void WriteJson(string path, JToken value) => Write(path, value.ToString(Formatting.Indented));
        private static void NewDirectory(string path)
        {
            RejectLink(path);
            if (File.Exists(path) || Directory.Exists(path)) throw new IOException("Preserve existing output: " + path);
            if (!Directory.Exists(Path.GetDirectoryName(path))) throw new DirectoryNotFoundException("Missing explicit output parent.");
            Directory.CreateDirectory(path);
        }
        private static IEnumerable<string> Files(string root)
        {
            RejectLink(root);
            foreach (var path in Directory.GetFiles(root)) { RejectLink(path); yield return path; }
            foreach (var path in Directory.GetDirectories(root))
                foreach (var child in Files(path)) yield return child;
        }
        private static JArray FileRows(string root) => new JArray(Files(root).OrderBy(path => path, StringComparer.Ordinal)
            .Select(path => new JObject { ["Path"] = Relative(root, path), ["Sha256"] = Sha(path), ["Length"] = new FileInfo(path).Length }));

        private static void Identity(JObject value)
        {
            if ((string)value["purpose"] != Purpose || !Same((string)value["projectPath"], Project) ||
                !Same((string)value["reception"], Reception) || !Same((string)value["repository"], Repository) ||
                (string)value["sourceCommit"] != SourceCommit || (string)value["cCommit"] != CCommit)
                throw new InvalidOperationException("Wrong MC26 external receiver identity.");
        }
        private static JObject Gate(string name)
        {
            var path = Path.Combine(Reception, name + ".json");
            var result = Read(path); Identity(result);
            if ((bool?)result["passed"] != true || (string)result["action"] != name) throw new InvalidDataException("Missing actual gate: " + name);
            var run = (string)result["run"];
            if (!Regex.IsMatch(run ?? "", @"^mc26-receiver-[a-z0-9]+(?:-[a-z0-9]+)*$")) throw new InvalidDataException("Unsafe gate run.");
            if (Sha(path) != Sha(Path.Combine(Reception, "Runs", run, name, "result.json"))) throw new InvalidDataException("Gate differs from original step.");
            return result;
        }
        private static Context Owned()
        {
            if (!Same(Path.GetDirectoryName(Application.dataPath), Project) || Pid == 16464 ||
                !Application.isBatchMode || Application.unityVersion != "2022.3.62f3c1")
                throw new InvalidOperationException("Receiver-only batch entry; never invoke on the main Editor.");
            RejectLink(Project); RejectLink(Reception);
            if (Directory.Exists(Path.Combine(Project, ".git")) || File.Exists(Path.Combine(Project, ".git"))) throw new InvalidOperationException("A receiver cannot be a source worktree.");
            var ownerPath = Path.Combine(Reception, "owner.json");
            var owner = Read(ownerPath); Identity(owner);
            if (!Same((string)owner["deliveryRoot"], Delivery) || (string)owner["unityVersion"] != Application.unityVersion ||
                Sha(ownerPath) != Sha(Path.Combine(Project, "domain-reception-owner.json"))) throw new InvalidDataException("External owner binding mismatch.");
            using (var process = System.Diagnostics.Process.GetCurrentProcess())
                if (!Same(process.MainModule.FileName, (string)owner["unityExecutable"])) throw new InvalidOperationException("Wrong actual Unity executable.");
            var arguments = Environment.GetCommandLineArgs();
            var index = Array.IndexOf(arguments, "-mc26-step");
            if (index < 0 || index + 1 >= arguments.Length || Array.LastIndexOf(arguments, "-mc26-step") != index) throw new ArgumentException("One explicit -mc26-step is required.");
            var step = Full(arguments[index + 1]); RejectLink(step);
            var request = Read(Path.Combine(step, "request.json")); Identity(request);
            var run = (string)request["run"]; var action = (string)request["action"]; var backend = (string)request["backend"];
            if (!Regex.IsMatch(run ?? "", @"^mc26-receiver-[a-z0-9]+(?:-[a-z0-9]+)*$") ||
                !new[] { "Prepare", "BuildMono", "BuildIL2CPP" }.Contains(action) ||
                !new[] { "Mono2x", "IL2CPP" }.Contains(backend) ||
                !Same(step, Path.Combine(Reception, "Runs", run, action)) || !Same((string)request["stepPath"], step) ||
                (string)request["ownerSha256"] != Sha(ownerPath)) throw new InvalidDataException("Wrong current invocation identity.");
            if ((action == "BuildMono" && backend != "Mono2x") || (action == "BuildIL2CPP" && backend != "IL2CPP")) throw new InvalidDataException("Backend/action mismatch.");
            if (EditorApplication.isPlayingOrWillChangePlaymode || EditorApplication.isCompiling || EditorApplication.isUpdating ||
                EditorUtility.scriptCompilationFailed || EditorUserBuildSettings.activeBuildTarget != BuildTarget.StandaloneWindows64)
                throw new InvalidOperationException("Receiver must have completed actual Win64 compilation, without Play Mode.");
            for (var i = 0; i < EditorSceneManager.sceneCount; i++)
                if (EditorSceneManager.GetSceneAt(i).isDirty) throw new InvalidOperationException("Never overwrite a dirty scene.");

            var imported = Gate("Import"); var audit = Gate("AuditPackage"); var consumer = Gate("InstallConsumer");
            var package = Path.Combine(Delivery, "YLFramework-C.unitypackage");
            if (!Same((string)request["packagePath"], package) || (bool?)imported["data"]?["helperPresent"] != false ||
                (bool?)imported["data"]?["consumerPresent"] != false || (string)imported["data"]?["packageSha256"] != Sha(package) ||
                (string)audit["data"]?["packageSha256"] != Sha(package) || (string)consumer["data"]?["packageSha256"] != Sha(package))
                throw new InvalidDataException("Product-only first-import/package/consumer binding failed.");
            var filesPath = (string)audit["data"]?["filesPath"];
            if ((string)audit["data"]?["filesSha256"] != Sha(filesPath)) throw new InvalidDataException("Package file manifest changed.");
            var productFiles = JArray.Parse(File.ReadAllText(filesPath));
            foreach (var item in productFiles)
            {
                var relative = (string)item["Path"];
                foreach (var root in new[] { Full(Repository), Project })
                {
                    var path = Path.Combine(root, relative); Relative(root, path);
                    if (root == Project && (bool)item["Folder"] && !File.Exists(path + ".meta") &&
                        !productFiles.Any(other => ((string)other["Path"]).StartsWith(relative + "/", StringComparison.Ordinal))) continue;
                    if (Sha(path + ".meta") != (string)item["MetaSha256"] || (!(bool)item["Folder"] && Sha(path) != (string)item["Sha256"]))
                        throw new InvalidDataException("Current source/receiver product differs from actual formal package: " + path);
                }
            }
            foreach (var item in consumer["data"]["files"])
                if (Sha((string)item["Path"]) != (string)item["Sha256"] || Sha((string)item["Source"]) != (string)item["Sha256"])
                    throw new InvalidDataException("Consumer/bootstrap changed since immutable installation.");
            VerifyPackages();
            var editorAssemblies = CompilationPipeline.GetAssemblies();
            foreach (var entry in new[] {
                new[] { "Assembly-CSharp", Harness + "/DomainAcceptanceProbe.cs" },
                new[] { "Assembly-CSharp-Editor", Harness + "/Editor/DomainAcceptanceAssets.cs" },
                new[] { "Assembly-CSharp-Editor", Harness + "/Editor/DomainReceptionBootstrap.cs" } })
                if (!editorAssemblies.Single(assembly => assembly.name == entry[0]).sourceFiles.Any(path => Same(SourcePath(path), Path.Combine(Project, entry[1]))))
                    throw new InvalidDataException("Consumer is not in the contracted default assembly: " + entry[1]);
            return new Context { Step = step, Run = run, Action = action, Backend = backend, Owner = owner, Audit = audit, Consumer = consumer, ProductFiles = productFiles };
        }

        private static UnityEditor.PackageManager.PackageInfo[] VerifyPackages()
        {
            var packages = UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages();
            foreach (var package in packages)
            {
                if (!package.name.StartsWith("com.unity.modules.", StringComparison.Ordinal) &&
                    package.name != "com.unity.ugui" && package.name != "com.unity.nuget.newtonsoft-json")
                    throw new InvalidDataException("Unexpected receiver package: " + package.name);
                if (package.source.ToString() != "BuiltIn" && package.source.ToString() != "Registry") throw new InvalidDataException("Non-registry/built-in receiver package.");
            }
            if (packages.Single(package => package.name == "com.unity.ugui").version != "1.0.0" ||
                packages.Single(package => package.name == "com.unity.nuget.newtonsoft-json").version != "3.2.1") throw new InvalidDataException("Pinned UPM versions not restored.");
            return packages;
        }
        private static string SourcePath(string path)
        {
            var full = Path.IsPathRooted(path) ? path : Path.Combine(Project, path);
            if (File.Exists(full)) return Full(full);
            var asset = path.Replace('\\', '/');
            if (asset.StartsWith("Packages/", StringComparison.Ordinal))
            {
                var package = UnityEditor.PackageManager.PackageInfo.FindForAssetPath(asset);
                if (package != null) return Full(Path.Combine(package.resolvedPath, asset.Substring(("Packages/" + package.name + "/").Length)));
            }
            throw new FileNotFoundException("Actual Unity compiler source not found.", path);
        }
        private static JArray SourceFiles(Context context)
        {
            var paths = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            foreach (var item in context.ProductFiles)
            {
                var path = Path.Combine(Project, (string)item["Path"]);
                if (!(bool)item["Folder"]) paths.Add(path);
                if (File.Exists(path + ".meta")) paths.Add(path + ".meta");
            }
            foreach (var item in context.Consumer["data"]["files"]) paths.Add((string)item["Path"]);
            foreach (var path in Files(Path.Combine(Project, Harness))) paths.Add(path);
            paths.Add(Path.Combine(Project, "Packages/manifest.json"));
            paths.Add(Path.Combine(Project, "Packages/packages-lock.json"));
            return new JArray(paths.OrderBy(path => path, StringComparer.Ordinal).Select(path => new JObject {
                ["Path"] = Relative(Project, path), ["Sha256"] = Sha(path), ["Length"] = new FileInfo(path).Length }));
        }

        public static void Prepare()
        {
            var context = Owned();
            if (context.Action != "Prepare") throw new InvalidOperationException("Prepare needs its own explicit action.");
            var folder = Harness + "/Generated/" + context.Run;
            if (Directory.Exists(Path.Combine(Project, folder)) || File.Exists(Path.Combine(Project, folder + ".meta"))) throw new IOException("Never overwrite/adopt a prepared fixture.");
            var editorAssembly = AppDomain.CurrentDomain.GetAssemblies().Single(assembly => assembly.GetName().Name == "Assembly-CSharp-Editor");
            var type = editorAssembly.GetType(Creator, true);
            var method = type.GetMethod("Create", BindingFlags.Public | BindingFlags.Static, null,
                new[] { typeof(string), typeof(string), typeof(string), typeof(string), typeof(bool) }, null);
            if (method == null || method.ReturnType != typeof(string)) throw new MissingMethodException("Creator needs public static string Create(string,string,string,string,bool).");
            var company = PlayerSettings.companyName; var product = PlayerSettings.productName;
            var prefix = Prefix(context);
            var scene = (string)method.Invoke(null, new object[] { folder, prefix, context.Run, context.Backend, false });
            if (string.IsNullOrEmpty(scene) || !scene.StartsWith(folder + "/", StringComparison.Ordinal) ||
                !scene.EndsWith(".unity", StringComparison.Ordinal) || !File.Exists(Path.Combine(Project, scene))) throw new InvalidDataException("Creator must return its new dedicated scene.");
            foreach (var root in new[] { "Skills", "Buffs", "Missions" })
                if (!AssetDatabase.LoadMainAssetAtPath(folder + "/Resources/" + prefix + "/" + root + ".asset")) throw new InvalidDataException("Missing contracted resource asset: " + root);
            if (PlayerSettings.companyName != company || PlayerSettings.productName != product) throw new InvalidDataException("Consumer changed company/product.");
            VerifyResourceKeys();
            var active = EditorSceneManager.GetActiveScene();
            if (active.path != scene || active.isDirty) throw new InvalidDataException("Dedicated scene must be saved and active.");
            var probes = active.GetRootGameObjects().SelectMany(root => root.GetComponentsInChildren<MonoBehaviour>(true))
                .Where(component => component && component.GetType().FullName == "YLFramework.Tests.DomainAcceptance.DomainAcceptanceProbe").ToArray();
            if (probes.Length != 1) throw new InvalidDataException("Dedicated scene must contain exactly one explicit acceptance component.");
            var fields = new SerializedObject(probes[0]);
            if (fields.FindProperty("_resourcePrefix").stringValue != prefix || fields.FindProperty("_run").stringValue != context.Run ||
                fields.FindProperty("_expectedBackend").stringValue != context.Backend || fields.FindProperty("_expectLegacyLogger").boolValue)
                throw new InvalidDataException("Actual saved acceptance component has the wrong prefix/run/backend/Logger binding.");
            var receipt = new JObject {
                ["Prepared"] = true, ["Project"] = Project, ["Pid"] = Pid, ["Unity"] = Application.unityVersion,
                ["Run"] = context.Run, ["Backend"] = context.Backend, ["Prefix"] = prefix, ["Folder"] = folder, ["Scene"] = scene,
                ["ExpectLegacyLogger"] = false, ["SourceCommit"] = SourceCommit, ["CCommit"] = CCommit,
                ["PackageSha256"] = context.Audit["data"]["packageSha256"], ["Files"] = FileRows(Path.Combine(Project, folder)),
                ["FolderMetaSha256"] = Sha(Path.Combine(Project, folder + ".meta")), ["SourceFiles"] = SourceFiles(context),
                ["Company"] = company, ["Product"] = product, ["Utc"] = DateTime.UtcNow.ToString("o")
            };
            WriteJson(Path.Combine(context.Step, "prepared.json"), receipt);
        }

        private static void VerifyResourceKeys()
        {
            var keys = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
            foreach (var asset in AssetDatabase.GetAllAssetPaths())
            {
                if (!asset.StartsWith("Assets/", StringComparison.Ordinal) || AssetDatabase.IsValidFolder(asset)) continue;
                var index = asset.IndexOf("/Resources/", StringComparison.Ordinal);
                if (index < 0) continue;
                var key = asset.Substring(index + "/Resources/".Length);
                key = key.Substring(0, key.Length - Path.GetExtension(key).Length);
                if (keys.TryGetValue(key, out var prior)) throw new InvalidDataException("Duplicate Resources key; preserve both fixtures, do not build: " + prior + " / " + asset);
                keys.Add(key, asset);
            }
        }
        private static bool ForbiddenAssembly(string name) => !Regex.IsMatch(name,
            @"^(?:Assembly-CSharp|YLFramework\.Runtime|YLFramework\.ResourceStartup\.Sample|mscorlib|netstandard|System(?:\.[A-Za-z0-9_]+)*|Mono(?:\.[A-Za-z0-9_]+)*|Microsoft\.CSharp|Microsoft\.Win32\.Primitives|UnityEngine(?:\.[A-Za-z0-9_]+)*|Unity(?:\.[A-Za-z0-9_]+)+|Newtonsoft\.Json|BouncyCastle\.Cryptography)$") || Regex.IsMatch(name,
            @"(?i)(?:MCP|MiniExcel|UnityEditor|nunit|TestRunner|(?:^|[.\-])Editor(?:$|[.\-])|(?:^|[.\-])Tests?(?:$|[.\-])|YLFramework\.Analyzers)");
        private static void CheckReferences(string name, IEnumerable<string> references)
        {
            if (ForbiddenAssembly(name)) throw new InvalidDataException("Forbidden Player assembly: " + name);
            foreach (var reference in references)
                if (ForbiddenAssembly(reference) || (name.StartsWith("YLFramework.", StringComparison.Ordinal) && reference.StartsWith("Assembly-CSharp", StringComparison.Ordinal)) ||
                    (name == "YLFramework.Runtime" && !Regex.IsMatch(reference, @"^(?:mscorlib|netstandard|System(?:\.[A-Za-z0-9_]+)*|UnityEngine(?:\.[A-Za-z0-9_]+)*|Unity\.ugui|Microsoft\.CSharp|Microsoft\.Win32\.Primitives|Newtonsoft\.Json|BouncyCastle\.Cryptography)$")))
                    throw new InvalidDataException("Forbidden Player reverse dependency: " + name + " -> " + reference);
        }
        private static JArray PlayerGraph()
        {
            var assemblies = CompilationPipeline.GetAssemblies(AssembliesType.Player);
            if (assemblies.Count(assembly => assembly.name == "YLFramework.Runtime") != 1 || assemblies.Count(assembly => assembly.name == "Assembly-CSharp") != 1)
                throw new InvalidDataException("Actual product/consumer Player compilation inputs are missing.");
            var rows = new JArray();
            foreach (var assembly in assemblies.OrderBy(item => item.name, StringComparer.Ordinal))
            {
                var references = assembly.assemblyReferences.Select(item => item.name).Concat(assembly.compiledAssemblyReferences.Select(Path.GetFileNameWithoutExtension))
                    .Distinct().OrderBy(name => name, StringComparer.Ordinal).ToArray();
                CheckReferences(assembly.name, references);
                rows.Add(new JObject { ["Name"] = assembly.name, ["References"] = new JArray(references),
                    ["SourceFiles"] = new JArray(assembly.sourceFiles.Select(path => new JObject { ["Path"] = path, ["Sha256"] = Sha(SourcePath(path)) })),
                    ["CompiledAssemblyReferences"] = new JArray(assembly.compiledAssemblyReferences) });
            }
            return rows;
        }
        private static JArray ActualPlayerAssemblies(string output, string backend)
        {
            var managed = backend == "Mono2x" ? Path.Combine(output, "DomainAcceptance_Data/Managed") :
                Path.Combine(output, "DomainAcceptance_BackUpThisFolder_ButDontShipItWithYourGame/Managed");
            if (!Directory.Exists(managed)) throw new DirectoryNotFoundException("Actual built Player managed inputs missing: " + managed);
            var rows = new JArray();
            foreach (var path in Files(managed).Where(path => path.EndsWith(".dll", StringComparison.OrdinalIgnoreCase)).OrderBy(path => path, StringComparer.Ordinal))
            {
                // Reflection-only bytes are never executed and do not bind to Editor-loaded Runtime assemblies.
                var assembly = System.Reflection.Assembly.ReflectionOnlyLoad(File.ReadAllBytes(path));
                var name = assembly.GetName().Name;
                var references = assembly.GetReferencedAssemblies().Select(reference => reference.Name).OrderBy(value => value, StringComparer.Ordinal).ToArray();
                CheckReferences(name, references);
                rows.Add(new JObject { ["Name"] = name, ["Path"] = Relative(output, path), ["Sha256"] = Sha(path),
                    ["Mvid"] = assembly.ManifestModule.ModuleVersionId.ToString(), ["References"] = new JArray(references) });
            }
            foreach (var name in new[] { "YLFramework.Runtime", "Assembly-CSharp" })
                if (rows.Count(row => (string)row["Name"] == name) != 1) throw new InvalidDataException("Missing actual built product/consumer assembly: " + name);
            return rows;
        }
        private static void CheckOutputFiles(string output)
        {
            foreach (var file in Files(output))
            {
                var relative = Relative(output, file);
                if (Regex.IsMatch(relative, @"(?i)(?:^|/)(?:Tests?|Integrations|Editor|Packages|PackageCache|\.git|\.env|\.mcp)(?:/|\.|$)|MiniExcel|MCP|nunit|TestRunner|YLFramework\.Editor|Assembly-CSharp-Editor|YLFramework\.Analyzers|(?:secret|credential|private[-_]?key)|\.(?:so|bundle|pfx|p12|pem|key|jks|keystore|unitypackage)$") ||
                    (file.EndsWith(".dll", StringComparison.OrdinalIgnoreCase) && ForbiddenAssembly(Path.GetFileNameWithoutExtension(file)) &&
                     !new[] { "GameAssembly.dll", "UnityPlayer.dll", "MonoPosixHelper.dll", "mono-2.0-bdwgc.dll" }.Contains(Path.GetFileName(file))))
                    throw new InvalidDataException("Forbidden Player output file: " + relative);
            }
        }
        private static JArray CopyLicenses(string output, UnityEditor.PackageManager.PackageInfo[] packages)
        {
            var destination = Path.Combine(output, "Licenses"); NewDirectory(destination);
            var rows = new JArray();
            var product = Path.Combine(Project, "Assets/YLFramework");
            foreach (var file in Files(product).Where(path => !path.EndsWith(".meta", StringComparison.Ordinal) &&
                Regex.IsMatch(Path.GetFileName(path), @"(?i)^(?:.*LICENSE.*|.*NOTICES?.*|SOURCE\.txt)$")))
            {
                var target = Path.Combine(destination, Relative(product, file).Replace('/', '-'));
                File.Copy(file, target, false);
                rows.Add(new JObject { ["Source"] = Relative(Project, file), ["Path"] = Relative(output, target), ["Sha256"] = Sha(target) });
            }
            foreach (var name in new[] { "com.unity.ugui", "com.unity.nuget.newtonsoft-json" })
            {
                var package = packages.Single(item => item.name == name); RejectLink(package.resolvedPath);
                var licenses = Directory.GetFiles(package.resolvedPath).Where(path => !path.EndsWith(".meta", StringComparison.Ordinal) &&
                    Regex.IsMatch(Path.GetFileName(path), @"(?i)LICENSE|NOTICE")).ToArray();
                if (!licenses.Any(path => Regex.IsMatch(Path.GetFileName(path), @"(?i)LICENSE"))) throw new FileNotFoundException("Applicable installed UPM license missing: " + name);
                foreach (var file in licenses)
                {
                    var target = Path.Combine(destination, name + "-" + Path.GetFileName(file)); File.Copy(file, target, false);
                    rows.Add(new JObject { ["Source"] = file, ["Path"] = Relative(output, target), ["Sha256"] = Sha(target), ["Package"] = name, ["Version"] = package.version });
                }
            }
            foreach (var required in new[] { "LICENSE.md", "Runtime-THIRD-PARTY-NOTICES.md", "Plugins-BouncyCastle-LICENSE.md" })
                if (!File.Exists(Path.Combine(destination, required))) throw new FileNotFoundException("Required product license missing.", required);
            return rows;
        }

        public static void Build()
        {
            var context = Owned();
            if (context.Action == "Prepare") throw new InvalidOperationException("Build needs an explicit BuildMono/BuildIL2CPP action.");
            var prepareStep = Path.Combine(Reception, "Runs", context.Run, "Prepare");
            var prepareResult = Read(Path.Combine(prepareStep, "result.json")); Identity(prepareResult);
            var preparedPath = Path.Combine(prepareStep, "prepared.json");
            if ((bool?)prepareResult["passed"] != true || (string)prepareResult["data"]?["proofSha256"] != Sha(preparedPath)) throw new InvalidDataException("Prepare must have normally completed in its own actual Editor process.");
            var prepared = Read(preparedPath);
            var folder = Harness + "/Generated/" + context.Run;
            var scene = (string)prepared["Scene"];
            if ((bool?)prepared["Prepared"] != true || (string)prepared["Run"] != context.Run || (string)prepared["Backend"] != context.Backend ||
                (string)prepared["Folder"] != folder || (string)prepared["Prefix"] != Prefix(context) || (bool?)prepared["ExpectLegacyLogger"] != false ||
                !Same((string)prepared["Project"], Project) || !scene.StartsWith(folder + "/", StringComparison.Ordinal)) throw new InvalidDataException("Prepared scene/run/backend binding mismatch.");
            var fixtureRoot = Path.Combine(Project, folder);
            var current = FileRows(fixtureRoot);
            if (!JToken.DeepEquals(current, prepared["Files"]) || Sha(fixtureRoot + ".meta") != (string)prepared["FolderMetaSha256"]) throw new InvalidDataException("Prepared assets changed; no regeneration in Build.");
            VerifyResourceKeys();
            var packages = VerifyPackages(); var sources = SourceFiles(context);
            var players = Path.Combine(Reception, "Players");
            if (!Directory.Exists(players)) NewDirectory(players);
            var runOutput = Path.Combine(players, context.Run); NewDirectory(runOutput);
            var output = Path.Combine(runOutput, context.Backend); NewDirectory(output);
            var backend = context.Backend == "Mono2x" ? ScriptingImplementation.Mono2x : ScriptingImplementation.IL2CPP;
            var company = PlayerSettings.companyName; var product = PlayerSettings.productName;
            if (company != (string)prepared["Company"] || product != (string)prepared["Product"]) throw new InvalidDataException("Company/product changed since Prepare.");
            var record = new JObject {
                ["Result"] = "NotBuilt", ["Errors"] = 0, ["AuditPassed"] = false, ["Project"] = Project, ["Pid"] = Pid,
                ["Unity"] = Application.unityVersion, ["UnityExecutable"] = context.Owner["unityExecutable"],
                ["Run"] = context.Run, ["Backend"] = context.Backend, ["SourceCommit"] = SourceCommit, ["CCommit"] = CCommit,
                ["PackageSha256"] = context.Audit["data"]["packageSha256"], ["Scene"] = scene, ["Prefix"] = Prefix(context),
                ["PrepareSha256"] = Sha(preparedPath), ["Company"] = company, ["Product"] = product,
                ["Options"] = "None", ["Development"] = false, ["Target"] = "StandaloneWindows64", ["Output"] = output,
                ["SourceFiles"] = sources,
                ["Packages"] = JArray.FromObject(packages.Select(package => new { package.name, package.version, Source = package.source.ToString(), package.packageId, package.resolvedPath,
                    Dependencies = package.dependencies.Select(dependency => new { dependency.name, dependency.version }).ToArray() })),
                ["StartedUtc"] = DateTime.UtcNow.ToString("o")
            };
            try
            {
                PlayerSettings.SetScriptingBackend(BuildTargetGroup.Standalone, backend);
                PlayerSettings.SetManagedStrippingLevel(BuildTargetGroup.Standalone, backend == ScriptingImplementation.IL2CPP ? ManagedStrippingLevel.Low : ManagedStrippingLevel.Disabled);
                if (backend == ScriptingImplementation.IL2CPP) PlayerSettings.SetIl2CppCompilerConfiguration(BuildTargetGroup.Standalone, Il2CppCompilerConfiguration.Release);
                record["Stripping"] = PlayerSettings.GetManagedStrippingLevel(BuildTargetGroup.Standalone).ToString();
                record["Compiler"] = backend == ScriptingImplementation.IL2CPP ? PlayerSettings.GetIl2CppCompilerConfiguration(BuildTargetGroup.Standalone).ToString() : "Mono";
                record["PlayerCompilationGraph"] = PlayerGraph();
                var report = BuildPipeline.BuildPlayer(new BuildPlayerOptions {
                    scenes = new[] { scene }, target = BuildTarget.StandaloneWindows64, targetGroup = BuildTargetGroup.Standalone,
                    locationPathName = Path.Combine(output, "DomainAcceptance.exe"), options = BuildOptions.None
                });
                record["Result"] = report.summary.result.ToString(); record["Errors"] = report.summary.totalErrors;
                record["Warnings"] = report.summary.totalWarnings; record["BuildGuid"] = report.summary.guid.ToString();
                record["Options"] = report.summary.options.ToString(); record["Seconds"] = report.summary.totalTime.TotalSeconds;
                record["Messages"] = JArray.FromObject(report.steps.SelectMany(step => step.messages).Where(message => message.type != LogType.Log)
                    .Select(message => new { Type = message.type.ToString(), message.content }));
                if (report.summary.result != BuildResult.Succeeded || report.summary.totalErrors != 0 || report.summary.options != BuildOptions.None ||
                    report.summary.platform != BuildTarget.StandaloneWindows64 || PlayerSettings.GetScriptingBackend(BuildTargetGroup.Standalone) != backend ||
                    PlayerSettings.GetManagedStrippingLevel(BuildTargetGroup.Standalone) != (backend == ScriptingImplementation.IL2CPP ? ManagedStrippingLevel.Low : ManagedStrippingLevel.Disabled) ||
                    (backend == ScriptingImplementation.IL2CPP && PlayerSettings.GetIl2CppCompilerConfiguration(BuildTargetGroup.Standalone) != Il2CppCompilerConfiguration.Release))
                    throw new InvalidOperationException("Actual non-Development Windows64 BuildPlayer failed.");
                record["ActualPlayerAssemblies"] = ActualPlayerAssemblies(output, context.Backend);
                if (backend == ScriptingImplementation.IL2CPP)
                {
                    foreach (var native in new[] { "GameAssembly.dll", "UnityPlayer.dll", "DomainAcceptance.exe", "DomainAcceptance_Data/il2cpp_data/Metadata/global-metadata.dat" })
                        if (!File.Exists(Path.Combine(output, native))) throw new FileNotFoundException("Actual IL2CPP native artifact missing.", native);
                }
                record["Licenses"] = CopyLicenses(output, packages);
                CheckOutputFiles(output);
                if (PlayerSettings.companyName != company || PlayerSettings.productName != product) throw new InvalidDataException("Company/product changed during Build.");
                if (!JToken.DeepEquals(sources, SourceFiles(context))) throw new InvalidDataException("Frozen source/assets changed during Build.");
                record["PlayerFiles"] = FileRows(output);
                record["AuditPassed"] = true;
            }
            catch (Exception error)
            {
                record["AuditPassed"] = false; record["Failure"] = error.ToString();
                record["RetainedFiles"] = FileRows(output);
                throw;
            }
            finally
            {
                record["EndedUtc"] = DateTime.UtcNow.ToString("o");
                var reportPath = Path.Combine(output, "BuildReport.json"); WriteJson(reportPath, record);
                Write(Path.Combine(output, "BuildReport.sha256"), Sha(reportPath) + "  BuildReport.json\n");
                WriteJson(Path.Combine(context.Step, "BuildReport.json"), record);
                Write(Path.Combine(context.Step, "BuildReport.sha256"), Sha(reportPath) + "  BuildReport.json\n");
            }
        }
    }
